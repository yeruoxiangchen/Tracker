# Objectron GT 中心模拟第二阶段采集：Shoe + Camera
# ==================================================
#
# 目的：
#   已知 Objectron 官方物体中心后，从每个完整视频片段的全部有效候选帧中，
#   以“相机中心相对物体中心的单位方向”执行训练一致的单 seed 球面 FPS，选 8 帧。
#
# 每个片段输出三份 Mesh：
#   1) 当前 SS30K+SLat30K，pose+mask runtime-O（GT 只用于选帧）；
#   2) 当前 SS30K+SLat30K，Objectron 完整真值 pose O（oracle 上界）；
#   3) strict ReconViaGen，使用完全相同的 8 张 RGB。
# 当前模型的两个 O 分支都会把 Mesh 轮廓投回原始 8 帧。
#
# 注意：这是一项 oracle 选帧定性诊断。GT rotation/scale 不进入第 1 个分支，
# 但 GT translation 进入了选帧，因此不能直接当作可部署结果或正式测试结论。

# T1. 四卡后台启动
# ---------------

cd /home/zjr/Tracker

mkdir -p "${ROOT}/logs"

if [ -s "${ROOT}/report.json" ]; then
  echo "reuse complete result: ${ROOT}/report.json"
elif systemctl --user is-active --quiet "${UNIT}.service"; then
  echo "already active: ${UNIT}.service"
else
  systemctl --user reset-failed "${UNIT}.service" 2>/dev/null || true
  systemd-run --user \
    --unit="${UNIT}" \
    --collect \
    --property=WorkingDirectory=/home/zjr/Tracker \
    --property=StandardOutput=append:${ROOT}/logs/master.log \
    --property=StandardError=append:${ROOT}/logs/master.log \
    /usr/bin/env \
      PROJECT_ROOT=/home/zjr/Tracker \
      PYTHON=/home/zjr/anaconda3/envs/reconviagen/bin/python \
      GT_CENTER_GPUS=0,1,2,3 \
      OUTPUT_ROOT="${ROOT}" \
      /bin/bash \
        pose_point_depth_mv/background_jobs/run_objectron_gt_center_guided8_shoe_camera_4gpu.sh
fi

echo "master log: ${ROOT}/logs/master.log"
echo "final report: ${ROOT}/report.json"
