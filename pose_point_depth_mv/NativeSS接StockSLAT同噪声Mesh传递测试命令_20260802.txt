    'legacy_objaverse': 8,
    'gap_objaverse': 8,
    'pilot_objaverse': 8,
    'omni': 8,
}
assert split['output']['val_manifest_sha256'] == hashlib.sha256(val_path.read_bytes()).hexdigest()
final_by_object = {row['object_uid']: row for row in split['phases']['final']}
t2_indices = [32, 35, 42, 45, 48, 53, 56, 59]
t2_rows = [final_by_object[val['samples'][index]['object_uid']] for index in t2_indices]
t2_sources = Counter(row['source'] for row in t2_rows)
t2_views = Counter(int(row['view_count']) for row in t2_rows)
assert t2_sources == {
    'legacy_objaverse': 2,
    'gap_objaverse': 2,
    'pilot_objaverse': 2,
    'omni': 2,
}, t2_sources
assert t2_views == {2: 3, 4: 3, 8: 2}, t2_views
print({
    'checkpoint_sha256': sha,
    'false_checks': sorted(false_checks),
    'final_objects': 32,
    'T2_sources': dict(t2_sources),
    'T2_views': dict(t2_views),
})
PY
  echo 'C1 PASS'
)
