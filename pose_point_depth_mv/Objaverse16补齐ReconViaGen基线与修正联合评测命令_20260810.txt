# O8：原版 Stable-X/trellis-vggt-v0-2，seed42；使用 O3 中与当前模型完全相同的
# 2/4/8视图RGB和GT mask。ReconViaGen执行官方1.10倍前景裁剪与518x518缩放，
# 使用VGGT，但不传显式相机、点云、GT Mesh或评测指标。

# O9：当前No-VGGT与ReconViaGen都先应用固定decoder轴变换

# 不做ICP、逐对象缩放或GT拟合。

# 旧 O7 被保留但不再作为最终几何数值：旧评测遗漏了固定decoder轴变换。

###############################################################################
# A. 推荐：后台运行O8-O9
###############################################################################

cd /home/zjr/Tracker

OBJAVERSE16_RECON_GPU=${OBJAVERSE16_RECON_GPU:-4}
ROOT=/data/zjr/objaverse16_no_vggt_mixed_20260810_v1
LOG=${ROOT}/logs/Objaverse16_reconviagen_addendum.log
PID=${ROOT}/logs/Objaverse16_reconviagen_addendum.pid

bash pose_point_depth_mv/background_jobs/launch_detached.sh \
  objaverse16-reconviagen-addendum \
  "${LOG}" \
  "${PID}" \
  env OBJAVERSE16_RECON_GPU="${OBJAVERSE16_RECON_GPU}" OBJAVERSE16_ROOT="${ROOT}" \
  bash pose_point_depth_mv/background_jobs/run_objaverse16_reconviagen_addendum.sh

echo "后台任务不依赖当前终端"
echo "监视: tail -f '${LOG}'"
echo "current terminal remains open"


###############################################################################
# B. 监视；Ctrl-C只退出tail/watch，不终止任务
###############################################################################

tail -f /data/zjr/objaverse16_no_vggt_mixed_20260810_v1/logs/Objaverse16_reconviagen_addendum.log

watch -n 10 '
date
echo "=== process ==="
PID_FILE=/data/zjr/objaverse16_no_vggt_mixed_20260810_v1/logs/Objaverse16_reconviagen_addendum.pid
if [ -s "${PID_FILE}" ]; then ps -p "$(cat "${PID_FILE}")" -o pid,ppid,sid,stat,etime,time,pcpu,pmem,rss,cmd; fi
echo "=== recent log ==="
tail -n 12 /data/zjr/objaverse16_no_vggt_mixed_20260810_v1/logs/Objaverse16_reconviagen_addendum.log 2>/dev/null
echo "=== completion ==="
cat /data/zjr/objaverse16_no_vggt_mixed_20260810_v1/logs/Objaverse16_reconviagen_addendum.exit_code 2>/dev/null || echo "exit_code尚未生成"
test -s /data/zjr/objaverse16_no_vggt_mixed_20260810_v1/O9_current_vs_reconviagen_axisfixed_20k_v1/report.json && echo "O9报告已生成" || echo "O9报告尚未生成"
'


###############################################################################
# C. 不后台运行：当前终端直接执行
###############################################################################

cd /home/zjr/Tracker
OBJAVERSE16_RECON_GPU=${OBJAVERSE16_RECON_GPU:-4} \
OBJAVERSE16_ROOT=/data/zjr/objaverse16_no_vggt_mixed_20260810_v1 \
