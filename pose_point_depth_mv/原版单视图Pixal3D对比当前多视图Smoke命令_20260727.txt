# S0A：固定 NATTEN CUDA wheel 和 NAF 的最小额外 Python 依赖。wheel 已在当前
# 机器安装并保存为只读文件；迁移机器时会从官方 release 下载并先验证 SHA。
# 整段放在关闭 errexit 的子 shell 中，失败会报告非零 RC，但不会关闭当前 shell。
(
  set +e
  set -uo pipefail
  S0A_CODE=0

  if cd /home/zjr/Tracker; then
    mkdir -p "${NATTEN_WHEEL_ROOT}"
    if [ -s "${NATTEN_WHEEL}" ]; then
      ACTUAL_NATTEN_SHA=$(sha256sum "${NATTEN_WHEEL}" | awk '{print $1}')

        echo "NATTEN wheel SHA mismatch: ${NATTEN_WHEEL}"
        echo "expected=${NATTEN_WHEEL_SHA256}"
        echo "actual=${ACTUAL_NATTEN_SHA}"
        S0A_CODE=97
      fi
    elif [ -e "${NATTEN_WHEEL}" ]; then
      echo "NATTEN wheel path exists but is empty: ${NATTEN_WHEEL}"
      S0A_CODE=98

        if [ "${ACTUAL_NATTEN_SHA}" != "${NATTEN_WHEEL_SHA256}" ]; then
          echo "downloaded NATTEN wheel SHA mismatch"

          echo "actual=${ACTUAL_NATTEN_SHA}"
          S0A_CODE=97
        else
          mv "${NATTEN_PARTIAL}" "${NATTEN_WHEEL}"

        "${PIXAL_PY}" -m pip install --no-deps einops==0.8.0
        S0A_CODE=$?
      else
        echo "reuse pinned einops ${INSTALLED_EINOPS}"
      fi
    fi
  else
    S0A_CODE=95
  fi

  echo "S0A_RC=${S0A_CODE}"
  if [ "${S0A_CODE}" -ne 0 ]; then
    echo "S0A failed; do not run S2"
  fi
  echo "current terminal remains open"
  true
)


# S0：CPU/网络，仅首次运行。下载固定源码归档和 checkpoint，分别验证 SHA，
# 解包后写文件级 manifest 并设为只读。已有完整安装会只做校验。
(
  set +e
  set -uo pipefail
  S0_CODE=0

  verify_sha256() {
    local path=$1
    local expected=$2
    local actual
    if [ ! -s "${path}" ]; then
      echo "missing dependency: ${path}"
      return 96
    fi
    actual=$(sha256sum "${path}" | awk '{print $1}')
    if [ "${actual}" != "${expected}" ]; then
      echo "SHA mismatch: ${path}"
      echo "expected=${expected}"
      echo "actual=${actual}"
      return 97
    fi
  }

  download_verified() {
    local url=$1
    local destination=$2
    local expected=$3
    local partial=${destination}.partial
    if [ -e "${destination}" ]; then
      verify_sha256 "${destination}" "${expected}"
      return $?
    fi
    mkdir -p "$(dirname "${destination}")"
    curl -L --fail --retry 3 --retry-delay 5 --continue-at - \
      --output "${partial}" "${url}"
    local curl_rc=$?
    if [ "${curl_rc}" -ne 0 ]; then
      echo "download failed; resumable partial preserved: ${partial}"
      return "${curl_rc}"
    fi
    verify_sha256 "${partial}" "${expected}" || return $?
    mv "${partial}" "${destination}"
  }

  if cd /home/zjr/Tracker; then
    download_verified \
      "https://github.com/valeoai/NAF/archive/${NAF_COMMIT}.tar.gz" \
      "${NAF_ARCHIVE}" "${NAF_ARCHIVE_SHA256}"
    S0_CODE=$?

    if [ "${S0_CODE}" -eq 0 ]; then
      download_verified \
        "https://github.com/valeoai/NAF/releases/download/model/naf_release.pth" \
        "${NAF_CKPT}" "${NAF_CKPT_SHA256}"
      S0_CODE=$?
    fi

    if [ "${S0_CODE}" -eq 0 ] && [ ! -e "${NAF_REPO}" ]; then
      NAF_EXTRACTED=${NAF_ROOT}/NAF-${NAF_COMMIT}
      if [ ! -e "${NAF_EXTRACTED}" ]; then
        tar --no-same-owner --no-same-permissions \
          -xzf "${NAF_ARCHIVE}" -C "${NAF_ROOT}"
        S0_CODE=$?
      fi
      if [ "${S0_CODE}" -eq 0 ]; then
        verify_sha256 "${NAF_EXTRACTED}/hubconf.py" "${NAF_HUBCONF_SHA256}"
        S0_CODE=$?
      fi
      if [ "${S0_CODE}" -eq 0 ]; then
        mv "${NAF_EXTRACTED}" "${NAF_REPO}"
      fi
    elif [ "${S0_CODE}" -eq 0 ] && [ ! -d "${NAF_REPO}" ]; then
      echo "NAF_REPO exists but is not a directory: ${NAF_REPO}"
      S0_CODE=98
    fi

    if [ "${S0_CODE}" -eq 0 ]; then
      verify_sha256 "${NAF_REPO}/hubconf.py" "${NAF_HUBCONF_SHA256}"
      S0_CODE=$?
    fi
    if [ "${S0_CODE}" -eq 0 ]; then
      verify_sha256 "${NAF_CKPT}" "${NAF_CKPT_SHA256}"
      S0_CODE=$?
    fi

    if [ "${S0_CODE}" -eq 0 ] && [ -s "${NAF_SOURCE_MANIFEST}" ]; then
      (cd "${NAF_REPO}" && sha256sum --check "${NAF_SOURCE_MANIFEST}")
      S0_CODE=$?
    elif [ "${S0_CODE}" -eq 0 ]; then
      NAF_MANIFEST_TMP=${NAF_SOURCE_MANIFEST}.tmp
      (
        cd "${NAF_REPO}" &&
        find . -type f -print0 | sort -z | xargs -0 sha256sum
      ) > "${NAF_MANIFEST_TMP}"
      S0_CODE=$?
      if [ "${S0_CODE}" -eq 0 ]; then
        mv "${NAF_MANIFEST_TMP}" "${NAF_SOURCE_MANIFEST}"
      fi
    fi

    if [ "${S0_CODE}" -eq 0 ]; then
      verify_sha256 \
        "${NAF_SOURCE_MANIFEST}" "${NAF_SOURCE_MANIFEST_SHA256}"
      S0_CODE=$?
    fi

    if [ "${S0_CODE}" -eq 0 ]; then
      chmod -R a-w "${NAF_REPO}"
      chmod a-w "${NAF_ARCHIVE}" "${NAF_CKPT}" "${NAF_SOURCE_MANIFEST}"
      echo "pinned local NAF ready: ${NAF_ROOT}"
    fi
  else
    S0_CODE=95
  fi

  echo "S0_RC=${S0_CODE}"
  echo "current terminal remains open"
  true
)
