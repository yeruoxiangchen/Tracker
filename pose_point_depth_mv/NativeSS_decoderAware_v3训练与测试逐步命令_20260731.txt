Native SS decoder-aware v3 训练与测试逐步命令
================================================

本文档不是脚本。每次只复制一个完整的 (...) 命令块到终端。
按 D0 -> D2 -> S0 -> S5 -> V0 -> V4 顺序运行；任一硬门失败就停止。
本轮只修改和验证 Native SS。不要启动 SS1K、Native SLAT 或 Mesh。
所有输出使用 fresh v3 目录，不覆盖 post-CFG v2 的 step500 结果。

==============================

  "${PY}" -c "import inspect; import pose_point_depth_mv.test_native_3d_condition as a; import pose_point_depth_mv.test_native_ss_occupancy as b; tests=[(n,f) for m in (a,b) for n,f in inspect.getmembers(m, inspect.isfunction) if n.startswith('test_')]; [(print(n, flush=True), f()) for n,f in tests]; print(f'{len(tests)} tests passed'); assert len(tests) == 19"
  echo 'D0 PASS'

D1. 旧 step500 的 16-object residual-scale 与 decoder-logit 诊断（GPU 5，长任务）
================================================================================

(
  set -uo pipefail
  cd /home/zjr/Tracker
  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  ROOT=/data/zjr/native3d_condition_ss_decoderaware_20260731_v3
  VAL=/data/zjr/native3d_condition_reviewed1k_inputs_20260730_v3/lifting_val64/manifest.json
  CKPT=/data/zjr/native3d_condition_ss_postcfg_20260731_v2/ss64_smoke_seed42_v1/checkpoints/step_000500.pt
  OUT=${ROOT}/diagnostic_old_step500_val16_scale025_100_seed424344_v1

  test -s "${VAL}" || exit 96
  test -s "${CKPT}" || exit 96
  if [ -s "${OUT}/report.json" ]; then
    echo "reuse complete D1: ${OUT}"
  elif [ -e "${OUT}" ]; then
    echo "D1 partial output exists; preserve and inspect: ${OUT}"
    exit 98
  else
    mkdir -p "$(dirname "${LOG}")"
    env \
      CUDA_VISIBLE_DEVICES=5 \
      HF_HUB_OFFLINE=1 \
      TRANSFORMERS_OFFLINE=1 \
      ATTN_BACKEND=flash_attn \
      SPCONV_ALGO=native \
      PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \

        --cache_manifest "${VAL}" \
        --checkpoint "${CKPT}" \
        --output_dir "${OUT}" \
        --pretrained Stable-X/trellis-vggt-v0-2 \
        --indices all \
        --max_objects 16 \
        --joint_seeds 42,43,44 \
        --scale_multipliers 0.25,0.5,0.75,1.0 \
        --common_thresholds=-0.5,-0.25,-0.1,0,0.1,0.25,0.5 \
        --logit_quantiles 0.01,0.1,0.5,0.9,0.99 \
        --steps 25 \
        --cfg_strength 5.0 \
        --cfg_interval 0.5,1.0 \
        --guidance_rescale 0 \
        --rescale_t 3.0 \
        --condition_scale 1.0 \
        --bootstrap_samples 5000 \
        --min_iou_gain_mean 0 \
        --min_iou_win_rate 0.5 \
        --min_recall_gain_mean 0 \
        --min_latent_mse_gain_mean 0 \
        --amp_dtype bf16 \
      2>&1 | tee "${LOG}"
    RC=${PIPESTATUS[0]}
    printf '%s\n' "${RC}" > "${LOG}.exit_code"
    echo "D1_RC=${RC}"
    exit "${RC}"
  fi
  echo 'current terminal remains open'
)
