#!/usr/bin/env bash
#
# 目的
# ====
# 从 T3 final32 中按来源和视图数冻结 4 个与性能无关的随机对象，输出：
#
#   GT Mesh | Native Full | Stock | Pixal3D official final
#
# 两套无纹理 normal/clay 结果：
#
# 1. canonical_pose：主审查口径。四路共用 GT 派生的显示尺度/中心和相机，
#    禁止逐方法 normalize、autoframe、ICP 或 best transform。
# 2. shape_aligned：仅供看形状的 GT-assisted proper Sim(3)，禁止 reflection；
#    不能用于相机位姿、AR 放置或正式模型排名结论。
#
# 随机与资格规则
# ==============
# selection_seed=20260802；在 final32 的每个 (source, view_count) 桶内按
# SHA256(seed|source|view_count|uid) 排序。GT 源 Mesh 大于 128 MiB 的对象
# 在排序前因人工渲染内存风险排除；该规则不读取任何 Stock/Full 指标。
# 冻结结果应为：
#
# legacy_objaverse  v2  746ee2a21f064c53863156e2ea33ed4e_seq001
# gap_objaverse     v8  objaverse_f6b109eead9f4d369e1509ac2abbbe67_seq001
# pilot_objaverse   v4  objaverse_0022445c2ea049ca8c2eefe0e52c5fff_seq001
# omni              v4  omni_banana_008_seq001
#
# Pixal3D 历史错误保护
# ===================
# - 禁止渲染 decoded pre-remesh mesh.obj；只读取 official-final：
#     o_voxel.postprocess.to_glb(..., remesh=True, texture_size=4096)
#     + inference.py official output rotation
# - 每个 case 使用一个全新 Python/CUDA 进程，避免 NAF/remesh 显存累计。
# - 每个 case 加载模型前要求物理 GPU 至少 22000 MiB 空闲。
# - 固定 Pixal snapshot 0b31...；固定 NAF commit 37f2...；全程离线。
# - 使用 sdpa、--low_vram、--skip_rembg、known_crop_fov、1024、12 steps。
# - --skip_rembg 只接收协议冻结的带前景和透明像素的 RGBA。
# - Pixal canonical 方向使用已通过 6-object 审计的 frozen transform v2；
#   不从当前 GT/预测 Mesh 拟合旋转、平移或尺度。
# - T3 native/stock OBJ 均为 transform_pose=False；渲染前只应用固定
#   decoder (x,y,z)->reference (x,z,-y) 轴转换。
#
# GPU
# ===
# 运行前按实际空闲卡设置，例如：
#
#   export PIXAL_GPU=5
#   export RENDER_GPU=5
#
# P1 是高显存 Pixal3D 推理；P2 是低得多的无纹理 Mesh 渲染。

set -euo pipefail
cd /home/zjr/Tracker

PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
PIXAL_PY=/home/zjr/anaconda3/envs/pixal3d/bin/python
PIXAL_GPU=${PIXAL_GPU:-5}
RENDER_GPU=${RENDER_GPU:-${PIXAL_GPU}}
MIN_FREE_GPU_MIB=22000

PIXAL_REVISION=0b31f9160aa400719af409098bff7936a932f726
PIXAL_SNAPSHOT=/home/zjr/.cache/huggingface/hub/models--TencentARC--Pixal3D/snapshots/${PIXAL_REVISION}
NAF_COMMIT=37f2dfc180f2de53d98bd601109c0da0dd6b0f43
NAF_ROOT=/data/zjr/models/valeoai_NAF_${NAF_COMMIT}
NAF_REPO=${NAF_ROOT}/repo
NAF_CKPT=${NAF_ROOT}/naf_release.pth
NAF_SOURCE_MANIFEST=${NAF_ROOT}/source_manifest.sha256

T3_REPORT=/data/zjr/native_ss_stock_slat_mesh_transfer_20260802_v1/final32_seed424344_exploratory_v1/report.json
VAL_MANIFEST=/data/zjr/native3d_condition_reviewed1k_inputs_20260731_v4/native_ss_sourcebalanced_split_seed20260801_v1/val64_sourcebalanced_manifest.json
SPLIT_AUDIT=/data/zjr/native3d_condition_reviewed1k_inputs_20260731_v4/native_ss_sourcebalanced_split_seed20260801_v1/split_audit.json
PIXAL_TRANSFORM=/data/zjr/pixal3d_official_final_fix_20260729/pixal3d_official_final_to_direct_canonical_caseisolated_v3.json

WORK=/data/zjr/native_ss_stock_slat_mesh_transfer_20260802_v1/human_review_gt_native_stock_pixal3d_random4_v1
PROTOCOL_ROOT=${WORK}/protocol_official_pixal_v1
PROTOCOL=${PROTOCOL_ROOT}/protocol.json
PIXAL_LOG_ROOT=${WORK}/logs/pixal_official_caseisolated
RENDER_ROOT=${WORK}/renders_textureless_v1
RENDER_LOG_ROOT=${WORK}/logs/render_textureless_v1
PROJECT_TOOLS=/home/zjr/Tracker/pose_point_depth_mv/tools

run_logged() {
  local log_path=$1
  shift
  mkdir -p "$(dirname "${log_path}")"
  set +e
  "$@" 2>&1 | tee "${log_path}"
  local command_rc=${PIPESTATUS[0]}
  set -e
  printf '%s\n' "${command_rc}" > "${log_path}.exit_code"
  return "${command_rc}"
}

require_pixal_gpu_memory() {
  local free_mib
  free_mib=$(
    nvidia-smi -i "${PIXAL_GPU}" \
      --query-gpu=memory.free \
      --format=csv,noheader,nounits | tr -d '[:space:]'
  )
  if ! [[ "${free_mib}" =~ ^[0-9]+$ ]]; then
    echo "cannot read free memory for physical GPU ${PIXAL_GPU}"
    return 95
  fi
  if [ "${free_mib}" -lt "${MIN_FREE_GPU_MIB}" ]; then
    echo "Pixal3D blocked before model load: GPU ${PIXAL_GPU} free=${free_mib} MiB, required=${MIN_FREE_GPU_MIB} MiB"
    return 96
  fi
  echo "Pixal3D GPU memory gate passed: GPU ${PIXAL_GPU}, free=${free_mib} MiB"
}


###############################################################################
# C0. 代码、模型、NAF 和冻结输入预检（CPU；不启动 Pixal pipeline）
###############################################################################

"${PY}" -m py_compile \
  pose_point_depth_mv/prepare_native_ss_pixal3d_review.py \
  pose_point_depth_mv/render_native_ss_pixal3d_fourway.py \
  pose_point_depth_mv/compare_pixal3d_singleview_smoke.py \
  pose_point_depth_mv/evaluate_direct_slat_pixal3d_utility.py \
  pose_point_depth_mv/render_direct_slat_fourway.py

"${PY}" -m unittest pose_point_depth_mv.test_native_ss_pixal3d_review

test -s "${T3_REPORT}"
test -s "${VAL_MANIFEST}"
test -s "${SPLIT_AUDIT}"
test -s "${PIXAL_TRANSFORM}"
test -s "${PIXAL_SNAPSHOT}/pipeline.json"
test -s "${NAF_REPO}/hubconf.py"
test -s "${NAF_CKPT}"
test -s "${NAF_SOURCE_MANIFEST}"

"${PIXAL_PY}" -u -m pose_point_depth_mv.compare_pixal3d_singleview_smoke \
  verify-naf \
  --naf_repo "${NAF_REPO}" \
  --naf_checkpoint "${NAF_CKPT}" \
  --naf_source_manifest "${NAF_SOURCE_MANIFEST}"


###############################################################################
# P0. 冻结 4 个对象、最大 mask 单视图、c2w/内参、GT 和 T3 Mesh（CPU）
###############################################################################

"${PY}" -u -m pose_point_depth_mv.prepare_native_ss_pixal3d_review \
  prepare \
  --t3_report "${T3_REPORT}" \
  --cache_manifest "${VAL_MANIFEST}" \
  --split_audit "${SPLIT_AUDIT}" \
  --output_dir "${PROTOCOL_ROOT}" \
  --seed 42 \
  --selection_seed 20260802 \
  --max_gt_source_mib 128 \
  --source_view_targets legacy_objaverse:2,gap_objaverse:8,pilot_objaverse:4,omni:4

"${PY}" -u -m pose_point_depth_mv.prepare_native_ss_pixal3d_review \
  verify --protocol "${PROTOCOL}"

"${PY}" -c \
  'import json,sys
p=json.load(open(sys.argv[1]))
actual=[c["uid"] for c in p["cases"]]
expected=[
 "746ee2a21f064c53863156e2ea33ed4e_seq001",
 "objaverse_f6b109eead9f4d369e1509ac2abbbe67_seq001",
 "objaverse_0022445c2ea049ca8c2eefe0e52c5fff_seq001",
 "omni_banana_008_seq001",
]
assert actual==expected,(actual,expected)
print("P0 frozen UIDs:")
for c in p["cases"]:
 print(c["case_id"],c["source"],"v{}".format(c["view_count"]),c["uid"],"selected_view={}".format(c["selected_view_id"]))' \
  "${PROTOCOL}"


###############################################################################
# P1. Pixal3D official-final 推理（高显存；严格每 case 一个新进程）
###############################################################################

mapfile -t CASE_IDS < <(
  "${PY}" -c \
    'import json,sys
for case in json.load(open(sys.argv[1]))["cases"]:
 print(case["case_id"])' \
    "${PROTOCOL}"
)

case_index=0
for case_id in "${CASE_IDS[@]}"; do
  case_index=$((case_index + 1))
  case "${case_id}" in
    *[!A-Za-z0-9._-]*)
      echo "unsafe frozen case ID: ${case_id}"
      exit 97
      ;;
  esac
  require_pixal_gpu_memory
  printf -v case_number '%02d' "${case_index}"
  echo "Pixal3D official-final ${case_index}/${#CASE_IDS[@]}: ${case_id}"
  run_logged "${PIXAL_LOG_ROOT}/${case_number}_${case_id}.log" env \
    CUDA_VISIBLE_DEVICES="${PIXAL_GPU}" \
    ATTN_BACKEND=sdpa \
    SPARSE_ATTN_BACKEND=sdpa \
    HF_HUB_OFFLINE=1 \
    TRANSFORMERS_OFFLINE=1 \
    PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
    "${PIXAL_PY}" -u -m pose_point_depth_mv.compare_pixal3d_singleview_smoke \
      infer \
      --protocol "${PROTOCOL}" \
      --case_ids "${case_id}" \
      --model_path "${PIXAL_SNAPSHOT}" \
      --naf_repo "${NAF_REPO}" \
      --naf_checkpoint "${NAF_CKPT}" \
      --naf_source_manifest "${NAF_SOURCE_MANIFEST}" \
      --device cuda \
      --low_vram \
      --skip_rembg \
      --camera_mode known_crop_fov \
      --resolution 1024 \
      --sampling_steps 12
done


###############################################################################
# P2. 四路无纹理 canonical + shape-aligned 渲染（每对象一个新进程）
###############################################################################

mapfile -t REVIEW_ROWS < <(
  "${PY}" -c \
    'import json,sys
for case in json.load(open(sys.argv[1]))["cases"]:
 print(case["case_id"]+"|"+case["uid"])' \
    "${PROTOCOL}"
)

render_index=0
for review_row in "${REVIEW_ROWS[@]}"; do
  render_index=$((render_index + 1))
  case_id=${review_row%%|*}
  uid=${review_row#*|}
  output_dir=${RENDER_ROOT}/${case_id}
  resume_args=()
  if [ -e "${output_dir}" ] && [ ! -s "${output_dir}/report.json" ]; then
    echo "resume inspected partial render: ${output_dir}"
    resume_args=(--resume)
  fi
  printf -v render_number '%02d' "${render_index}"
  run_logged "${RENDER_LOG_ROOT}/${render_number}_${case_id}.log" env \
    PATH="${PROJECT_TOOLS}:${PATH}" \
    CUDA_VISIBLE_DEVICES="${RENDER_GPU}" \
    MPLCONFIGDIR=/tmp/matplotlib \
    NUMBA_CACHE_DIR=/tmp/numba_cache \
    TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
    PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
    "${PY}" -u -m pose_point_depth_mv.render_native_ss_pixal3d_fourway \
      --protocol "${PROTOCOL}" \
      --pixal_transform "${PIXAL_TRANSFORM}" \
      --output_dir "${output_dir}" \
      --uid "${uid}" \
      --seed 42 \
      --device cuda \
      --review_modes canonical_pose,shape_aligned \
      --render_frames 72 \
      --render_resolution 320 \
      --contact_frames 6 \
      --fps 12 \
      --display_margin 0.9 \
      --alignment_seed 20260802 \
      --candidate_samples 1000 \
      --alignment_samples 4000 \
      --candidate_iterations 8 \
      --final_iterations 30 \
      "${resume_args[@]}"
done


###############################################################################
# P3. 输出清单（CPU）
###############################################################################

"${PY}" -c \
  'import json,sys
from pathlib import Path
protocol=Path(sys.argv[1]); render_root=Path(sys.argv[2])
p=json.load(open(protocol))
print("Four-way textureless review complete:")
for case in p["cases"]:
 root=render_root/case["case_id"]
 report=root/"report.json"
 assert report.is_file(),report
 r=json.load(open(report)); assert r["complete"] is True
 print("\n",case["source"],case["uid"])
 print(" HTML:",root/"index.html")
 print(" canonical:",root/"canonical_pose"/"normal_contact_sheet.png")
 print(" shape-aligned (GT-assisted):",root/"shape_aligned"/"normal_contact_sheet.png")' \
  "${PROTOCOL}" "${RENDER_ROOT}"

echo "Protocol: ${PROTOCOL}"
echo "Render root: ${RENDER_ROOT}"
echo "current terminal remains open"
