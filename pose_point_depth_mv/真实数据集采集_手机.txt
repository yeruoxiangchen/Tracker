#!/usr/bin/env bash
set -euo pipefail

# 用途：
#   1. 从手机服务保留的 RGB/mask/pose 重建一份修正后的不可变数据集；
#   2. 保留旧的被拒绝 reconstruction 目录，以新 identity 继续 official
#      Native-SS step2000 + Native-SLat step25000 生成；
#   3. 即使新 runtime-O 仍未通过 pose-mask 质量门，也可以显式地作为
#      diagnostic 继续。质量失败记录不会被改成 passed。
#
# 用户消息中的两个路径完全相同，因此默认只处理唯一能确定的会话：
#   20260816_035545_862
# 如果第二个会话是其他 ID，用英文逗号追加，例如：
#   SESSION_IDS=20260816_035545_862,20260816_040547_970 MODE=prepare bash "$0"
# 当前服务器上另外两个同类被拒绝会话是
# 20260816_035300_136 和 20260816_040547_970；请按实际想指定的第二个选择，
# 本脚本不会猜测或自动批量处理它们。
#
# 三步执行：
#   MODE=prepare bash 'pose_point_depth_mv/真实数据集采集_手机.txt'
#   MODE=dry_run GPU=4 bash 'pose_point_depth_mv/真实数据集采集_手机.txt'
#   MODE=run     GPU=4 bash 'pose_point_depth_mv/真实数据集采集_手机.txt'

cd /home/zjr/Tracker

PY=${PY:-/home/zjr/anaconda3/envs/reconviagen/bin/python}
OUTPUT_ROOT=${OUTPUT_ROOT:-/home/zjr/Tracker/pose_point_depth_mv/outputs/可视AR}
GPU=${GPU:-4}
MODE=${MODE:-prepare}
SESSION_IDS=${SESSION_IDS:-20260816_035545_862}
MODEL_STEP=${MODEL_STEP:-25000}
SEED=${SEED:-42}

case "${MODE}" in
  prepare|dry_run|run) ;;
  *)
    echo "ERROR: MODE must be prepare, dry_run, or run; got ${MODE}" >&2
    exit 64
    ;;
esac

if ! test -x "${PY}"; then
  echo "ERROR: Python is missing or not executable: ${PY}" >&2
  exit 65
fi

IFS=',' read -r -a SOURCE_SESSIONS <<<"${SESSION_IDS}"
if test "${#SOURCE_SESSIONS[@]}" -eq 0; then
  echo "ERROR: SESSION_IDS is empty" >&2
  exit 64
fi

for SOURCE_SESSION in "${SOURCE_SESSIONS[@]}"; do
  case "${SOURCE_SESSION}" in
    ''|*[!A-Za-z0-9._-]*)
      echo "ERROR: unsafe source session ID: ${SOURCE_SESSION}" >&2
      exit 64
      ;;
  esac

  SOURCE_DATA=${OUTPUT_ROOT}/runtime/data/${SOURCE_SESSION}
  SOURCE_MASKS=${OUTPUT_ROOT}/runtime/masks/${SOURCE_SESSION}
  FIXED_SESSION=${SOURCE_SESSION}_axisuv_v5
  FIXED_DATASET=${OUTPUT_ROOT}/datasets/${FIXED_SESSION}
  FIXED_CAPTURE_REPORT=${FIXED_DATASET}/capture_report.json
  RECON_SESSION=real_official_slat_step${MODEL_STEP}_phone_${SOURCE_SESSION}_axisuv_v5_seed${SEED}_diagnostic_v1
  RECON_DIR=${OUTPUT_ROOT}/reconstructions/${RECON_SESSION}

  echo "============================================================"
  echo "source_session=${SOURCE_SESSION}"
  echo "fixed_dataset=${FIXED_DATASET}"
  echo "reconstruction=${RECON_DIR}"
  echo "mode=${MODE} gpu=${GPU}"

  test -d "${SOURCE_DATA}"
  test -d "${SOURCE_MASKS}"
  test -s "${SOURCE_DATA}/poses.txt"
  test -s "${SOURCE_DATA}/frame_metadata.jsonl"

  RGB_COUNT=$(find "${SOURCE_DATA}" -maxdepth 1 -type f -name 'frame_*.jpg' | wc -l)
  MASK_COUNT=$(find "${SOURCE_MASKS}" -maxdepth 1 -type f \
    -name 'frame_[0-9][0-9][0-9][0-9].png' | wc -l)
  if test "${RGB_COUNT}" -lt 8 || test "${MASK_COUNT}" -lt "${RGB_COUNT}"; then
    echo "ERROR: incomplete source RGB/mask set: rgb=${RGB_COUNT} mask=${MASK_COUNT}" >&2
    exit 66
  fi

  if test "${MODE}" = prepare; then
    if ! test -s "${FIXED_CAPTURE_REPORT}"; then
      "${PY}" -u -m pose_point_depth_mv.reconstruct_existing_ar_session \
        --source_session_id "${SOURCE_SESSION}" \
        --session_id "${FIXED_SESSION}" \
        --output_root "${OUTPUT_ROOT}" \
        --capture_only
    else
      echo "reuse immutable corrected dataset: ${FIXED_DATASET}"
    fi

    "${PY}" - "${FIXED_CAPTURE_REPORT}" <<'PY'
import json
import sys

path = sys.argv[1]
report = json.load(open(path, encoding="utf-8"))
axis = report["sparse_metadata"]["camera_axis_diagnostics"]
contract = axis["saved_image_axis_contract"]
selected = float(axis["selected_image_camera_rotation_degrees"])
display_uv = float(contract["display_matrix_uv_rotation_degrees"])
assert report["passed"] is True
assert report["geometry_mode"] == "pose_mask_only"
assert report["point_cloud_consumed"] is False
assert axis["selection_source"] == "xrcpuimage_none_direct_display_uv_v5"
assert contract["display_matrix_application"] == (
    "direct_pose_camera_to_native_cpu_image_uv_rotation"
)
assert abs(selected - display_uv) < 1.0e-6
print(json.dumps({
    "prepared": True,
    "capture_report": path,
    "selected_image_camera_rotation_degrees": selected,
    "contract_source": axis["selection_source"],
    "frame_count": report["selected_frame_count"],
}, indent=2, ensure_ascii=False))
PY
    continue
  fi

  if ! test -s "${FIXED_CAPTURE_REPORT}"; then
    echo "ERROR: corrected dataset is absent; run MODE=prepare first" >&2
    exit 67
  fi

  if test "${MODE}" = dry_run; then
    LAUNCH_DRY_RUN=1
  else
    LAUNCH_DRY_RUN=0
  fi

  # 用户明确要求对被拒绝数据继续生成，因此这里显式启用
  # diagnostic bypass。runtime_input_manifest.json 仍保留真实的
  # formal_input_passed/checks；该结果不得作为合格输入或正式结论。
  GPU="${GPU}" \
  MODEL_STEP="${MODEL_STEP}" \
  SEED="${SEED}" \
  SESSION_ID="${RECON_SESSION}" \
  RUN_TAG=axisuv_v5_diagnostic_v1 \
  DRY_RUN="${LAUNCH_DRY_RUN}" \
  DIAGNOSTIC_BYPASS_POSE_MASK_QUALITY=1 \
    bash pose_point_depth_mv/background_jobs/run_real_official_slat_retest_one.sh \
      "${FIXED_DATASET}"

  if test "${MODE}" = run; then
    REPORT=${RECON_DIR}/reconstruction_report.json
    test -s "${REPORT}"
    "${PY}" - "${REPORT}" <<'PY'
import json
import sys

r = json.load(open(sys.argv[1], encoding="utf-8"))
assert r["passed"] is True
assert r["identity"]["diagnostic_bypass_pose_mask_quality"] is True
print(json.dumps({
    "passed": True,
    "formal_claim_allowed": r["formal_claim_allowed"],
    "run_dir": r["run_dir"],
    "world_glb": r["meshes"]["world_glb"],
    "contact_sheet": r["previews"]["contact_sheet"],
}, indent=2, ensure_ascii=False))
PY
  fi
done

# 后台执行示例（先完成 MODE=prepare）：
#   tmux new-session -d -s phone_axisuv_v5 \
