run_logged() (
  local log_path=$1
  local exit_path=${log_path}.exit_code
  shift
  set +e
  "$@" 2>&1 | tee "${log_path}"
  local pipe_status=("${PIPESTATUS[@]}")
  local command_rc=${pipe_status[0]}
  local tee_rc=${pipe_status[1]}
  {
    echo "command exit code: ${command_rc}"
    echo "tee exit code: ${tee_rc}"
  } | tee -a "${log_path}"
  printf '%s\n' "${command_rc}" > "${exit_path}"
  if [ "${command_rc}" -ne 0 ]; then
    return "${command_rc}"
  fi
  return "${tee_rc}"
)

run_runtime_repeat_process() (
  local runtime_id=$1
  local attention_backend=$2
  local spconv_algo=$3
  local amp_dtype=$4
  local process_index=$5
  local decoder_repeats=$6
  local slat_repeats=$7
  local output_dir=${RUNTIME_DIAG_ROOT}/${runtime_id}/process_${process_index}
  if [ -s "${output_dir}/report.json" ]; then
    echo "reuse immutable runtime report: ${output_dir}/report.json"
    "${PY}" -c \
      'import json,sys; r=json.load(open(sys.argv[1])); raise SystemExit(0 if r.get("complete") is True else 3)' \
      "${output_dir}/report.json"
    return $?
  fi
  if [ -e "${output_dir}" ]; then
    echo "partial runtime output exists; preserve and inspect: ${output_dir}"
    return 98
  fi
  mkdir -p "$(dirname "${output_dir}")"
  run_logged "${output_dir}.log" \
    env \
      CUDA_VISIBLE_DEVICES=4 \
      HF_HUB_OFFLINE=1 \
      TRANSFORMERS_OFFLINE=1 \
      ATTN_BACKEND="${attention_backend}" \
      SPARSE_ATTN_BACKEND="${attention_backend}" \
      SPCONV_ALGO="${spconv_algo}" \
      CUBLAS_WORKSPACE_CONFIG=:4096:8 \
      MPLCONFIGDIR=/tmp/matplotlib \
      NUMBA_CACHE_DIR=/tmp/numba_cache \
      TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
      PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
      PYTHONFAULTHANDLER=1 \
    "${PY}" -u -m pose_point_depth_mv.diagnose_direct_slat_runtime_repeat \
      --protocol "${SOURCE_PREFLIGHT_PROTOCOL_DIR}/protocol.json" \
      --blind_key_file "${SOURCE_PREFLIGHT_PROTOCOL_DIR}/blind_key.txt" \
      --source_preflight_output "${SOURCE_PREFLIGHT_OUT}" \
      --output_dir "${output_dir}" \
      --runtime_id "${runtime_id}" \
      --process_index "${process_index}" \
      --object_uids 04d712a0dae0421cae9083484f4332ee \
      --joint_seeds 43,44 \
      --branches stock,full \
      --decoder_repeats "${decoder_repeats}" \
      --slat_repeats "${slat_repeats}" \
      --surface_samples 20000 \
      --attention_backend "${attention_backend}" \
      --spconv_algo "${spconv_algo}" \
      --amp_dtype "${amp_dtype}" \
      --deterministic_algorithms off \
      --allow_tf32 false \
      --device cuda
)
