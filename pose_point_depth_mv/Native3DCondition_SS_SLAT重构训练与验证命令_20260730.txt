fail() {
  echo "ERROR: $*" >&2
  exit 96
}

require_env() {
  local name=$1
  local value=${!name:-}
  [ -n "${value}" ] || fail "set ${name} before PHASE=${PHASE}"
}

require_file() {
  [ -s "$1" ] || fail "required non-empty file is missing: $1"
}

require_dir() {
  [ -d "$1" ] || fail "required directory is missing: $1"
}

run_logged() {
  local log=$1
  shift
  mkdir -p "$(dirname "${log}")"
  echo "log=${log}"
  echo "monitor: tail -f '${log}'"
  set +e
  "$@" 2>&1 | tee "${log}"
  local rc=${PIPESTATUS[0]}
  set -e
  printf '%s\n' "${rc}" > "${log}.exit_code"
  return "${rc}"
}

run_new_output() {
  local output=$1
  local log=$2
  shift 2
  if [ -s "${output}/report.json" ]; then
    echo "reuse immutable complete output: ${output}"
    return 0
  fi
  if [ -e "${output}" ]; then
    fail "partial output exists; preserve and inspect before retry: ${output}"
  fi
  run_logged "${log}" "$@"
}
