run_logged() {
  LOG_PATH=$1
  shift
  set +e
  "$@" 2>&1 | tee "${LOG_PATH}"
  COMMAND_RC=${PIPESTATUS[0]}
  printf '%s\n' "${COMMAND_RC}" > "${LOG_PATH}.exit_code"
  return "${COMMAND_RC}"
}

run_direct_mesh() {
  DIRECT_OUT=$1
  shift
  if [ -s "${DIRECT_OUT}/report.json" ]; then
    echo "reuse immutable Direct Mesh: ${DIRECT_OUT}"
  elif [ -e "${DIRECT_OUT}" ]; then
    echo "Direct Mesh partial output exists; preserve and inspect: ${DIRECT_OUT}"
    return 91
  else
    run_logged "${DIRECT_OUT}.log" env \
      CUDA_VISIBLE_DEVICES="${GPU}" \
      HF_HUB_OFFLINE=1 \
      TRANSFORMERS_OFFLINE=1 \
      ATTN_BACKEND=flash_attn \
      SPCONV_ALGO=native \
      PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
      "${PY}" -u -m pose_point_depth_mv.export_direct_slat_mesh_pairs \
        --cache_manifest "${VAL_CACHE}/manifest.json" \
        --train_cache_manifest "${TRAIN_CACHE}/manifest.json" \
        --checkpoint "${PREFLIGHT_CKPT}" \
        --output_dir "${DIRECT_OUT}" \
        --pretrained "${PRETRAINED}" \
        --steps 25 \
        --cfg_strength 5.0 \
        --cfg_interval 0.5,1.0 \
        --rescale_t 3.0 \
        --slat_guided_delta_policy post_cfg_v2 \
        --surface_samples 20000 \
        --bootstrap_samples 5000 \
        --amp_dtype bf16 \
        --noise_identity_mode object_uid_v1 \
        "$@"
  fi
}

prepare_pixal_protocol() {
  DIRECT_REPORT=$1
  PIXAL_OUT=$2
  JOINT_SEEDS=$3
  "${PY}" -u -m pose_point_depth_mv.compare_pixal3d_singleview_smoke \
    prepare \
    --current_report "${DIRECT_REPORT}" \
    --cache_manifest "${VAL_CACHE}/manifest.json" \
    --output_dir "${PIXAL_OUT}" \
    --selection_mode report_objects_v1 \
    --joint_seeds "${JOINT_SEEDS}" \
    --view_counts 2 4 8 \
    --selection_seed 20260728
}

run_pixal_native() {
  PIXAL_OUT=$1
  PIXAL_PROTOCOL=${PIXAL_OUT}/protocol.json
  if [ ! -s "${PIXAL_PROTOCOL}" ]; then
    echo "missing Pixal3D protocol: ${PIXAL_PROTOCOL}"
    return 92
  fi
  "${PY}" -c \
    'import json,sys; from pathlib import Path; from pose_point_depth_mv.compare_pixal3d_singleview_smoke import pixal3d_mesh_path,sha256_file,validate_protocol; pth=Path(sys.argv[1]).resolve(); p=validate_protocol(pth); ok=True
for c in p["cases"]:
 m=pixal3d_mesh_path(pth,c["case_id"]); r=m.with_name("result.json")
 if not m.is_file() or not r.is_file(): ok=False; break
 x=json.loads(r.read_text(encoding="utf-8"))
 if x.get("protocol_sha256")!=p["protocol_sha256"] or x.get("mesh_sha256")!=sha256_file(m): ok=False; break
raise SystemExit(0 if ok else 1)' \
    "${PIXAL_PROTOCOL}"
  PIXAL_REUSE_RC=$?
  if [ "${PIXAL_REUSE_RC}" -eq 0 ]; then
    echo "reuse immutable Pixal3D outputs: ${PIXAL_OUT}"
    return 0
  fi
  PIXAL_LOG=${PIXAL_OUT}/pixal3d_infer_sdpa.log
  if [ -e "${PIXAL_LOG}" ]; then
    PIXAL_LOG=${PIXAL_OUT}/pixal3d_infer_sdpa.resume1.log
  fi
  run_logged "${PIXAL_LOG}" env \
    CUDA_VISIBLE_DEVICES="${GPU}" \
    ATTN_BACKEND=sdpa \
    SPARSE_ATTN_BACKEND=sdpa \
    HF_HUB_OFFLINE=1 \
    TRANSFORMERS_OFFLINE=1 \
    PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
    "${PIXAL_PY}" -u -m pose_point_depth_mv.compare_pixal3d_singleview_smoke \
      infer \
      --protocol "${PIXAL_PROTOCOL}" \
      --model_path "${PIXAL_SNAPSHOT}" \
      --naf_repo "${NAF_REPO}" \
      --naf_checkpoint "${NAF_CKPT}" \
      --naf_source_manifest "${NAF_SOURCE_MANIFEST}" \
      --device cuda \
      --low_vram \
      --skip_rembg \
      --camera_mode known_crop_fov \
      --resolution 1024 \
      --sampling_steps 12
}
