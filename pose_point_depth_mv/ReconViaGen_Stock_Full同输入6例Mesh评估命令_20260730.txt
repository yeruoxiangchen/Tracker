#!/usr/bin/env bash
#
# 当前 Direct-SLAT v5.1 step100 的 6-object / seed42 探索性比较：
#   1. 读取 Direct 报告中已经冻结的 6 个 uid。
#   2. 将每个 uid 对应的相同多视图 RGB + mask 直接送入原版 ReconViaGen。
#   3. 用统一坐标约定和统一采样指标比较：
#        ReconViaGen original / Direct stock / Direct full。
#
# 默认只占用一张 GPU。运行前按实际空闲卡设置，例如：
#   GPU=4 bash pose_point_depth_mv/ReconViaGen_Stock_Full同输入6例Mesh评估命令_20260730.txt

set -euo pipefail

cd /home/zjr/Tracker

PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
GPU=${GPU:-4}

DIRECT_ROOT=/data/zjr/direct_slat_v5_1_terminal_rollout_20260729/train897_step100_seed42_2gpu_v3/mesh6_components_seed42_v1
DIRECT_REPORT=${DIRECT_ROOT}/report.json
DIRECT_RUN_CONFIG=${DIRECT_ROOT}/run_config.json

RECON_ROOT=/data/zjr/reconviagen_original_sameinput_step100_mesh6_seed42_20260730_v1
EVAL_ROOT=/data/zjr/reconviagen_stock_full_sameinput_step100_mesh6_seed42_20260730_v1
LOG_ROOT=/data/zjr/reconviagen_stock_full_sameinput_step100_mesh6_seed42_20260730_v1_logs

if [ ! -x "${PY}" ]; then
  echo "blocked: Python is missing: ${PY}"
  exit 96
fi
if [ ! -s "${DIRECT_REPORT}" ] || [ ! -s "${DIRECT_RUN_CONFIG}" ]; then
  echo "blocked: Direct step100 6-case report/run_config is missing"
  exit 96
fi

# 同时得到 Direct cache manifest 和它所绑定的原始 lifting manifest。
DIRECT_CACHE=$(
  "${PY}" -c \
    'import json,sys; print(json.load(open(sys.argv[1]))["cache_manifest"])' \
    "${DIRECT_RUN_CONFIG}"
)
SOURCE_LIFTING=$(
  "${PY}" -c \
    'import json,sys; print(json.load(open(sys.argv[1]))["source_lifting_manifest"])' \
    "${DIRECT_CACHE}"
)

if [ ! -s "${SOURCE_LIFTING}" ]; then
  echo "blocked: source lifting manifest is missing: ${SOURCE_LIFTING}"
  exit 96
fi

# 这一步明确拒绝误用 18-record 的 seed42/43/44 报告。
"${PY}" -c \
  'import json,sys
r=json.load(open(sys.argv[1]))
assert r["format"]=="pose_point_depth_mv.direct_slat_mesh_exploratory.v2"
assert r["checkpoint_step"]==100
assert r["object_count"]==6
assert r["joint_seeds"]==[42]
assert len(r["records"])==6
assert len({x["uid"] for x in r["records"]})==6
assert all(int(x["joint_seed"])==42 for x in r["records"])
print("Direct input audit passed: 6 objects x seed42 = 6 records")
for x in r["records"]:
    print(x["pair_id"], x["uid"], x["joint_seed"])' \
  "${DIRECT_REPORT}"

mkdir -p "${RECON_ROOT}" "${LOG_ROOT}"

# 每个子进程都只接收 source cache 已冻结的 RGB/mask；没有重新渲染或构造输入。
while IFS=$'\t' read -r PAIR_ID SAMPLE_UID SEED; do
  CASE_OUT=${RECON_ROOT}/${PAIR_ID}
  CASE_LOG=${LOG_ROOT}/${PAIR_ID}.log

  if [ -s "${CASE_OUT}/report.json" ]; then
    echo "reuse original ReconViaGen: ${PAIR_ID}"
    continue
  fi
  if [ -e "${CASE_OUT}" ]; then
    echo "blocked: partial ReconViaGen output exists; preserve and inspect: ${CASE_OUT}"
    exit 98
  fi

  echo "run original ReconViaGen: ${PAIR_ID} uid=${SAMPLE_UID} seed=${SEED} GPU=${GPU}"
  CUDA_VISIBLE_DEVICES="${GPU}" \
  HF_HUB_OFFLINE=1 \
  TRANSFORMERS_OFFLINE=1 \
  ATTN_BACKEND=flash_attn \
  SPCONV_ALGO=native \
  MPLCONFIGDIR=/tmp/matplotlib \
  NUMBA_CACHE_DIR=/tmp/numba_cache \
  TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
  PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
  "${PY}" -u -m pose_point_depth_mv.infer_reconviagen_stock_from_direct_cache \
    --source_lifting_manifest "${SOURCE_LIFTING}" \
    --uid "${SAMPLE_UID}" \
    --output_dir "${CASE_OUT}" \
    --pretrained Stable-X/trellis-vggt-v0-2 \
    --device cuda \
    --low_vram \
    --seed "${SEED}" \
    --ss_steps 30 \
    --ss_guidance 7.5 \
    --ss_guidance_rescale 0.7 \
    --ss_rescale_t 5.0 \
    --slat_steps 12 \
    --slat_guidance 7.5 \
    --slat_guidance_rescale 0.5 \
    --slat_rescale_t 3.0 \
    --multiimage_algo multidiffusion \
    2>&1 | tee "${CASE_LOG}"
  CASE_RC=${PIPESTATUS[0]}
  if [ "${CASE_RC}" -ne 0 ]; then
    echo "original ReconViaGen failed: pair=${PAIR_ID} rc=${CASE_RC}"
    exit "${CASE_RC}"
  fi
done < <(
  "${PY}" -c \
    'import json,sys
r=json.load(open(sys.argv[1]))
for x in r["records"]:
    print("{}\t{}\t{}".format(x["pair_id"],x["uid"],int(x["joint_seed"])))' \
    "${DIRECT_REPORT}"
)

RECON_REPORT_COUNT=$(
  find "${RECON_ROOT}" -mindepth 2 -maxdepth 2 -type f -name report.json | wc -l
)
if [ "${RECON_REPORT_COUNT}" -ne 6 ]; then
  echo "blocked: expected 6 original ReconViaGen reports, got ${RECON_REPORT_COUNT}"
  exit 96
fi

if [ -s "${EVAL_ROOT}/report.json" ]; then
  echo "reuse completed three-way evaluation: ${EVAL_ROOT}/report.json"
elif [ -e "${EVAL_ROOT}" ]; then
  echo "blocked: partial immutable evaluation exists; preserve and inspect: ${EVAL_ROOT}"
  exit 98
else
  "${PY}" -u -m pose_point_depth_mv.evaluate_reconviagen_stock_full_mesh \
    --direct_report "${DIRECT_REPORT}" \
    --recon_root "${RECON_ROOT}" \
    --output_dir "${EVAL_ROOT}" \
    --surface_samples 100000 \
    --thresholds 0.01,0.02,0.05 \
    --bootstrap_samples 10000 \
    --metric_seed 20260730 \
    --alignment_modes canonical_pose \
    2>&1 | tee "${LOG_ROOT}/three_way_evaluation.log"
  EVAL_RC=${PIPESTATUS[0]}
  if [ "${EVAL_RC}" -ne 0 ]; then
    echo "three-way evaluation failed: rc=${EVAL_RC}"
    exit "${EVAL_RC}"
  fi
fi

"${PY}" -c \
  'import json,sys
r=json.load(open(sys.argv[1]))
assert r["format"]=="pose_point_depth_mv.reconviagen_stock_full_mesh_accuracy.v1"
assert r["object_count"]==6 and r["record_count"]==6 and r["joint_seeds"]==[42]
assert r["same_input_audit"]["passed"] is True
assert len(r["same_input_audit"]["records"])==6
assert all(x["passed"] is True for x in r["same_input_audit"]["records"])
print(json.dumps({
  "passed": True,
  "formal": r["formal"],
  "objects": r["object_count"],
  "records": r["record_count"],
  "seeds": r["joint_seeds"],
  "canonical_pose": {
    "methods": r["modes"]["canonical_pose"]["method_summary"],
    "comparisons": r["modes"]["canonical_pose"]["comparison_summary"],
  },
},indent=2,ensure_ascii=False))' \
  "${EVAL_ROOT}/report.json"

echo "summary: ${EVAL_ROOT}/summary.txt"
echo "report: ${EVAL_ROOT}/report.json"
echo "current terminal remains open"
