#!/usr/bin/env bash
# GT 固定；Native Full、Stock、Pixal3D 分别以 proper SE(3) ICP 对齐到 GT。
# 只消除旋转和平移，严格禁止缩放与镜像；结果仅用于几何形状人工审查。

set -euo pipefail
cd /home/zjr/Tracker

PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
RENDER_GPU=${RENDER_GPU:-5}
PROTOCOL=/data/zjr/native_ss_stock_slat_mesh_transfer_20260802_v1/human_review_gt_native_stock_pixal3d_random4_v1/protocol_official_pixal_v1/protocol.json
PIXAL_TRANSFORM=/data/zjr/pixal3d_official_final_fix_20260729/pixal3d_official_final_to_direct_canonical_caseisolated_v3.json
VIS_ROOT=/home/zjr/Tracker/pose_point_depth_mv/outputs/可视化Mesh
OUT=${VIS_ROOT}/四路Mesh_GT刚体对齐_20260802_v2
CASE_ROOT=${OUT}/逐例审计
LOG_ROOT=${OUT}/日志
PROJECT_TOOLS=/home/zjr/Tracker/pose_point_depth_mv/tools

mapfile -t REVIEW_ROWS < <(
  "${PY}" -c \
    'import json,sys
for case in json.load(open(sys.argv[1]))["cases"]:
 print(case["case_id"]+"|"+case["uid"])' \
    "${PROTOCOL}"
)

PNG_NAMES=(
  01_历史Objaverse_2视图_GT刚体对齐四路对比.png
  02_补充Objaverse_8视图_GT刚体对齐四路对比.png
  03_试渲染Objaverse_4视图_GT刚体对齐四路对比.png
  04_Omni香蕉_4视图_GT刚体对齐四路对比.png
)

if [ "${#REVIEW_ROWS[@]}" -ne 4 ]; then
  echo "blocked: expected 4 frozen cases, got ${#REVIEW_ROWS[@]}"
  exit 96
fi

mkdir -p "${CASE_ROOT}" "${LOG_ROOT}"
for case_index in "${!REVIEW_ROWS[@]}"; do
  review_row=${REVIEW_ROWS[${case_index}]}
  case_id=${review_row%%|*}
  uid=${review_row#*|}
  case_out=${CASE_ROOT}/${case_id}
  resume_args=()
  if [ -e "${case_out}" ] && [ ! -s "${case_out}/report.json" ]; then
    resume_args=(--resume)
  fi

  PATH="${PROJECT_TOOLS}:${PATH}" \
  CUDA_VISIBLE_DEVICES="${RENDER_GPU}" \
  MPLCONFIGDIR=/tmp/matplotlib \
  NUMBA_CACHE_DIR=/tmp/numba_cache \
  TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
  PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
  "${PY}" -u -m pose_point_depth_mv.render_native_ss_pixal3d_fourway \
    --protocol "${PROTOCOL}" \
    --pixal_transform "${PIXAL_TRANSFORM}" \
    --output_dir "${case_out}" \
    --uid "${uid}" \
    --seed 42 \
    --device cuda \
    --review_modes rigid_pose_gt \
    --render_frames 36 \
    --render_resolution 320 \
    --contact_frames 6 \
    --fps 12 \
    --display_margin 0.9 \
    --alignment_seed 20260802 \
    --candidate_samples 1000 \
    --alignment_samples 4000 \
    --candidate_iterations 8 \
    --final_iterations 30 \
    "${resume_args[@]}" \
    2>&1 | tee "${LOG_ROOT}/$((case_index + 1))_${case_id}.log"

  sheet=${case_out}/rigid_pose_gt/normal_contact_sheet.png
  test -s "${sheet}"
  cp -f "${sheet}" "${OUT}/${PNG_NAMES[${case_index}]}"
done

"${PY}" -c \
  'from pathlib import Path
import sys
root=Path(sys.argv[1]); files=sorted(root.glob("*.png"))
assert len(files)==4, files
print("刚体对齐四路 PNG：")
for path in files: print(path)' \
  "${OUT}"

"${PY}" -u -m pose_point_depth_mv.collect_fourway_review_assets \
  --review_root "${CASE_ROOT}" \
  --output_dir "${OUT}/四路OBJ分类" \
  --review_mode rigid_pose_gt \
  --expected_cases 4

echo "逐例 Mesh、视频和变换审计: ${CASE_ROOT}"
echo "按对象分类的 16 个 OBJ: ${OUT}/四路OBJ分类"
echo "current terminal remains open"
