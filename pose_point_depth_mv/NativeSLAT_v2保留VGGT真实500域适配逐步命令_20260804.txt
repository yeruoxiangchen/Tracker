# OmniObject3D 真实视频子集下载、事后去重与 500/64/64 冻结命令
# 日期：2026-08-04
# 版本：v2（覆盖“下载前过早筛选”的旧思路）
#
# 正确顺序：
#   D0  核对本地 Scan.obj、远端 videos_processed 和现有 29-object pilot。
#   D1  仅根据“远端类别归档存在 + 本地该类有 Scan.obj”冻结下载子集。
#       此时不排除 reviewed1k 中用过的 Omni 对象。
#   D2  逐类别归档、可恢复下载。
#   D3  下载完成后冻结 gzip/尺寸报告。
#   D4  解析归档目录，冻结严格的逐对象字段盘点（v1审计证据）。
#   D4V2 从D4 v1显式分离eligible/rejected；允许有限坏对象被审计拒绝。
#   D5  从冻结 reviewed1k split 生成 135-object 旧 Omni 排除清单。
#   D5P 冻结 pilot29 计数账本，仅记录与 reviewed1k 的重叠，不排除 pilot。
#   D6  在 D4V2 eligible集合上事后仅排除这 135 个对象，再冻结
#       500 real-train + 64 real-dev + 64 untouched real-holdout。
#       pilot29 不排除，D6 只记录它们在各 split 中的去向。
#
# 下载规模不是“直接宣称 1000 个可训样本”：
#   pilot 中 35 个本地 Scan 候选只有 29 个实际 video/mesh 配对。为了在下载后
#   排除最多 135 个旧 reviewed1k Omni，并仍保留 628 个新物理对象，
#   覆盖率估算为：
#
#     ceil((500 + 64 + 64 + 135) / (29 / 35)) = 921
#
#   D1 因此选择 >=1000 个本地 Scan.obj 候选，给自动失败留少量余量。
#   OpenXLab 的实际下载粒度是类别 tar.gz，无法在下载前得到精确 object 交集。
#
# 边界：
#   - 本文件不包含 R8、模型训练或评估命令。
#   - D4 v1不得覆盖；D4V2必须保留全部rejected对象及失败原因。
#   - D6 的 split 仍是 raw video/mesh 候选，training_ready 必须为 false。
#   - 旧 Omni 指 reviewed1k 中 119 train + 16 已消费 val，不是仅排除 119 train。
#   - pilot29 不属于 reviewed1k 排除规则；除 battery_003 因与 reviewed1k
#     重叠而被排除外，其余 28 个正常参与 split。
#   - 去重同时检查规范化 object ID 和本地 Scan.obj 解析路径。
#   - 不要一次运行整份文件；每步先检查 RC 和 JSON。
#   - 所有失败只退出子 shell，不会关闭当前交互终端。
#   - 既有 complete 结果复用；partial/损坏结果保留，不覆盖、不删除。

  REMOTE=/data/zjr/dataset10k_20260726/manifests/omniobject3d_remote_list_20260726.txt
  SCANS=/data/zjr/OmniObject3D/raw_scans_extracted_omni215_v1/categories
  ARCHIVES=/data/zjr/OmniObject3D/raw/omniobject3d___OmniObject3D-New/raw/videos_processed
  ROOT=/data/zjr/omni_real_video500_download_20260804_v2
  PLAN=${ROOT}/manifests/D1_download_plan.json
  LIST=${ROOT}/manifests/D1_categories.txt

  if [ -s "${PLAN}" ] && [ -s "${LIST}" ]; then
    "${PY}" - "${PLAN}" "${LIST}" <<'PY'
import hashlib
import json
import sys
from pathlib import Path
plan = json.loads(Path(sys.argv[1]).read_text(encoding="utf-8"))
payload = Path(sys.argv[2]).read_text(encoding="utf-8")
categories = [line for line in payload.splitlines() if line]
assert plan["format"] == "pose_point_depth_mv.omni_real_video_download_plan.v2"

print(json.dumps({k: plan[k] for k in (

  else
    "${PY}" - "${PILOT_LIST}" "${REMOTE}" "${SCANS}" "${ARCHIVES}" "${PLAN}" "${LIST}" <<'PY'
import hashlib
import json
import re
import sys
from datetime import datetime, timezone
from pathlib import Path

pilot_list = Path(sys.argv[1]).resolve()
remote_list = Path(sys.argv[2]).resolve()
scan_root = Path(sys.argv[3]).resolve()
archive_root = Path(sys.argv[4]).resolve()
plan_path = Path(sys.argv[5]).resolve()
list_path = Path(sys.argv[6]).resolve()
target_candidates = 1000
minimum_categories = 48
units = {"": 1, "K": 2**10, "M": 2**20, "G": 2**30, "T": 2**40}
pattern = re.compile(
    r"\|\s*(/raw/videos_processed/([^/]+)\.tar\.gz)\s*"
    r"\|\s*([0-9.]+)\s*([KMGT]?)\s*\|"
)

available = {}
for line in remote_list.read_text(encoding="utf-8", errors="replace").splitlines():
    match = pattern.search(line)
    if not match:
        continue
    remote_path, category, number, unit = match.groups()
    category_root = scan_root / category
    if not category_root.is_dir():
        continue
    object_ids = sorted(
        path.name
        for path in category_root.iterdir()
        if path.is_dir() and (path / "Scan" / "Scan.obj").is_file()
    )
    if not object_ids:
        continue
    available[category] = {
        "category": category,
        "remote_path": remote_path,
        "remote_bytes": int(float(number) * units[unit]),
        "scan_mesh_object_count": len(object_ids),
        "scan_mesh_object_ids": object_ids,
    }

pilot_categories = [
    line.strip()
    for line in pilot_list.read_text(encoding="utf-8").splitlines()
    if line.strip() and not line.lstrip().startswith("#")
]
missing_pilot = sorted(set(pilot_categories).difference(available))
if missing_pilot:
    raise RuntimeError(f"pilot categories absent from video/mesh overlap: {missing_pilot}")

# 仅用归档字节效率和类别数规划下载；故意不读取 reviewed1k split。
selected = set(pilot_categories)
candidate_count = sum(
    available[category]["scan_mesh_object_count"] for category in selected
)
ranked = sorted(
    (row for category, row in available.items() if category not in selected),
    key=lambda row: (
        row["remote_bytes"] / row["scan_mesh_object_count"],
        row["remote_bytes"],
        row["category"],
    ),
)
for row in ranked:
    if candidate_count >= target_candidates and len(selected) >= minimum_categories:
        break
    selected.add(row["category"])
    candidate_count += row["scan_mesh_object_count"]
if candidate_count < target_candidates or len(selected) < minimum_categories:
    raise RuntimeError(
        f"insufficient candidates/categories: {candidate_count}/{len(selected)}"
    )

categories = sorted(selected)
rows = []
for category in categories:
    row = dict(available[category])
    archive = archive_root / f"{category}.tar.gz"
    row["archive"] = str(archive)
    row["archive_present_bytes"] = archive.stat().st_size if archive.is_file() else 0
    row["already_downloaded"] = archive.is_file() and archive.stat().st_size > 0
    rows.append(row)

list_payload = "".join(f"{category}\n" for category in categories)
list_sha256 = hashlib.sha256(list_payload.encode("utf-8")).hexdigest()
payload = {
    "format": "pose_point_depth_mv.omni_real_video_download_plan.v2",
    "created_at_utc": datetime.now(timezone.utc).isoformat(),
    "target_scan_mesh_candidates": target_candidates,
    "minimum_category_count": minimum_categories,
    "category_count": len(rows),
    "scan_mesh_candidate_count": sum(row["scan_mesh_object_count"] for row in rows),
    "estimated_archive_bytes": sum(row["remote_bytes"] for row in rows),
    "estimated_missing_archive_bytes": sum(
        row["remote_bytes"] for row in rows if not row["already_downloaded"]
    ),
    "selection_uses_reviewed1k_exclusion": False,
    "selected_categories": categories,
    "category_list_sha256": list_sha256,
    "categories": rows,
    "selection_policy": (
        "retain audited pilot categories, then add categories by compressed bytes "
        "per local Scan.obj until >=1000 mesh candidates and >=48 categories; "
        "reviewed1k object IDs are intentionally not consulted before download"
    ),
    "scope_guard": (
        "Download plan only. Actual video IDs are unknown until archive inventory; "
        "old reviewed1k Omni objects are filtered only after that inventory."
    ),
}
plan_path.parent.mkdir(parents=True, exist_ok=True)
list_path.write_text(list_payload, encoding="utf-8")
plan_path.write_text(
    json.dumps(payload, indent=2, ensure_ascii=False) + "\n", encoding="utf-8"
)
print(json.dumps({k: payload[k] for k in (
    "category_count", "scan_mesh_candidate_count", "estimated_archive_bytes",
    "estimated_missing_archive_bytes", "selection_uses_reviewed1k_exclusion",
    "category_list_sha256", "scope_guard"
)}, indent=2, ensure_ascii=False))
PY
  fi

  sha256sum "${PLAN}" "${LIST}"
  echo "D1 passed; inspect estimated_missing_archive_bytes, then run D2"
)

        echo "[D2] reuse gzip-verified archive: ${ARCHIVE}"
      else
        echo "[D2] corrupt/partial archive preserved: ${ARCHIVE}"
        FAILURES=$((FAILURES + 1))
      fi
      continue
    fi
    if [ -e "${ARCHIVE}" ]; then
      echo "[D2] empty/non-regular target preserved: ${ARCHIVE}"
      FAILURES=$((FAILURES + 1))
      continue
    fi

    if ! "${PY}" -u pose_point_depth_mv/dataset_tools/download_omniobject3d.py \
      download \
      --dataset-repo omniobject3d/OmniObject3D-New \
      --source-path "${REMOTE_PATH}" \
      --target-path "${TARGET}"; then
      echo "[D2] download failed; preserve partial and continue: ${CATEGORY}"
      FAILURES=$((FAILURES + 1))
      continue
    fi
    if ! test -s "${ARCHIVE}" || ! gzip -t "${ARCHIVE}"; then
      echo "[D2] downloaded archive failed verification; preserve: ${ARCHIVE}"
      FAILURES=$((FAILURES + 1))
      continue
    fi
    echo "[D2] complete category=${CATEGORY} bytes=$(stat -c %s "${ARCHIVE}")"
  done < "${LIST}"

  echo "D2_failure_count=${FAILURES}"
  test "${FAILURES}" -eq 0 || exit 2
) 2>&1 | tee /data/zjr/omni_real_video500_download_20260804_v2/logs/D2_download.log

D2_RC=${PIPESTATUS[0]}

  OUTPUT=${ROOT}/audit/D3_DOWNLOAD_SUMMARY.json

  if [ -s "${OUTPUT}" ]; then
    "${PY}" -c 'import json,sys; p=json.load(open(sys.argv[1])); print(json.dumps({k:p[k] for k in ("passed","category_count","archive_complete_count","scan_object_count","total_archive_bytes","scope_guard")},indent=2,ensure_ascii=False)); assert p["passed"] is True' "${OUTPUT}"

import json
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path

plan_path = Path(sys.argv[1]).resolve()
list_path = Path(sys.argv[2]).resolve()
archive_root = Path(sys.argv[3]).resolve()
output = Path(sys.argv[4]).resolve()
plan = json.loads(plan_path.read_text(encoding="utf-8"))
categories = [line.strip() for line in list_path.read_text().splitlines() if line.strip()]
assert categories == plan["selected_categories"]
planned = {row["category"]: row for row in plan["categories"]}
rows = []
for index, category in enumerate(categories, start=1):
    archive = archive_root / f"{category}.tar.gz"
    present = archive.is_file() and archive.stat().st_size > 0
    gzip_rc = subprocess.run(["gzip", "-t", str(archive)], check=False).returncode if present else 96
    source = planned[category]
    rows.append({
        "category": category,
        "archive": str(archive),
        "archive_present": present,
        "archive_bytes": archive.stat().st_size if present else 0,
        "gzip_test_rc": int(gzip_rc),
        "scan_object_count": source["scan_mesh_object_count"],
        "scan_object_ids": source["scan_mesh_object_ids"],
    })
    print(f"[D3] {index}/{len(categories)} {category} gzip_rc={gzip_rc}", flush=True)

payload = {
    "format": "pose_point_depth_mv.omni_real_video_download_summary.v2",
    "created_at_utc": datetime.now(timezone.utc).isoformat(),
    "download_plan": str(plan_path),
    "selection_uses_reviewed1k_exclusion": False,
    "category_count": len(rows),
    "archive_complete_count": sum(
        row["archive_present"] and row["gzip_test_rc"] == 0 for row in rows
    ),
    "scan_object_count": sum(row["scan_object_count"] for row in rows),
    "total_archive_bytes": sum(row["archive_bytes"] for row in rows),
    "categories": rows,
    "scope_guard": (
        "Archive integrity only. Actual video object IDs and reviewed1k overlap "
        "are deliberately evaluated after this stage."
    ),
}
payload["passed"] = bool(rows) and (
    payload["archive_complete_count"] == payload["category_count"]
    and payload["scan_object_count"] >= 1000
)
output.parent.mkdir(parents=True, exist_ok=True)
output.write_text(json.dumps(payload, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
print(json.dumps({k: payload[k] for k in (
    "passed", "category_count", "archive_complete_count", "scan_object_count",
    "total_archive_bytes", "scope_guard"
)}, indent=2, ensure_ascii=False))
if not payload["passed"]:
    raise SystemExit(2)
PY
  fi
)

import hashlib
import json
import sys
from datetime import datetime, timezone
from pathlib import Path

train_path = Path(sys.argv[1]).resolve()
val_path = Path(sys.argv[2]).resolve()
audit_path = Path(sys.argv[3]).resolve()
scan_root = Path(sys.argv[4]).resolve()
output = Path(sys.argv[5]).resolve()
train = json.loads(train_path.read_text(encoding="utf-8"))
val = json.loads(val_path.read_text(encoding="utf-8"))
audit = json.loads(audit_path.read_text(encoding="utf-8"))
assert audit["passed"] is True

def omni_uids(manifest):
    return sorted({
        str(row["object_uid"])
        for row in manifest["samples"]
        if str(row.get("object_uid", "")).startswith("omni_")
    })

train_uids = omni_uids(train)
val_uids = omni_uids(val)
assert len(train_uids) == 119
assert len(val_uids) == 16
assert set(train_uids).isdisjoint(val_uids)

scan_by_id = {}
for object_root in scan_root.glob("*/*"):
    scan = object_root / "Scan" / "Scan.obj"
    if not scan.is_file():
        continue
    object_id = object_root.name
    if object_id in scan_by_id:
        raise RuntimeError(f"duplicate local Scan object ID: {object_id}")
    scan_by_id[object_id] = scan.resolve()

rows = []
for phase, values in (("reviewed1k_train", train_uids), ("reviewed1k_consumed_val", val_uids)):
    for uid in values:
        object_id = uid.removeprefix("omni_")
        scan = scan_by_id.get(object_id)
        if scan is None:
            raise RuntimeError(f"reviewed1k Omni object has no local Scan.obj: {uid}")
        rows.append({
            "source_phase": phase,
            "reviewed1k_object_uid": uid,
            "real_object_id": object_id,
            "scan_obj": str(scan),
        })

payload = {

    "train_manifest": str(train_path),
    "val_manifest": str(val_path),
    "split_audit": str(audit_path),
    "train_omni_count": len(train_uids),
    "consumed_val_omni_count": len(val_uids),
    "excluded_object_count": len(rows),
    "object_id_hash": object_id_hash,
    "objects": rows,
    "passed": len(rows) == 135 and len(set(object_ids)) == 135,
}
output.parent.mkdir(parents=True, exist_ok=True)
output.write_text(json.dumps(payload, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
print(json.dumps({k: payload[k] for k in (
    "passed", "train_omni_count", "consumed_val_omni_count",
    "excluded_object_count", "object_id_hash"
)}, indent=2))
if not payload["passed"]:
    raise SystemExit(2)
PY
  fi
)
