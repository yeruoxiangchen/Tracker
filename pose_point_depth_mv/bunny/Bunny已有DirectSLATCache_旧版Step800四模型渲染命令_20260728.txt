#!/usr/bin/env bash
#
# Bunny Racer：已有 materialized Direct-SLAT cache 时，运行旧版 step800 后半段，
# 并生成 Reference / Pixal3D / ReconViaGen stock / Direct Full 四列法线对比。
#
# 公平的实测 cache 用法（下面两个 REPLACE_* 必须替换为真实值，不能原样执行）：
#   export BUNNY_DIRECT_SLAT_CACHE=REPLACE_WITH_MATERIALIZED_BUNNY_CACHE_DIRECTORY
#   export BUNNY_DIRECT_SLAT_UID=REPLACE_WITH_EXACT_BUNNY_SEQUENCE_UID
#   export BUNNY_DIRECT_SLAT_INPUT_MODE=calibrated_real
#   BUNNY_GPU=5 bash \
#     pose_point_depth_mv/bunny/Bunny已有DirectSLATCache_旧版Step800四模型渲染命令_20260728.txt
#
# 若 sparse points/TM2W 是从 reference Mesh 模拟得到的，只能运行显式 oracle：
#   export BUNNY_DIRECT_SLAT_INPUT_MODE=reference_oracle
#   export APPROVE_BUNNY_REFERENCE_ORACLE=1
# 该模式会使用带 oracle 后缀的方法名和显示名，不与 calibrated_real 混淆。
#
# 重要：
# - 这不是 RGB-to-Mesh 完整重建入口，不负责创建 Direct-SS/Direct-SLAT cache。
# - 原始 Bunny 五张缩略图没有 K/T、点云或 TM2W，不能直接生成 Direct Full。
# - BUNNY_DIRECT_SLAT_CACHE 必须已物化，且包含指定 UID 的 seed 42 support。
# - step800 使用 checkpoint 自身的 legacy rollout 默认值；这里有意不传后来新增的
#   slat_delta_scale、slat_delta_rms_ratio_cap、slat_guided_delta_policy。
# - 已完成的三列结果只复用，不覆盖；四列结果使用新的 method_id/review_id。

cd /home/zjr/Tracker
set +e
set -o pipefail

PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
BUNNY_GPU=${BUNNY_GPU:-5}
PROJECT_TOOLS=/home/zjr/Tracker/pose_point_depth_mv/tools

OUT=/home/zjr/Tracker/pose_point_depth_mv/bunny/outputs/bunny_human_review_20260727_v1
PROTOCOL=${OUT}/protocol.json
LOG_ROOT=${OUT}/logs
ADAPTER_ROOT=${OUT}/adapters

PRETRAINED=Stable-X/trellis-vggt-v0-2
TRAIN_CACHE_MANIFEST=/data/ar_slat_direct_train_all_step000900_seed424344_lhslatv2_20260723/manifest.json
DIRECT_SS_CKPT=/home/zjr/Tracker/pose_point_depth_mv/outputs/direct_flow_p0_trainall_s1000_seed42_2gpu_bf16_v3/checkpoints/step_000900.pt
DIRECT_SLAT_CKPT=/home/zjr/Tracker/pose_point_depth_mv/outputs/direct_slat_step900support_trainall_s1000_seed42_2gpu_bf16_v3/checkpoints/step_000800.pt

EXPECTED_SS_SHA256=522d1618062e108c64ab71a97a328d07ce0374946f7d13228de883bb9ad0d2a1
EXPECTED_SLAT_SHA256=b13a89ddc6b4821665521ca8b0db116cf924b25993088f2a6bbcee4594acc69c

CACHE_ROOT=${BUNNY_DIRECT_SLAT_CACHE:-}
BUNNY_UID=${BUNNY_DIRECT_SLAT_UID:-}
INPUT_MODE=${BUNNY_DIRECT_SLAT_INPUT_MODE:-}
CACHE_MANIFEST=${CACHE_ROOT}/manifest.json

METHOD_ID=direct_ss900_slat800_full_legacy
DISPLAY_NAME="Direct-SS step900 + Direct-SLAT step800 Full"
REVIEW_ID=pixal3d_vs_reconviagen_vs_direct_ss900_slat800_full_legacy_v1
REFERENCE_AS_INPUT=0
ADAPTER_CONFIRM_ARGS=()

if [ "${INPUT_MODE}" = reference_oracle ]; then
  METHOD_ID=direct_ss900_slat800_full_legacy_oracle
  DISPLAY_NAME="Direct-SS900 + Direct-SLAT800 Full oracle/sensor simulation"
  REVIEW_ID=pixal3d_vs_reconviagen_vs_direct_ss900_slat800_full_legacy_oracle_v1
  REFERENCE_AS_INPUT=1
  ADAPTER_CONFIRM_ARGS+=(--allow_reference_as_input)
fi

ADAPTER=${ADAPTER_ROOT}/${METHOD_ID}.json
REVIEW_DIR=${OUT}/comparison/${REVIEW_ID}
METHOD_RESULT=${OUT}/methods/${METHOD_ID}/result.json

mkdir -p "${LOG_ROOT}" "${ADAPTER_ROOT}"


###############################################################################
# S0. CPU preflight：协议、模式、checkpoint、cache UID/provenance
###############################################################################

S0_RC=0
test -s "${PROTOCOL}" || S0_RC=96
test -x "${PROJECT_TOOLS}/ninja" || S0_RC=96
test -s "${TRAIN_CACHE_MANIFEST}" || S0_RC=96
test -s "${DIRECT_SS_CKPT}" || S0_RC=96
test -s "${DIRECT_SLAT_CKPT}" || S0_RC=96

if [ -z "${CACHE_ROOT}" ] || [ -z "${BUNNY_UID}" ]; then
  echo "S0 blocked: set BUNNY_DIRECT_SLAT_CACHE and BUNNY_DIRECT_SLAT_UID"
  S0_RC=96
elif [ ! -s "${CACHE_MANIFEST}" ]; then
  echo "S0 blocked: missing materialized Bunny cache manifest: ${CACHE_MANIFEST}"
  S0_RC=96
fi

if [ "${INPUT_MODE}" != calibrated_real ] && [ "${INPUT_MODE}" != reference_oracle ]; then
  echo "S0 blocked: set BUNNY_DIRECT_SLAT_INPUT_MODE=calibrated_real"
  echo "or explicitly use reference_oracle with APPROVE_BUNNY_REFERENCE_ORACLE=1"
  S0_RC=96
elif [ "${INPUT_MODE}" = reference_oracle ] && \
     [ "${APPROVE_BUNNY_REFERENCE_ORACLE:-0}" != 1 ]; then
  echo "S0 blocked: reference_oracle requires APPROVE_BUNNY_REFERENCE_ORACLE=1"
  S0_RC=96
fi

if [ "${S0_RC}" -eq 0 ]; then
  "${PY}" - \
    "${CACHE_MANIFEST}" \
    "${BUNNY_UID}" \
    "${DIRECT_SS_CKPT}" \
    "${EXPECTED_SS_SHA256}" \
    "${DIRECT_SLAT_CKPT}" \
    "${EXPECTED_SLAT_SHA256}" <<'PY'
import hashlib
import json
import sys
from pathlib import Path


def sha256(path):
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


manifest_path = Path(sys.argv[1]).resolve()
uid = sys.argv[2]
ss_checkpoint = Path(sys.argv[3]).resolve()
expected_ss_sha256 = sys.argv[4]
slat_checkpoint = Path(sys.argv[5]).resolve()
expected_slat_sha256 = sys.argv[6]

if sha256(ss_checkpoint) != expected_ss_sha256:
    raise RuntimeError("Direct-SS step900 checkpoint SHA-256 mismatch")
if sha256(slat_checkpoint) != expected_slat_sha256:
    raise RuntimeError("Direct-SLAT step800 checkpoint SHA-256 mismatch")

manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
if manifest.get("format") != "pose_point_depth_mv.direct_slat_cache.v1":
    raise RuntimeError(f"unexpected cache format={manifest.get('format')!r}")
if manifest.get("materialized") is not True:
    raise RuntimeError("Bunny Direct-SLAT cache is not materialized")

matches = [
    row
    for row in manifest.get("samples", [])
    if str(row.get("uid")) == uid and int(row.get("support_seed", -1)) == 42
]
if len(matches) != 1:
    raise RuntimeError(
        f"expected exactly one uid={uid!r}, seed=42 cache row; found {len(matches)}"
    )

config = manifest.get("config", {})
if int(config.get("expected_ss_step", -1)) != 900:
    raise RuntimeError("cache was not generated by the bound Direct-SS step900")
if str(config.get("ss_flow_checkpoint_sha256", "")) != expected_ss_sha256:
    raise RuntimeError("cache Direct-SS checkpoint binding differs from step900")

source_lifting = Path(str(config.get("source_lifting_manifest", "")))
if not source_lifting.is_absolute():
    source_lifting = manifest_path.parent / source_lifting
source_lifting = source_lifting.resolve()
if not source_lifting.is_file():
    raise RuntimeError(f"cache source lifting manifest is missing: {source_lifting}")
expected_lifting_sha256 = str(config.get("source_lifting_manifest_sha256", ""))
if expected_lifting_sha256 and sha256(source_lifting) != expected_lifting_sha256:
    raise RuntimeError("source lifting manifest SHA-256 mismatch")

lifting = json.loads(source_lifting.read_text(encoding="utf-8"))
lifting_matches = [
    row for row in lifting.get("samples", []) if str(row.get("uid")) == uid
]
if len(lifting_matches) != 1:
    raise RuntimeError(
        f"source lifting manifest must contain uid={uid!r} exactly once"
    )

print(
    json.dumps(
        {
            "status": "passed",
            "cache_manifest": str(manifest_path),
            "uid": uid,
            "support_seed": 42,
            "view_count": int(matches[0].get("view_count", matches[0].get("views", 0))),
            "source_lifting_manifest": str(source_lifting),
            "source_cache_manifest": lifting.get("source_cache_manifest", ""),
            "direct_ss_step": 900,
            "direct_slat_step": 800,
        },
        indent=2,
    )
)
PY
  S0_RC=$?
fi

echo "S0_RC=${S0_RC}"
if [ "${S0_RC}" -ne 0 ]; then
  echo "S0 failed; no GPU inference or four-column rendering was started"
  echo "The REPLACE_* values in this file are documentation placeholders."
  echo "A real materialized Bunny Direct-SLAT cache does not currently exist here."
fi
echo "current terminal remains open"
if [ "${S0_RC}" -ne 0 ]; then
  exit "${S0_RC}"
fi


###############################################################################
# S1. 冻结 adapter；运行旧版 step800 legacy Full rollout
###############################################################################

S1_RC=0
if [ "${S0_RC}" -ne 0 ]; then
  S1_RC=96
elif [ -s "${METHOD_RESULT}" ]; then
  echo "reuse immutable Direct step800 Bunny result: ${METHOD_RESULT}"
else
  "${PY}" - \
    "${ADAPTER}" \
    "${CACHE_MANIFEST}" \
    "${BUNNY_UID}" \
    "${TRAIN_CACHE_MANIFEST}" \
    "${DIRECT_SS_CKPT}" \
    "${DIRECT_SLAT_CKPT}" \
    "${REFERENCE_AS_INPUT}" \
    "${INPUT_MODE}" <<'PY'
import json
import sys
from pathlib import Path

adapter_path = Path(sys.argv[1]).resolve()
cache_manifest = Path(sys.argv[2]).resolve()
uid = sys.argv[3]
train_manifest = Path(sys.argv[4]).resolve()
ss_checkpoint = Path(sys.argv[5]).resolve()
slat_checkpoint = Path(sys.argv[6]).resolve()
reference_as_input = bool(int(sys.argv[7]))
input_mode = sys.argv[8]

cache = json.loads(cache_manifest.read_text(encoding="utf-8"))
source_lifting = Path(cache["config"]["source_lifting_manifest"])
if not source_lifting.is_absolute():
    source_lifting = cache_manifest.parent / source_lifting
source_lifting = source_lifting.resolve()
lifting = json.loads(source_lifting.read_text(encoding="utf-8"))
source_cache = Path(str(lifting.get("source_cache_manifest", "")))
if source_cache and not source_cache.is_absolute():
    source_cache = source_lifting.parent / source_cache
source_cache = source_cache.resolve() if str(source_cache) else None

input_paths = [str(cache_manifest), str(source_lifting)]
if source_cache is not None and source_cache.is_file():
    input_paths.append(str(source_cache))

payload = {
    "format": "pose_point_depth_mv.bunny_command_adapter.v1",
    "adapter_id": "direct_ss900_direct_slat800_legacy_bunny_v1",
    "cwd": "/home/zjr/Tracker",
    "environment": {
        "HF_HUB_OFFLINE": "1",
        "TRANSFORMERS_OFFLINE": "1",
        "ATTN_BACKEND": "flash_attn",
        "SPCONV_ALGO": "native",
        "MPLCONFIGDIR": "/tmp/matplotlib",
        "NUMBA_CACHE_DIR": "/tmp/numba_cache",
        "TORCH_EXTENSIONS_DIR": "/tmp/torch_extensions",
        "PYTORCH_CUDA_ALLOC_CONF": "expandable_segments:True",
    },
    "command": [
        "/home/zjr/anaconda3/envs/reconviagen/bin/python",
        "-u",
        "-m",
        "pose_point_depth_mv.export_direct_slat_mesh_pairs",
        "--cache_manifest",
        str(cache_manifest),
        "--checkpoint",
        str(slat_checkpoint),
        "--train_cache_manifest",
        str(train_manifest),
        "--output_dir",
        "{output_dir}/native_run",
        "--pretrained",
        "Stable-X/trellis-vggt-v0-2",
        "--joint_seeds",
        "42",
        "--uids",
        uid,
        "--max_objects",
        "1",
        "--steps",
        "25",
        "--cfg_strength",
        "5.0",
        "--cfg_interval",
        "0.5,1.0",
        "--rescale_t",
        "3.0",
        "--support_scale",
        "1.0",
        "--surface_samples",
        "20000",
        "--bootstrap_samples",
        "5000",
        "--amp_dtype",
        "bf16",
    ],
    "expected_mesh": (
        "native_run/mesh_pairs/obj_0000_seed_42/full/mesh_canonical.obj"
    ),
    "input_view_indices": [0, 1, 2, 3, 4],
    "checkpoint_paths": [str(ss_checkpoint), str(slat_checkpoint)],
    "input_paths": input_paths,
    "code_paths": [
        "/home/zjr/Tracker/pose_point_depth_mv/export_direct_slat_mesh_pairs.py",
        "/home/zjr/Tracker/pose_point_depth_mv/direct_flow.py",
        "/home/zjr/Tracker/pose_point_depth_mv/direct_slat_flow.py",
    ],
    "input_contract": {
        "images": "five frozen Bunny RGBA views",
        "camera_sparse_tm2w_source": str(source_lifting),
        "direct_slat_cache": str(cache_manifest),
        "cache_uid": uid,
        "support_seed": 42,
        "input_mode": input_mode,
        "target_mesh_role": (
            "evaluation/visual reference only unless input_mode=reference_oracle"
        ),
    },
    "reference_mesh_declared_as_model_input": reference_as_input,
    "notes": [
        "Direct-SS step900 corrected support and Direct-SLAT step800 Full.",
        "Legacy step800 checkpoint defaults are used without later bounded wrappers.",
        "The fourth column is a geometry-only normal rendering.",
    ],
}

serialized = json.dumps(payload, indent=2) + "\n"
if adapter_path.is_file():
    existing = adapter_path.read_text(encoding="utf-8")
    if existing != serialized:
        raise RuntimeError(
            f"immutable adapter differs; choose a new method ID: {adapter_path}"
        )
else:
    adapter_path.parent.mkdir(parents=True, exist_ok=True)
    temporary = adapter_path.with_name(adapter_path.name + ".tmp")
    temporary.write_text(serialized, encoding="utf-8")
    temporary.replace(adapter_path)
print(adapter_path)
PY
  S1_ADAPTER_RC=$?

  if [ "${S1_ADAPTER_RC}" -ne 0 ]; then
    S1_RC=${S1_ADAPTER_RC}
  else
    CUDA_VISIBLE_DEVICES="${BUNNY_GPU}" \
    "${PY}" -u -m pose_point_depth_mv.bunny_review.trained_adapter command \
      --protocol "${PROTOCOL}" \
      --adapter "${ADAPTER}" \
      --method_id "${METHOD_ID}" \
      --display_name "${DISPLAY_NAME}" \
      "${ADAPTER_CONFIRM_ARGS[@]}" \
      2>&1 | tee "${LOG_ROOT}/${METHOD_ID}.log"
    S1_RC=${PIPESTATUS[0]}
  fi
fi

printf '%s\n' "${S1_RC}" > "${LOG_ROOT}/${METHOD_ID}.log.exit_code"
echo "S1_RC=${S1_RC}"
echo "method result: ${METHOD_RESULT}"
echo "current terminal remains open"


###############################################################################
# S2. 四路状态必须全部 complete
###############################################################################

S2_RC=0
if [ "${S1_RC}" -ne 0 ]; then
  S2_RC=96
else
  "${PY}" -m pose_point_depth_mv.bunny_review.status \
    --protocol "${PROTOCOL}" \
    --methods "reference,pixal3d,reconviagen_stock,${METHOD_ID}" \
    --require_complete
  S2_RC=$?
fi

echo "S2_RC=${S2_RC}"
echo "current terminal remains open"


###############################################################################
# S3. 与现有图相同的统一法线渲染，四种结果并列
###############################################################################

S3_RC=0
if [ "${S2_RC}" -ne 0 ]; then
  echo "S3 blocked: four requested methods are not all complete"
  S3_RC=96
else
  PATH="${PROJECT_TOOLS}:${PATH}" \
  CUDA_VISIBLE_DEVICES="${BUNNY_GPU}" \
  ATTN_BACKEND=flash_attn \
  SPCONV_ALGO=native \
  MPLCONFIGDIR=/tmp/matplotlib \
  NUMBA_CACHE_DIR=/tmp/numba_cache \
  TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
  "${PY}" -u -m pose_point_depth_mv.bunny_review.finalize \
    --protocol "${PROTOCOL}" \
    --methods "reference,pixal3d,reconviagen_stock,${METHOD_ID}" \
    --review_id "${REVIEW_ID}" \
    --device cuda \
    --render_frames 72 \
    --render_resolution 320 \
    --contact_frames 6 \
    --fps 12 \
    2>&1 | tee "${LOG_ROOT}/${REVIEW_ID}.log"
  S3_RC=${PIPESTATUS[0]}
fi

printf '%s\n' "${S3_RC}" > "${LOG_ROOT}/${REVIEW_ID}.log.exit_code"
echo "S3_RC=${S3_RC}"
if [ "${S3_RC}" -eq 0 ]; then
  echo "HTML: ${REVIEW_DIR}/index.html"
  echo "四列并排图: ${REVIEW_DIR}/normal_contact_sheet.png"
  echo "四列并排视频: ${REVIEW_DIR}/normal_side_by_side.mp4"
else
  echo "四列比较尚未生成；检查 S0--S3 的首个非零 RC"
fi
echo "current terminal remains open"
