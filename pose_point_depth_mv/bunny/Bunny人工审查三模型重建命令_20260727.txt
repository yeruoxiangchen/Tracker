#!/usr/bin/env bash
#
# Bunny Racer 人工审查：Pixal3D / ReconViaGen stock / 当前训练模型
#
# 重要：
# 1. pose_point_depth_mv/bunny/ 保存原始模型资产、运行命令和审查结果；
# 2. 可复用推理代码仍位于 pose_point_depth_mv/bunny_review/；
# 3. Pixal3D 使用 view 0；ReconViaGen/训练模型默认使用 view 0--4；
# 4. 五张实拍缩略图没有 K/T、深度、稀疏点云或 TM2W；
# 5. Direct-SS/Direct-SLAT 的 .pt 不能传给 ReconViaGen 原生 LoRA loader。
#
# 每块命令都保留当前终端，并输出独立 RC。不要删除 partial output；改新 method_id
# 或 review_id 后重跑，便于审计。

cd /home/zjr/Tracker

PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
PIXAL_PY=/home/zjr/anaconda3/envs/pixal3d/bin/python
OUT=/home/zjr/Tracker/pose_point_depth_mv/bunny/outputs/bunny_human_review_20260727_v1
PROTOCOL=${OUT}/protocol.json

NAF_ROOT=/data/zjr/models/valeoai_NAF_37f2dfc180f2de53d98bd601109c0da0dd6b0f43
NAF_REPO=${NAF_ROOT}/repo
NAF_CKPT=${NAF_ROOT}/naf_release.pth
NAF_SOURCE_MANIFEST=${NAF_ROOT}/source_manifest.sha256

PRETRAINED=Stable-X/trellis-vggt-v0-2
PIXAL_GPU=1
RECON_GPU=2
TRAINED_GPU=3
RENDER_GPU=1


###############################################################################
# A0. 冻结 Bunny 五视图、白背景 alpha、参考 Mesh 与协议
###############################################################################

A0_RC=0
if [ -s "${PROTOCOL}" ]; then
  echo "reuse immutable Bunny protocol: ${PROTOCOL}"
  "${PY}" -m pose_point_depth_mv.bunny_review.status \
    --protocol "${PROTOCOL}" \
    --methods reference,pixal3d,reconviagen_stock,trained_full \
    || A0_RC=$?
elif [ -e "${OUT}" ]; then
  echo "A0 blocked: partial output exists without protocol: ${OUT}"
  A0_RC=98
else
  "${PY}" -u -m pose_point_depth_mv.bunny_review.prepare \
    --bunny_root /home/zjr/Tracker/pose_point_depth_mv/bunny \
    --output_dir "${OUT}" \
    --view_indices 0,1,2,3,4 \
    --single_view_index 0 \
    --white_threshold 245 \
    || A0_RC=$?
fi
echo "A0_RC=${A0_RC}"
echo "current terminal remains open"


###############################################################################
# A1. 原版 Pixal3D：单视图 view 0
###############################################################################

A1_RC=0
if [ ! -s "${PROTOCOL}" ]; then
  echo "A1 blocked: run A0 first"
  A1_RC=96
elif [ ! -d "${NAF_REPO}" ] || [ ! -s "${NAF_CKPT}" ] || \
     [ ! -s "${NAF_SOURCE_MANIFEST}" ]; then
  echo "A1 blocked: pinned local NAF is incomplete: ${NAF_ROOT}"
  A1_RC=96
else
  CUDA_VISIBLE_DEVICES="${PIXAL_GPU}" \
  HF_HUB_OFFLINE=1 \
  TRANSFORMERS_OFFLINE=1 \
  ATTN_BACKEND=flash_attn \
  SPCONV_ALGO=native \
  PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
  "${PIXAL_PY}" -u -m pose_point_depth_mv.bunny_review.infer_pixal3d \
    --protocol "${PROTOCOL}" \
    --method_id pixal3d \
    --display_name "Pixal3D single-view" \
    --model_path TencentARC/Pixal3D \
    --naf_repo "${NAF_REPO}" \
    --naf_checkpoint "${NAF_CKPT}" \
    --naf_source_manifest "${NAF_SOURCE_MANIFEST}" \
    --device cuda \
    --low_vram \
    --fixed_fov 0.857556 \
    --resolution 1024 \
    --sampling_steps 12 \
    --seed 42 \
    || A1_RC=$?
fi
echo "A1_RC=${A1_RC}"
echo "current terminal remains open"


###############################################################################
# A2. ReconViaGen stock：五视图 0--4
###############################################################################

A2_RC=0
if [ ! -s "${PROTOCOL}" ]; then
  echo "A2 blocked: run A0 first"
  A2_RC=96
else
  CUDA_VISIBLE_DEVICES="${RECON_GPU}" \
  HF_HUB_OFFLINE=1 \
  TRANSFORMERS_OFFLINE=1 \
  ATTN_BACKEND=flash_attn \
  SPCONV_ALGO=native \
  MPLCONFIGDIR=/tmp/matplotlib \
  NUMBA_CACHE_DIR=/tmp/numba_cache \
  TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
  PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
  "${PY}" -u -m pose_point_depth_mv.bunny_review.infer_reconviagen \
    --protocol "${PROTOCOL}" \
    --method_id reconviagen_stock \
    --display_name "ReconViaGen stock 5-view" \
    --pretrained "${PRETRAINED}" \
    --device cuda \
    --low_vram \
    --seed 42 \
    --ss_steps 30 \
    --slat_steps 12 \
    --multiimage_algo multidiffusion \
    --mesh_simplify 0.95 \
    --texture_size 1024 \
    || A2_RC=$?
fi
echo "A2_RC=${A2_RC}"
echo "current terminal remains open"

###############################################################################

###############################################################################

###############################################################################
# A4. 状态：确认四个方法均 complete 后再渲染
###############################################################################

"${PY}" -m pose_point_depth_mv.bunny_review.status \
  --protocol "${PROTOCOL}" \
  --methods reference,pixal3d,reconviagen_stock,trained_full \
  --require_complete


###############################################################################
# A5. 统一固定视角法线渲染、并排 MP4/contact sheet/HTML
###############################################################################

A5_RC=0
CUDA_VISIBLE_DEVICES="${RENDER_GPU}" \
ATTN_BACKEND=flash_attn \
SPCONV_ALGO=native \
MPLCONFIGDIR=/tmp/matplotlib \
NUMBA_CACHE_DIR=/tmp/numba_cache \
TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
"${PY}" -u -m pose_point_depth_mv.bunny_review.finalize \
  --protocol "${PROTOCOL}" \
  --methods reference,pixal3d,reconviagen_stock,trained_full \
  --review_id pixal_stock_current_v1 \
  --device cuda \
  --render_frames 72 \
  --render_resolution 320 \
  --contact_frames 6 \
  --fps 12 \
  || A5_RC=$?

echo "A5_RC=${A5_RC}"
echo "HTML: ${OUT}/comparison/pixal_stock_current_v1/index.html"
echo "并排图: ${OUT}/comparison/pixal_stock_current_v1/normal_contact_sheet.png"
