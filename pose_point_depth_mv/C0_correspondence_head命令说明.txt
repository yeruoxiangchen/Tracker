# C0 Explicit View Correspondence Head 命令说明
# 日期：2026-07-17
#
# 一、当前模型结构
#
# 输入（来自现有lifting cache，不重建数据）：
#
#   每个view、每个16^3 voxel：
#     sampled VGGT/DINO visual feature [V,4096,3072]
#     pose/ray/depth geometry          [V,4096,18]
#     correct branch view weight      [V,4096]
#
#   visual encoder ─┐
#                   ├─ per-view joint interaction
#   geometry encoder┘       │
#                           ▼
#              same-voxel symmetric view pairs
#                           │
#                           ▼
#               per-pair logit -> per-voxel score
#                           │
#                           ▼
#                  fixed-weight sample score
#
# 该阶段不读取x_t、stock velocity或GT latent，不加载ReconViaGen/SS Flow，
# 也不训练Flow LoRA。唯一目标是验证正确image-pose-depth绑定是否可识别。
#
# 训练控制：
#   pose_cyclic1
#   depth_view_cyclic1
#   visual_view_cyclic1
#
# 未见过的保留控制：
#   pose_reverse
#   depth_spatial
#
# 所有correct/control共用correct view weight与support，避免通过覆盖范围变化分类。
# 训练损失包含sample BCE、paired ranking、hard-negative ranking和弱voxel BCE。
#
# 二、执行约定
#
# 1. 按V7.1 -> V7.2 -> V7.3 -> V7.4 -> V7.5 -> V7.6执行。
# 2. 每个命令块最终exit 0，不会因为科学FAIL关闭终端。
# 3. 真实程序状态写入*.exit_code；科学判定写入*.decision_code。
# 4. output已存在时不覆盖，写98并停止该块。
# 5. V7.6只有train16和fresh48都PASS才允许进入多seed。

# ========================================================================
# V7.6 paired gate与高信号结果提取
# ========================================================================
(
  set -uo pipefail
  cd /home/zjr/Tracker || exit 0

  RUN=pose_point_depth_mv/outputs/c0_correspondence_train16_s400_seed42_bf16_20260717

  /home/zjr/anaconda3/envs/reconviagen/bin/python -c '
import json
import sys
from pathlib import Path

root = Path(sys.argv[1])
reports = {}
for split in ("train16", "fresh48"):
    path = root / f"eval_{split}_all_controls" / "report.json"
    if not path.is_file():
        print(split, "MISSING", path)
        continue
    report = json.loads(path.read_text(encoding="utf-8"))
    reports[split] = report
    print("\n====", split)
    print("passed=", report["passed"])
    print("correct score=", report["correct_score"]["object"])
    print("view permutation diff=", report["view_permutation_max_abs_diff"])
    for mode, row in report["controls"].items():
        margin = row["correct_margin"]
        print(
            mode,
            "role=", row["role"],
            "mean/median/win=",
            margin["object"]["mean"],
            margin["object"]["median"],
            margin["object_win_rate"],
            "CI=", margin["object_bootstrap_95_ci"],
        )
paired = (
    set(reports) == {"train16", "fresh48"}
    and all(report["passed"] for report in reports.values())
)
print("\npaired train/fresh decision=", "PASS" if paired else "FAIL")
raise SystemExit(0 if paired else 2)
' "${RUN}" 2>&1 | tee "${RUN}/paired_result_summary.txt"
  CODE=${PIPESTATUS[0]}
  echo "${CODE}" > "${RUN}.decision_code"
  echo "V7.6 paired decision code: ${CODE}"
  exit 0
)


# 三、裁决
#
# 只有V7.6 decision code=0，才进入seed43/44复现。
#
# PASS要求train16和fresh48同时满足：
#   correct score均值 >= 0；
#   每个训练control和两个未见held-out control的score均值 <= 0；
#   correct-control mean/median > 0；
#   object win rate >= 65%；
#   object bootstrap 95% CI下界 > 0；
#   common view permutation分数差 <= 1e-5。
#
# 若train PASS、fresh FAIL：说明head记忆训练对象或corruption统计，停止C0 V1。
# 若训练controls PASS、held-out controls FAIL：说明head记住了特定corruption模式，停止C0 V1。
# 只有三个seed均在fresh48通过，才进入C1局部occupancy/decoder-logit监督；此时仍不启用Flow LoRA。
