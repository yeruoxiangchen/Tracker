Objaverse 5K：剩余半集 Wide-Orbit v2 构建与双根后处理
=========================================================

目标
----

1. 冻结旧 render v1 的 shard_000-007，保留为 limited-view 数据。
2. 停止旧 render worker，旧根中 shard_008-011 的半成品只保留、不读取、不删除。
3. 用同一份冻结 source plan 的 shard_008-015 构建 wide-orbit v2：
   - 24 个连续 AR-like 候选视图；
   - 方位跨度随机 285-330 度；
   - 按物体中心方位角确定性选 8 张；
   - 硬门 coverage >= 240 度、maximum gap <= 120 度；
   - 继续执行 mask、bbox、Blender bounds 和 projection-support 门。
4. CPU watcher 同时读取旧根 000-007 和新根 008-015；DINO watcher继续复用原 work root。

新根与旧根严格分离，任何命令都不会删除旧数据。

A1. 冻结旧半集并停止旧 render worker
=====================================

# 先停止旧 worker 冻结旧根，再确认恰好只有 shard_000-007 完成。
# 如果数字不是8，不要继续；先检查新增完成的 shard，
# 再调整新半集边界，避免同一 shard 在两个根重复。

(
  set -euo pipefail
  OLD_RENDER=/data/zjr/objaverse5k_single_subject_render512_20260810_v1

  for WORKER_INDEX in 0 1 2 3; do
    systemctl --user stop \
      "objaverse5k-render-w${WORKER_INDEX}-20260810-v1.service" || true
  done

  for WORKER_INDEX in 0 1 2 3; do
    test "$(systemctl --user is-active \
      "objaverse5k-render-w${WORKER_INDEX}-20260810-v1.service" || true)" != active
  done

  OLD_READY=$(find "${OLD_RENDER}/objaverse" \
    -name _WORKER_COMPLETE.json | wc -l)
  echo "old completed shards: ${OLD_READY}"
  test "${OLD_READY}" -eq 8

  for INDEX in $(seq 0 7); do
    test -f "${OLD_RENDER}/objaverse/shard_$(printf '%03d' "${INDEX}")/_WORKER_COMPLETE.json"
  done

  systemctl --user --no-pager --plain list-units \
    'objaverse5k-render-w*-20260810-v1.service'
)

=================================================

(
  set -euo pipefail
  cd /home/zjr/Tracker

  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  PLAN=/data/zjr/objaverse5k_single_subject_stratified_20260810_v1/build_plan.json

        NUMBA_CACHE_DIR=/tmp/numba_cache \
        TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
        PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
        "${PY}" -u \
          pose_point_depth_mv/dataset_tools/run_mixed_multiview_render_worker.py \
            --source_plan "${PLAN}" \
            --render_root "${RENDER}" \
            --preview_root "${PREVIEW}" \
            --log_root "${LOG_ROOT}" \
            --worker_index "${WORKER_INDEX}" \
            --num_workers 4 \
            --sources objaverse \
            --python "${PY}" \
            --blender_path /tmp/blender-3.0.1-linux-x64/blender \
            --blender_engine CYCLES \
            --blender_samples 16 \
            --blender_bounds_tolerance 0.001 \
            --blender_cycles_device CUDA \
            --seed 20260811 \
            --sequences_per_object 2 \

            --shard_start 8 \
            --shard_end 16 \

A6. Wide 半集完成后的覆盖合同审计
=================================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
WIDE=/data/zjr/objaverse5k_single_subject_render512_wideorbit_20260811_v2

"${PY}" - "${WIDE}" <<'PY'
import glob, json, os, statistics, sys
root = sys.argv[1]
manifests = sorted(glob.glob(os.path.join(root, "objaverse/shard_*/manifest.json")))
assert len(manifests) == 8, manifests
coverage, gaps, samples, failures = [], [], 0, 0
for path in manifests:
    payload = json.load(open(path, encoding="utf-8"))
    assert payload["build_config"]["frame_selection_policy"] == "object_azimuth_balanced"
    for row in payload["samples"]:
        stats = row["frame_selection_stats"]
        coverage.append(float(stats["azimuth_coverage_degrees"]))
        gaps.append(float(stats["maximum_azimuth_gap_degrees"]))
        assert coverage[-1] >= 240.0, (path, row["uid"], stats)
        assert gaps[-1] <= 120.0, (path, row["uid"], stats)
    samples += len(payload["samples"])
    failures += len(payload["failures"])
print(json.dumps({
    "passed": True,
    "manifests": len(manifests),
    "samples": samples,
    "failures": failures,
    "coverage_min_median_max": [min(coverage), statistics.median(coverage), max(coverage)],
    "max_gap_min_median_max": [min(gaps), statistics.median(gaps), max(gaps)],
}, indent=2))
PY
