run_logged() {
  LOG_PATH=$1
  shift
  set +e
  "$@" 2>&1 | tee "${LOG_PATH}"
  COMMAND_RC=${PIPESTATUS[0]}
  printf '%s\n' "${COMMAND_RC}" > "${LOG_PATH}.exit_code"
  return "${COMMAND_RC}"
}

"${PY}" - "${WORK}" "${N3_REPORT}" <<'PY'
import glob
import json
import math
import os
import sys
from pathlib import Path

root = Path(sys.argv[1])
output = Path(sys.argv[2])
expected_paths = [
    root / f"train4_repeat{repeat}_seed42_v2_fix1" / "report.json"
    for repeat in range(3)
]
paths = [path for path in expected_paths if path.is_file()]
missing_paths = [str(path) for path in expected_paths if not path.is_file()]
rows = []
for path in paths:
    report = json.load(open(path, encoding="utf-8"))
    history = report.get("history", [])
    audit = report.get("mechanism_training_audit", {})
    counts = audit.get("counts", {})
    finite = all(
        math.isfinite(float(step[key]))
        for step in history
        for key in (
            "loss", "flow_loss", "stock_loss", "grad_norm",
            "support_adapter_grad_norm", "lora_grad_norm",
            "raw_delta_ratio_max", "raw_delta_excess_loss",
        )
    )
    checks = {
        "completed_step4": report.get("completed") is True and report.get("step") == 4,
        "history_count4": len(history) == 4,
        "finite": finite,
        "support_grad_nonzero": max(float(x["support_adapter_grad_norm"]) for x in history) > 0,
        "lora_grad_nonzero": max(float(x["lora_grad_norm"]) for x in history) > 0,
        "wrong_support_executed": int(counts.get("wrong_support_events", 0)) > 0,
        "support_dropout_executed": int(counts.get("support_dropout_events", 0)) > 0,
        "post_stock_fallback": report.get("post_training_stock_equivalence", {}).get("passed") is True,
    }
    rows.append({
        "path": str(path),
        "checks": checks,
        "mechanism_counts": counts,
        "mechanism_means": audit.get("means", {}),
        "rank_pass_rate": audit.get("rank_pass_rate"),
        "passed": all(checks.values()),
    })

payload = {
    "format": "pose_point_depth_mv.direct_slat_v2_train4_repeat_audit.v2",
    "expected_runs": 3,
    "found_runs": len(paths),
    "missing_reports": missing_paths,
    "runs": rows,
    "diagnosis": (
        "run the complete N2 block before N3"
        if missing_paths
        else "all N2 reports found; inspect per-run checks"
    ),
    "passed": not missing_paths and len(rows) == 3 and all(row["passed"] for row in rows),
}
output.write_text(json.dumps(payload, indent=2), encoding="utf-8")
print(json.dumps(payload, indent=2))
raise SystemExit(0 if payload["passed"] else 3)
PY
N3_RC=$?
