    pose_point_depth_mv/dataset_tools/freeze_mixed10k_source_inputs.py \

      --objaverse_manifest "${OBJ_ROOT}/manifest_5000_30000.json" \

      --omni_archive_root "${OMNI_ARCHIVE_ROOT}" \
      --omni_exclude_remote_path /raw/raw_scans/package.tar.gz \

elif [ -e "${SOURCE_INPUTS}" ]; then

  echo "${SOURCE_INPUTS}"

  "${PY}" -u \
    pose_point_depth_mv/dataset_tools/freeze_mixed10k_source_inputs.py \
      --objaverse_uids_file "${OBJ_ROOT}/uids.txt" \
      --objaverse_manifest "${OBJ_ROOT}/manifest_0_5000.json" \
      --objaverse_manifest "${OBJ_ROOT}/manifest_5000_30000.json" \
      --objaverse_candidate_count 11546 \
      --selection_seed 20260727 \
      --omni_remote_paths_file "${WORK}/manifests/omni_raw_scans_remote_paths.txt" \
      --omni_archive_root "${OMNI_ARCHIVE_ROOT}" \
      --omni_exclude_remote_path /raw/raw_scans/package.tar.gz \
      --expected_omni_archives 215 \
      --omni_exclusion_reason \
        "OpenXLab HTTP 405 after three attempts; recorded in omni_raw_scans_filewise_state.json" \
      --output_dir "${SOURCE_INPUTS}"
  INPUT_RC=$?

echo "INPUT_RC=${INPUT_RC}"

    'import json,sys

###############################################################################
# S2. 后台启动 Omni 安全解包和 mixed source plan 冻结（CPU/磁盘 IO）
###############################################################################

cd /home/zjr/Tracker

PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
WORK=/data/zjr/dataset10k_20260726
OBJ_ROOT=/data/zjr/Objaverse
OMNI_ARCHIVE_ROOT=/data/zjr/OmniObject3D/raw/omniobject3d___OmniObject3D-New/raw/raw_scans
SOURCE_INPUTS=${WORK}/source_inputs_obj11546_omni215_v1
OMNI_EXTRACT_ROOT=/data/zjr/OmniObject3D/raw_scans_extracted_omni215_v1
SOURCE_FREEZE=${WORK}/source_freeze_obj11546_omni215_v1
UNIT=tracker-mixed10k-source-freeze-obj11546-omni215-v1
SOURCE_START_RC=97

if [ -s "${SOURCE_FREEZE}/source_report.json" ] && \
   [ -s "${SOURCE_FREEZE}/build_plan.json" ]; then
  echo "reuse complete mixed source freeze: ${SOURCE_FREEZE}"
  SOURCE_START_RC=0
elif [ -e "${SOURCE_FREEZE}" ]; then
  echo "S2 blocked: immutable source-freeze directory exists without complete reports:"
  echo "${SOURCE_FREEZE}"
  SOURCE_START_RC=98
elif [ ! -s "${SOURCE_INPUTS}/input_report.json" ]; then
  echo "S2 blocked: first run S1; input_report.json is missing"
  SOURCE_START_RC=96
elif systemctl --user is-active --quiet "${UNIT}"; then
  echo "already active: ${UNIT}"
  SOURCE_START_RC=0
else
  systemctl --user reset-failed "${UNIT}" >/dev/null 2>&1 || true
  systemd-run --user \
    --unit="${UNIT}" \
    --collect \
    --property=WorkingDirectory=/home/zjr/Tracker \
    --property=Nice=5 \
    /usr/bin/env \
      PYTHONUNBUFFERED=1 \
    "${PY}" -u \
      pose_point_depth_mv/dataset_tools/prepare_mixed_mesh10k_sources.py \
        --sources both \
        --objaverse_uids_file "${SOURCE_INPUTS}/objaverse_candidate_uids.txt" \
        --objaverse_limit 11546 \
        --objaverse_manifest "${OBJ_ROOT}/manifest_0_5000.json" \
        --objaverse_manifest "${OBJ_ROOT}/manifest_5000_30000.json" \
        --min_objaverse_coverage 1.0 \
        --objaverse_shards 16 \
        --omni_remote_paths_file "${SOURCE_INPUTS}/omni_remote_paths.txt" \
        --omni_archive_root "${OMNI_ARCHIVE_ROOT}" \
        --omni_extract_root "${OMNI_EXTRACT_ROOT}" \
        --expected_omni_archives 215 \
        --min_omni_objects 4000 \
        --min_omni_obj_bytes 1024 \
        --omni_extract_workers 2 \
        --omni_shards 8 \
        --output_dir "${SOURCE_FREEZE}"
  SOURCE_START_RC=$?
fi

echo "SOURCE_START_RC=${SOURCE_START_RC}"
echo "unit=${UNIT}"
echo "current terminal remains open"

# 说明：
#   - 只开 2 个解包进程，避免对 /data 造成过高并发 IO。
#   - 不使用 GPU；可以和 GPU 训练/渲染并行，但仍会占用磁盘带宽。
#   - 已完整发布的 Omni 类别带 _EXTRACT_REPORT.json，重启时会校验后复用。
#   - 中途失败时保留证据，不会把半成品发布为正式 SOURCE_FREEZE。

    'import json,sys

  "passed":r["passed"],
  "sources":r["sources"],
  "formal_mixed_source_plan":r["formal_mixed_source_plan"],

  "objaverse_coverage":r["objaverse"]["coverage"],

  "omni_shards":p["omni"]["shard_count"],
  "build_plan_sha256":r["build_plan_sha256"],
  "hard_guards":r["hard_guards"],
}

  journalctl --user-unit="${UNIT}" -n 40 --no-pager 2>/dev/null || true
