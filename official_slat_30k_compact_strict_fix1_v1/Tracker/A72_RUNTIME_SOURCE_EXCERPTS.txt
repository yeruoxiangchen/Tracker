    46	def parse_indices(spec: str, size: int) -> list[int]:
    47	    text = str(spec).strip().lower()
    48	    if text in {"", "all"}:
    49	        return list(range(size))
    50	    output: list[int] = []
    51	    for item in text.split(","):
    52	        item = item.strip()
    53	        if not item:
    54	            continue
    55	        if "-" in item:
    56	            start, end = item.split("-", 1)
    57	            output.extend(range(int(start), int(end) + 1))
    58	        else:
    59	            output.append(int(item))
    60	    bad = [index for index in output if index < 0 or index >= size]
    61	    if bad:
    62	        raise IndexError(f"indices outside size={size}: {bad}")

   240	class ObjectBalancedDistributedSampler(Sampler[int]):
   241	    """Draw one sequence per object per epoch, then shard objects across ranks."""
   242	
   243	    def __init__(
   244	        self,
   245	        rows: list[dict[str, Any]],
   246	        *,
   247	        num_replicas: int,
   248	        rank: int,
   249	        seed: int,
   250	    ) -> None:
   251	        self.num_replicas = int(num_replicas)
   252	        self.rank = int(rank)
   253	        self.seed = int(seed)
   254	        self.epoch = 0
   255	        self.by_object: dict[str, list[int]] = {}
   256	        for index, row in enumerate(rows):
   257	            object_uid = str(row.get("object_uid", row.get("uid", "")))
   258	            if not object_uid:
   259	                raise ValueError("training cache contains an empty object UID")
   260	            self.by_object.setdefault(object_uid, []).append(index)
   261	        if not self.by_object:
   262	            raise ValueError("object-balanced sampler received no objects")
   263	        self.num_samples = math.ceil(len(self.by_object) / self.num_replicas)
   264	        self.total_size = self.num_samples * self.num_replicas

   267	        self.epoch = int(epoch)

   269	    def __iter__(self):
   270	        generator = torch.Generator().manual_seed(self.seed + self.epoch)
   271	        object_uids = sorted(self.by_object)
   272	        order = torch.randperm(len(object_uids), generator=generator).tolist()
   273	        selected = []
   274	        for object_index in order:
   275	            candidates = self.by_object[object_uids[object_index]]
   276	            choice = int(
   277	                torch.randint(len(candidates), (1,), generator=generator).item()
   278	            )
   279	            selected.append(candidates[choice])
   280	        if len(selected) < self.total_size:
   281	            selected.extend(selected[: self.total_size - len(selected)])
   282	        selected = selected[self.rank : self.total_size : self.num_replicas]
   283	        return iter(selected)

   707	        )

   710	def load_ema_state(
   711	    model: nn.Module, state: dict[str, torch.Tensor]
   712	) -> dict[str, torch.Tensor]:
   713	    named = trainable_named_parameters(model)
   714	    if set(state) != set(named):
   715	        raise ValueError("checkpoint EMA/trainable names differ")
   716	    return {
   717	        name: value.to(device=named[name].device, dtype=torch.float32).clone()
   718	        for name, value in state.items()
   719	    }

   722	def ema_state_finite(state: dict[str, torch.Tensor]) -> bool:
   723	    return bool(state) and all(
   724	        bool(torch.isfinite(value).all().item()) for value in state.values()
   725	    )

   728	def gradient_norms(model: nn.Module) -> dict[str, float]:
   729	    values = {"lora": 0.0, "condition": 0.0, "view_fusion": 0.0}
   730	    for name, parameter in model.named_parameters():
   731	        if parameter.grad is not None:
   732	            key = (
   733	                "lora"
   734	                if "lora_" in name
   735	                else "view_fusion"
   736	                if name.startswith("view_fusion.")
   737	                else "condition"
   738	            )
   739	            values[key] += float(parameter.grad.detach().float().square().sum().item())
   740	    return {key: value**0.5 for key, value in values.items()}

   743	def save_checkpoint(
   744	    path: Path,
   745	    *,
   746	    model: nn.Module,
   747	    optimizer: torch.optim.Optimizer,
   748	    scaler: Any,
   749	    step: int,
   750	    micro_step: int,
   751	    args: argparse.Namespace,
   752	    model_summary: dict[str, Any],
   753	    data_identity: dict[str, Any],
   754	    history: list[dict[str, Any]],
   755	    ema_state: dict[str, torch.Tensor],
   756	    ema_last_decay: float,
   757	) -> None:
   758	    trainable = [value for value in model.parameters() if value.requires_grad]
   759	    if not parameters_finite(trainable) or not optimizer_state_finite(optimizer):
   760	        raise RuntimeError(f"refusing to save non-finite Native-SLAT state: {path}")
   761	    if not finite_tree(scaler.state_dict()) or not ema_state_finite(ema_state):
   762	        raise RuntimeError(f"refusing to save non-finite scaler/EMA: {path}")
   763	    path.parent.mkdir(parents=True, exist_ok=True)
   764	    state_builder = (
   765	        v2_trainable_state_dict
   766	        if args.architecture == "v2"
   767	        else trainable_state_dict
   768	    )
   769	    payload = {
   770	        "format": architecture_checkpoint_format(args),
   771	        "step": int(step),
   772	        "micro_step": int(micro_step),
   773	        "model_trainable_state": state_builder(model),
   774	        "ema_trainable_state": {
   775	            name: value.detach().cpu().clone() for name, value in ema_state.items()
   776	        },
   777	        "ema": {
   778	            "target_decay": float(args.ema_decay),
   779	            "updates": int(step),
   780	            "last_decay": float(ema_last_decay),
   781	        },
   782	        "optimizer": optimizer.state_dict(),
   783	        "scaler": scaler.state_dict(),
   784	        "args": checkpoint_args(args),
   785	        "model_summary": model_summary,
   786	        "data_identity": data_identity,
   787	        "history": history,
   788	        "rng": {
   789	            "python": random.getstate(),
   790	            "numpy": np.random.get_state(),
   791	            "torch": torch.get_rng_state(),
   792	            "cuda": torch.cuda.get_rng_state(),
   793	        },
   794	    }
   795	    temporary = path.with_name(f".{path.name}.tmp-{os.getpid()}")
   796	    try:
   797	        torch.save(payload, temporary)
   798	        os.replace(temporary, path)
   799	    finally:
   800	        if temporary.exists():
   801	            temporary.unlink()

     7	class SparseConv3d(nn.Module):
     8	    def __init__(self, in_channels, out_channels, kernel_size, stride=1, dilation=1, padding=None, bias=True, indice_key=None):
     9	        super(SparseConv3d, self).__init__()
    10	        if 'spconv' not in globals():
    11	            import spconv.pytorch as spconv
    12	        algo = None
    13	        if SPCONV_ALGO == 'native':
    14	            algo = spconv.ConvAlgo.Native
    15	        elif SPCONV_ALGO == 'implicit_gemm':
    16	            algo = spconv.ConvAlgo.MaskImplicitGemm
    17	        if stride == 1 and (padding is None):
    18	            self.conv = spconv.SubMConv3d(in_channels, out_channels, kernel_size, dilation=dilation, bias=bias, indice_key=indice_key, algo=algo)
    19	        else:
    20	            self.conv = spconv.SparseConv3d(in_channels, out_channels, kernel_size, stride=stride, dilation=dilation, padding=padding, bias=bias, indice_key=indice_key, algo=algo)
    21	        self.stride = tuple(stride) if isinstance(stride, (list, tuple)) else (stride, stride, stride)
    22	        self.padding = padding
    23	
    24	    def forward(self, x: SparseTensor) -> SparseTensor:
    25	        spatial_changed = any(s != 1 for s in self.stride) or (self.padding is not None)
    26	        new_data = self.conv(x.data)
    27	        new_shape = [x.shape[0], self.conv.out_channels]
    28	        new_layout = None if spatial_changed else x.layout
    29	
    30	        if spatial_changed and (x.shape[0] != 1):
    31	            # spconv was non-1 stride will break the contiguous of the output tensor, sort by the coords
    32	            fwd = new_data.indices[:, 0].argsort()
    33	            bwd = torch.zeros_like(fwd).scatter_(0, fwd, torch.arange(fwd.shape[0], device=fwd.device))
    34	            sorted_feats = new_data.features[fwd]
    35	            sorted_coords = new_data.indices[fwd]
    36	            unsorted_data = new_data
    37	            new_data = spconv.SparseConvTensor(sorted_feats, sorted_coords, unsorted_data.spatial_shape, unsorted_data.batch_size)  # type: ignore
    38	
    39	        out = SparseTensor(
    40	            new_data, shape=torch.Size(new_shape), layout=new_layout,
    41	            scale=tuple([s * stride for s, stride in zip(x._scale, self.stride)]),
    42	            spatial_cache=x._spatial_cache,
    43	        )
    44	
    45	        if spatial_changed and (x.shape[0] != 1):
    46	            out.register_spatial_cache(f'conv_{self.stride}_unsorted_data', unsorted_data)
    47	            out.register_spatial_cache(f'conv_{self.stride}_sort_bwd', bwd)
    48	 
    49	        return out

    24	    """
    25	    Naive implementation of scaled dot product attention.
    26	    """
    27	    q = q.permute(0, 2, 1, 3)   # [N, H, L, C]
    28	    k = k.permute(0, 2, 1, 3)   # [N, H, L, C]
    29	    v = v.permute(0, 2, 1, 3)   # [N, H, L, C]

    32	    attn_weight = torch.softmax(attn_weight, dim=-1)
    33	    out = attn_weight @ v
    34	    out = out.permute(0, 2, 1, 3)   # [N, L, H, C]
