cd /home/zjr/Tracker
set -o pipefail

# 目标：
# 1. 先断点补齐原冻结5K，不改变其selection；
# 2. 将5K用硬链接映射到新的30K根（不复制约20GiB数据，不破坏旧路径）；
# 3. 从shard-0007起冻结并下载额外25K，且把旧5K UID作为硬排除集合；
# 4. 审计5K+25K严格不重叠、文件大小/目录集合精确、抽样tar/NPZ schema正确；
# 5. 最终新数据根可直接传给ReconViaGen的--data_root。

PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
REVISION=c9175c52d4a45feb9536bec2f1168c50ecdc7765
DOWNLOAD_WORKERS=${DOWNLOAD_WORKERS:-8}

BASE5=/data/zjr/ProObjaverse-300K-ReconViaGen-5K
BASE_SELECTION=${BASE5}/selection.json
DATA30=/data/zjr/ProObjaverse-300K-ReconViaGen-30K
STATE=/data/zjr/ProObjaverse-300K-ReconViaGen-30K-state
STATE25=${STATE}/additional25k
AUDIT=${STATE}/combined_audit

# Hugging Face/Xet缓存也放到/data；下载大文件时不占home。
export HF_HOME=${HF_HOME:-/data/zjr/huggingface}
export HF_HUB_CACHE=${HF_HUB_CACHE:-${HF_HOME}/hub}
export HF_XET_CACHE=${HF_XET_CACHE:-${HF_HOME}/xet}
export HF_HUB_DOWNLOAD_TIMEOUT=${HF_HUB_DOWNLOAD_TIMEOUT:-180}
export HF_HUB_ETAG_TIMEOUT=${HF_HUB_ETAG_TIMEOUT:-30}
export HF_HUB_DISABLE_PROGRESS_BARS=1

mkdir -p "${BASE5}/logs" "${DATA30}" "${STATE25}/logs" "${AUDIT}" \
  "${HF_HUB_CACHE}" "${HF_XET_CACHE}"

# D0：续传并严格补齐原5K。若已经完成，5000对只做快速size验证。
"${PY}" -u ReconViaGen/download_proobjaverse_subset.py \
  --output-root "${BASE5}" \
  --revision "${REVISION}" \
  --count 5000 \
  --seed 20260813 \
  --shard-start 1 \
  --shard-stop 295 \
  --candidate-margin 500 \
  --max-workers "${DOWNLOAD_WORKERS}" \
  --retries 8 \
  --min-free-extra-gib 50 \
  2>&1 | tee -a "${BASE5}/logs/download_resume_for_30k.log"
D0_RC=${PIPESTATUS[0]}
echo "D0_RC=${D0_RC}"

# D1：验证base报告和selection身份，再将5K硬链接进新的30K数据根。
if [ "${D0_RC}" -ne 0 ]; then
  echo "D1 blocked: base 5K is incomplete"
  D1_RC=96
else
  "${PY}" -c \
    'import json,sys
s=json.load(open(sys.argv[1])); r=json.load(open(sys.argv[2]))
assert s["format"]=="reconviagen.proobjaverse_paired_subset.v1"
assert s["selected_pair_count"]==5000
assert r["complete"] is True and r["selected_pair_count"]==5000
assert r["completed_pair_count"]==5000 and r["failed_pair_count"]==0
assert r["selection_sha256"]==s["selection_sha256"]
print("base5 verified:",r["completed_pair_count"])' \
    "${BASE_SELECTION}" "${BASE5}/download_report.json" \
    || D1_RC=$?
  D1_RC=${D1_RC:-0}
fi

if [ "${D1_RC}" -eq 0 ]; then
  "${PY}" -u ReconViaGen/prepare_proobjaverse_30k.py link-base \
    --source-root "${BASE5}" \
    --target-root "${DATA30}" \
    --selection "${BASE_SELECTION}" \
    --expected-count 5000 \
    --report "${STATE}/base5_hardlink_report.json" \
    2>&1 | tee "${STATE}/base5_hardlink.log"
  D1_RC=${PIPESTATUS[0]}
fi
echo "D1_RC=${D1_RC}"

# D2：只查询官方仓库并冻结额外25K，不下载大文件。
# shard-0007起步避免重新扫描旧5K所在的0001--0006；exclude-selection仍提供独立防重门。
if [ "${D1_RC}" -ne 0 ]; then
  echo "D2 blocked: base hard-link materialization failed"
  D2_RC=96
else
  "${PY}" -u ReconViaGen/download_proobjaverse_subset.py \
    --output-root "${DATA30}" \
    --state-root "${STATE25}" \
    --exclude-selection "${BASE_SELECTION}" \
    --revision "${REVISION}" \
    --count 25000 \
    --seed 20260813 \
    --shard-start 7 \
    --shard-stop 295 \
    --candidate-margin 2500 \
    --plan-only \
    2>&1 | tee "${STATE25}/logs/plan.log"
  D2_RC=${PIPESTATUS[0]}
fi
echo "D2_RC=${D2_RC}"

# 冻结后立即检查：25K清单必须显式绑定并排除旧5K。
if [ "${D2_RC}" -eq 0 ]; then
  "${PY}" -c \
    'import json,sys
b=json.load(open(sys.argv[1])); e=json.load(open(sys.argv[2]))
bu={x["uid"] for x in b["selected"]}; eu={x["uid"] for x in e["selected"]}
assert len(bu)==5000 and len(eu)==25000 and not (bu & eu)
assert e["excluded_uid_count"]==5000
assert any(x["selection_sha256"]==b["selection_sha256"] for x in e["excluded_selection_bindings"])
print("extension frozen: pairs=25000 overlap=0")
print("expected additional GiB:",round(e["expected_download_bytes"]/1024**3,2))
print("scanned shards:",e["scanned_shards"][0]["shard"],"to",e["scanned_shards"][-1]["shard"])' \
    "${BASE_SELECTION}" "${STATE25}/selection.json" \
    || D2_RC=$?
fi
echo "D2_AUDITED_RC=${D2_RC}"

# D3：下载额外25K。可任意中断；重跑整个txt会先验证/复用已经完整的对象。
if [ "${D2_RC}" -ne 0 ]; then
  echo "D3 blocked: additional 25K planning failed"
  D3_RC=96
else
  "${PY}" -u ReconViaGen/download_proobjaverse_subset.py \
    --output-root "${DATA30}" \
    --state-root "${STATE25}" \
    --exclude-selection "${BASE_SELECTION}" \
    --revision "${REVISION}" \
    --count 25000 \
    --seed 20260813 \
    --shard-start 7 \
    --shard-stop 295 \
    --candidate-margin 2500 \
    --max-workers "${DOWNLOAD_WORKERS}" \
    --retries 8 \
    --min-free-extra-gib 50 \
    2>&1 | tee -a "${STATE25}/logs/download.log"
  D3_RC=${PIPESTATUS[0]}
fi
echo "D3_RC=${D3_RC}"

# D4：冻结最终30K合并清单；检查两个selection不重叠、数据根无缺失/额外对象，
# 并稳定抽64对打开NPZ、tar、camera JSON及一张WebP做schema审计。
if [ "${D3_RC}" -ne 0 ]; then
  echo "D4 blocked: additional 25K download incomplete"
  D4_RC=96
else
  "${PY}" -u ReconViaGen/prepare_proobjaverse_30k.py audit \
    --data-root "${DATA30}" \
    --selection "${BASE_SELECTION}" \
    --selection "${STATE25}/selection.json" \
    --expected-counts 5000,25000 \
    --expected-total 30000 \
    --schema-samples 64 \
    --seed 20260813 \
    --output-dir "${AUDIT}" \
    2>&1 | tee "${AUDIT}/audit.log"
  D4_RC=${PIPESTATUS[0]}
fi
echo "D4_RC=${D4_RC}"

# D5：用ReconViaGen实际TarDataset索引一次，确认训练代码看见的正好是30000对象。
if [ "${D4_RC}" -ne 0 ]; then
  echo "D5 blocked: combined audit failed"
  D5_RC=96
else
  DATA30="${DATA30}" PYTHONPATH=/home/zjr/Tracker/ReconViaGen \
  "${PY}" -c \
    'import os
from dataset import TarDataset
d=TarDataset(os.environ["DATA30"],num_views=6,random_sample=-1,image_size=518)
assert len(d)==30000,len(d)
print("ReconViaGen TarDataset exact length:",len(d))' \
    2>&1 | tee "${AUDIT}/dataset_index_smoke.log"
  D5_RC=${PIPESTATUS[0]}
fi
echo "D5_RC=${D5_RC}"

if [ "${D5_RC}" -eq 0 ]; then
  echo "30K dataset ready: ${DATA30}"
  echo "combined selection: ${AUDIT}/combined_selection_30k.json"
  echo "audit report: ${AUDIT}/audit_report.json"
  du -sh "${DATA30}"
fi
