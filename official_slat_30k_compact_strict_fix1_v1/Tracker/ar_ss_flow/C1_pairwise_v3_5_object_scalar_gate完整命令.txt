C1 pairwise-v3.5 / C1.6：Object-level Scalar Gate完整命令
================================================================================

目标：
1. 不重新训练pairwise-v3 checkpoint。
2. 不再使用voxel/local/region gate作为正式部署候选。
3. 对每个多视图样本，把4096个raw visual-only pairwise confidence汇总为一个object scalar。
4. 只在indices 0-15选择以下统计量之一：
   - mean
   - median
   - trimmed_mean_10
   - top20_mean
5. 使用训练对象标定绝对gate：
   g_object = clip((object_score - tau_low) / (tau_high - tau_low), 0, 1)
6. indices 16-63完全冻结验证，不重新选择统计量或阈值。
7. correct、wrong pose、visual shuffle分别计算object scalar/gate。
8. 只有seed42通过后才运行seed43/44；seed43/44冻结使用seed42选择的统计量，但各自在自己的train16上标定阈值。
9. 只有三seed全部通过，才允许设计正式C2：
   cond_final = cond_base + g_object * delta_cond
   stock condition本身不加gate、不修改。

ensure_volume_audit () {
  SEED="$1"
  NAME="$2"
  INDICES="$3"
  RUN=ar_ss_flow/outputs/corr_pairwise_v3_s200_seed${SEED}
  OUT=${RUN}/deployment_local_audit_${NAME}_v2

  if [ "${FORCE_AUDIT}" != "1" ] && \
     [ -s "${OUT}/report.json" ] && \
     [ -s "${OUT}/volume_samples.npz" ]; then
    echo "Reusing ${OUT}"
    return 0
  fi

  rm -rf "${OUT}"
  rm -f "${OUT}.log" "${OUT}.exit_code"

  (
    CUDA_VISIBLE_DEVICES=${GPU} \
    /home/zjr/anaconda3/envs/reconviagen/bin/python -u \
      -m ar_ss_flow.audit_pairwise_local_deployment \
      --cache_manifest /data/ar_ss_flow_pose_lifting_overfit64_v1_20260714/manifest.json \
      --checkpoint "${RUN}/checkpoints/last.pt" \
      --output_dir "${OUT}" \
      --indices "${INDICES}" \
      --device cuda \
      --seed "${SEED}" \
      --source_view_count 4 \
      --negative_modes pose_cyclic1,pose_cyclic2,pose_reverse \
      --neighborhood_radius 1 \
      --min_source_views 2 \
      --min_common_voxels 64
  ) 2>&1 | tee "${OUT}.log"

  STATUS=${PIPESTATUS[0]}
  echo "${STATUS}" | tee "${OUT}.exit_code"
  if [ "${STATUS}" -ne 0 ]; then
    echo "Audit failed: ${OUT}"
    return "${STATUS}"
  fi
  test -s "${OUT}/report.json" || return 10
  test -s "${OUT}/volume_samples.npz" || return 11
}

if [ "${SEED42_STATUS}" = "0" ]; then
  for SEED in 43 44; do
    ensure_volume_audit "${SEED}" train16 0-15
    TRAIN_STATUS=$?
    if [ "${TRAIN_STATUS}" = "0" ]; then
      ensure_volume_audit "${SEED}" fresh48 16-63
      FRESH_STATUS=$?
    else
      FRESH_STATUS=99
    fi
    echo "seed=${SEED} train audit status=${TRAIN_STATUS}"
