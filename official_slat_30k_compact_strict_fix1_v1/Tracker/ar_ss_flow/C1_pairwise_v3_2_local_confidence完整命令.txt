C1 pairwise-v3.2：局部区域 confidence 校准完整命令
============================================================

目标：
1. 不重新训练 pairwise-v3 checkpoint。
2. 对 visual-only voxel confidence 比较三种固定后处理：
   - raw
   - 3x3x3 common-support local mean
   - 3x3x3 common-support local top-k mean
3. 聚合方法和 tau_low/tau_high 只在 indices 0-15 上选择。
4. indices 16-63 只做冻结验证。
5. 局部方法必须 fresh AUC 平均比 raw 至少提高 0.03，且任何 hard negative 不得明显下降。

约定：
- Tracker：/home/zjr/Tracker
- Python：/home/zjr/anaconda3/envs/reconviagen/bin/python
- Cache：/data/ar_ss_flow_pose_lifting_overfit64_v1_20260714/manifest.json
- 工具包：/home/zjr/Downloads/c1_pairwise_v3_2_local_tools.zip
- 默认 GPU=0，可先 export GPU=1 改卡。

注意：
- 本命令不会在科学门失败时执行 exit 2，因此不会关闭当前交互式终端。
- exit_code=2 表示程序正常完成但科学门失败。
- exit_code 为其他非零值才表示文件、参数或程序错误。

cd /home/zjr/Tracker || exit 1
export GPU=${GPU:-0}

SEED=42
RUN=ar_ss_flow/outputs/corr_pairwise_v3_s200_seed${SEED}

run_local_audit () {
  NAME="$1"
  INDICES="$2"
  OUT="${RUN}/deployment_local_audit_${NAME}_v2"

  rm -rf "${OUT}"
  rm -f "${OUT}.log" "${OUT}.exit_code"

  echo "===================================================================================================="
  echo "seed=${SEED} local audit ${NAME}, indices=${INDICES}"

  (
    CUDA_VISIBLE_DEVICES=${GPU} \
    /home/zjr/anaconda3/envs/reconviagen/bin/python -u \
      -m ar_ss_flow.audit_pairwise_local_deployment \
      --cache_manifest /data/ar_ss_flow_pose_lifting_overfit64_v1_20260714/manifest.json \
      --checkpoint "${RUN}/checkpoints/last.pt" \
      --output_dir "${OUT}" \
      --indices "${INDICES}" \
      --device cuda \
      --seed "${SEED}" \
      --source_view_count 4 \
      --negative_modes pose_cyclic1,pose_cyclic2,pose_reverse \
      --neighborhood_radius 1 \
      --min_source_views 2 \
      --min_common_voxels 64
  ) 2>&1 | tee "${OUT}.log"

  STATUS=${PIPESTATUS[0]}
  echo "${STATUS}" | tee "${OUT}.exit_code"

  if [ "${STATUS}" -ne 0 ]; then
    echo "Audit failed: ${OUT}"
    echo "Inspect: tail -n 150 ${OUT}.log"
    return "${STATUS}"
  fi

  test -s "${OUT}/report.json" || { echo "Missing report.json"; return 10; }
  test -s "${OUT}/volume_samples.npz" || { echo "Missing volume_samples.npz"; return 11; }
  ls -lh "${OUT}/report.json" "${OUT}/volume_samples.npz"
}

run_local_audit train16 0-15
TRAIN_STATUS=$?

if [ "${TRAIN_STATUS}" -eq 0 ]; then
  run_local_audit fresh48 16-63
  FRESH_STATUS=$?
else
  FRESH_STATUS=99
  echo "Skipping fresh audit because train audit failed."
fi

echo "train audit status=${TRAIN_STATUS}"
echo "fresh audit status=${FRESH_STATUS}"
