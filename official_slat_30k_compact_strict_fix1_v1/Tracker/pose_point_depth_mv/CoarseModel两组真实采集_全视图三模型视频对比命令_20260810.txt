#!/usr/bin/env bash
set -euo pipefail

# 用途
# ====
#
# 两组指定真实采集：
#   A. CoarseModel/datasets/reconviagen_20260512_083943：全部 18 张图像。
#      其中 13 张保留原 COLMAP 位姿；5 张用 AR pose 与最近 5 个已注册相机的
#      局部 proper-Sim(3)+相机轴基拟合补齐。适配器会先做留一误差审计。
#   B. CoarseModel/datasets/reconviagen_20260513_062152：全部 16 张图像，全部
#      保留原 COLMAP 位姿。
#
# 三路输出：
#   01 当前 mixed No-VGGT（A=18 views，B=16 views）的 normal turntable MP4；
#   02 原目录 reconviagen_output/reconstructed_object.mp4 的逐字节副本；
#   03 Pixal3D official-final（单视图模型）的 normal turntable MP4；
#   04 另附三路显示自动取景并排视频和抽帧图，仅供人工看几何。
#
# 注意：当前模型训练时最多采样 16 views；A 的 18 views 是有意进行的轻微 OOD
# 开发诊断。Pixal3D 本身仍是单视图方法，只会自动选择掩码面积最大的输入帧。
# 04/05 会用每段视频唯一且固定的前景裁切框，把主体最大边统一到约 72% 画幅；
# 不会逐帧缩放，因此不会产生呼吸/抖动。ReconViaGen 在并排图中只取原视频
# 右半区的法线 Mesh，02 的原始 MP4 仍逐字节保留。
# 该操作只统一显示占屏比例，不会改变 Mesh，也不代表世界尺度、位姿或相机对齐。
#
# 每一节都能逐段复制；也可以一次执行整个文件。默认 GPU=5，运行前按实际空闲卡改。


# V0. 静态检查与最小测试（CPU，通常 < 2 秒）
# ===========================================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python

${PY} -m py_compile \
  pose_point_depth_mv/dataset_tools/prepare_coarsemodel_real_raw_cache.py \
  pose_point_depth_mv/render_coarsemodel_current_recon_pixal_videos.py

MPLCONFIGDIR=/tmp/matplotlib ${PY} -m unittest -v \
  pose_point_depth_mv.dataset_tools.test_prepare_coarsemodel_real_raw_cache \
  pose_point_depth_mv.test_render_coarsemodel_current_recon_pixal_videos

echo "V0 complete"
echo "current terminal remains open"


# V1. 两组全图像 raw cache + 位姿补齐审计（CPU，约 5 秒）
# =======================================================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
ROOT=/data/zjr/coarsemodel_real_allviews_threeway_20260810_v1
RAW=${ROOT}/raw_cache_allviews_v1

${PY} -u -m pose_point_depth_mv.dataset_tools.prepare_coarsemodel_real_raw_cache \
  --dataset /home/zjr/Tracker/CoarseModel/datasets/reconviagen_20260512_083943 \
  --dataset /home/zjr/Tracker/CoarseModel/datasets/reconviagen_20260513_062152 \
  --output_dir "${RAW}" \
  --min_registered_pairs 8 \
  --include_all_images \
  --resume

${PY} -c '
import json,sys
r=json.load(open(sys.argv[1])); rows={x["object_id"]:x for x in r["objects"]}
expected={"reconviagen_20260512_083943":(18,13,5),"reconviagen_20260513_062152":(16,16,0)}
assert r["passed"] is True and set(rows)==set(expected)
for key,value in expected.items():
    row=rows[key]
    actual=(row["input_view_count"],row["registered_pair_count"],row["phone_pose_augmented_count"])
    assert actual==value,(key,actual,value)
    assert row["pose_completion_audit"]["passed"] is True
    assert row["legacy_reconviagen_video"]
print(json.dumps({k:{"input/registered/completed":expected[k],
 "pose_completion_loo":rows[k]["pose_completion_audit"]} for k in expected},indent=2,ensure_ascii=False))
' "${RAW}/raw_cache_report.json"

echo "V1 complete: ${RAW}/raw_cache_report.json"
echo "current terminal remains open"


# V2. 分别构建 18-view 与 16-view runtime-O 输入（CPU，约 10 秒）
# ============================================================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
ROOT=/data/zjr/coarsemodel_real_allviews_threeway_20260810_v1
RAW=${ROOT}/raw_cache_allviews_v1
RUNTIME_A=${ROOT}/runtime_0512_all18_v1
RUNTIME_B=${ROOT}/runtime_0513_all16_v1

${PY} -u -m pose_point_depth_mv.dataset_tools.prepare_omni_real_runtime_inputs \
  --raw_cache_report "${RAW}/raw_cache_report.json" \
  --output_dir "${RUNTIME_A}" \
  --object coarsemodel_real:reconviagen_20260512_083943 \
  --selected_view_count 18 \
  --view_selection_policy lexical_even_valid_mask_fallback \
  --min_object_points 64 \
  --min_mask_observations 2 \
  --min_mask_support_ratio 0.50 \
  --min_completed_objects 1 \
  --resume

${PY} -u -m pose_point_depth_mv.dataset_tools.prepare_omni_real_runtime_inputs \
  --raw_cache_report "${RAW}/raw_cache_report.json" \
  --output_dir "${RUNTIME_B}" \
  --object coarsemodel_real:reconviagen_20260513_062152 \
  --selected_view_count 16 \
  --view_selection_policy lexical_even_valid_mask_fallback \
  --min_object_points 64 \
  --min_mask_observations 2 \
  --min_mask_support_ratio 0.50 \
  --min_completed_objects 1 \
  --resume

${PY} -c '
import json,sys
for path,views in zip(sys.argv[1:],[18,16]):
    r=json.load(open(path)); assert r["passed"] is True and r["completed_object_count"]==1
    row=r["objects"][0]; assert row["selected_view_count"]==views,(path,row["selected_view_count"],views)
    print(row["object_key"],"views=",views,
          "mask_supported_points=",row["runtime_frame_stats"]["support"]["mask_supported_point_count"])
' "${RUNTIME_A}/runtime_input_manifest.json" "${RUNTIME_B}/runtime_input_manifest.json"

echo "V2 complete"
echo "current terminal remains open"


# V3. DINO-only 编码（单 GPU，两组通常合计约 2--5 分钟）
# ======================================================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
GPU=${GPU:-5}
ROOT=/data/zjr/coarsemodel_real_allviews_threeway_20260810_v1
RUNTIME_A=${ROOT}/runtime_0512_all18_v1
RUNTIME_B=${ROOT}/runtime_0513_all16_v1
MODEL_A=${ROOT}/dino_0512_all18_v1
MODEL_B=${ROOT}/dino_0513_all16_v1

for SPEC in "${RUNTIME_A}:${MODEL_A}" "${RUNTIME_B}:${MODEL_B}"; do
  RUNTIME=${SPEC%%:*}
  MODEL=${SPEC#*:}
  CUDA_VISIBLE_DEVICES="${GPU}" \
  HF_HUB_OFFLINE=1 \
  TRANSFORMERS_OFFLINE=1 \
  ATTN_BACKEND=flash_attn \
  SPCONV_ALGO=native \
  MPLCONFIGDIR=/tmp/matplotlib \
  NUMBA_CACHE_DIR=/tmp/numba_cache \
  TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
  ${PY} -u -m pose_point_depth_mv.dataset_tools.prepare_omni_real_dino_only_model_inputs \
    --runtime_input_manifest "${RUNTIME}/runtime_input_manifest.json" \
    --output_dir "${MODEL}" \
    --device cuda \
    --resume
done

echo "V3 complete"
echo "current terminal remains open"


# V4. 当前 mixed No-VGGT 模型推理（单 GPU，两组通常数分钟）
# ========================================================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
GPU=${GPU:-5}
ROOT=/data/zjr/coarsemodel_real_allviews_threeway_20260810_v1
RUN=/data/zjr/native_no_vggt_mixed_real376_synth868_20260808_v1
SS_CKPT=${RUN}/ss_mixed_step2000_seed42_1gpu_v1/checkpoints/step_002000.pt
SLAT_CKPT=${RUN}/slat_mixed_step2000_seed42_2gpu_v1/checkpoints/step_002000.pt
SS_CONTRACT=${RUN}/contracts/ss_real_full_ema_v1.json
SLAT_CONTRACT=${RUN}/contracts/slat_real_full_ema_v1.json
STOCK_FREEZE=/data/zjr/native_slat_genrecon_v2_mixed1k_20260802_v1/stock_slat_freeze_v2.json
MODEL_A=${ROOT}/dino_0512_all18_v1
MODEL_B=${ROOT}/dino_0513_all16_v1
CURRENT_A=${ROOT}/current_0512_all18_seed42_v1
CURRENT_B=${ROOT}/current_0513_all16_seed42_v1

for SPEC in "${MODEL_A}:${CURRENT_A}" "${MODEL_B}:${CURRENT_B}"; do
  MODEL=${SPEC%%:*}
  CURRENT=${SPEC#*:}
  if [ -s "${CURRENT}/inference_manifest.json" ]; then
    echo "reuse current inference: ${CURRENT}/inference_manifest.json"
    continue
  fi
  CUDA_VISIBLE_DEVICES="${GPU}" \
  HF_HUB_OFFLINE=1 \
  TRANSFORMERS_OFFLINE=1 \
  ATTN_BACKEND=flash_attn \
  SPCONV_ALGO=native \
  MPLCONFIGDIR=/tmp/matplotlib \
  NUMBA_CACHE_DIR=/tmp/numba_cache \
  TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
  PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
  ${PY} -u -m pose_point_depth_mv.infer_omni_real_native_no_vggt_mixed \
    --model_input_manifest "${MODEL}/model_input_manifest.json" \
    --native_ss_checkpoint "${SS_CKPT}" \
    --native_slat_checkpoint "${SLAT_CKPT}" \
    --ss_migration_contract "${SS_CONTRACT}" \
    --slat_migration_contract "${SLAT_CONTRACT}" \
    --stock_slat_freeze "${STOCK_FREEZE}" \
    --output_dir "${CURRENT}" \
    --seeds 42 \
    --weights ema \
    --amp_dtype bf16 \
    --device cuda
done

echo "V4 complete"
echo "current terminal remains open"


# V5. Pixal3D official-final 单视图推理（单 GPU，两组通常约 4--8 分钟）
# ==================================================================

cd /home/zjr/Tracker

ROOT=/data/zjr/coarsemodel_real_allviews_threeway_20260810_v1
RUNTIME_A=${ROOT}/runtime_0512_all18_v1
RUNTIME_B=${ROOT}/runtime_0513_all16_v1
PIXAL_A=${ROOT}/pixal_0512_seed42_v1

PIXAL_REVISION=0b31f9160aa400719af409098bff7936a932f726
PIXAL_SNAPSHOT=/home/zjr/.cache/huggingface/hub/models--TencentARC--Pixal3D/snapshots/${PIXAL_REVISION}
NAF_COMMIT=37f2dfc180f2de53d98bd601109c0da0dd6b0f43
NAF_ROOT=/data/zjr/models/valeoai_NAF_${NAF_COMMIT}

for SPEC in "${RUNTIME_A}:${PIXAL_A}" "${RUNTIME_B}:${PIXAL_B}"; do
  RUNTIME=${SPEC%%:*}
  PIXAL=${SPEC#*:}
  if [ -s "${PIXAL}/inference_manifest.json" ]; then
    echo "reuse Pixal3D inference: ${PIXAL}/inference_manifest.json"
    continue
  fi
  CUDA_VISIBLE_DEVICES="${GPU}" \
  HF_HUB_OFFLINE=1 \
  TRANSFORMERS_OFFLINE=1 \
  ATTN_BACKEND=sdpa \
  SPARSE_ATTN_BACKEND=sdpa \
  SPCONV_ALGO=native \
  MPLCONFIGDIR=/tmp/matplotlib \
  NUMBA_CACHE_DIR=/tmp/numba_cache \
  TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
  PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
  ${PIXAL_PY} -u -m pose_point_depth_mv.infer_omni_real_pixal3d \
    --runtime_input_manifest "${RUNTIME}/runtime_input_manifest.json" \
    --output_dir "${PIXAL}" \
    --model_path "${PIXAL_SNAPSHOT}" \
    --naf_repo "${NAF_ROOT}/repo" \
    --naf_checkpoint "${NAF_ROOT}/naf_release.pth" \
    --naf_source_manifest "${NAF_ROOT}/source_manifest.sha256" \
    --device cuda \
    --low_vram \
    --seeds 42 \
    --resolution 1024 \
    --sampling_steps 12 \
    --isolate_objects \
    --isolate_batch_size 1
done

echo "V5 complete"
echo "current terminal remains open"

# V1--V5 已完成时只需重跑本节。使用新的 v2 目录，保留原 v1 审查结果。

# V7. 最终只读核验
# ===============

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
VIS=/home/zjr/Tracker/pose_point_depth_mv/outputs/可视化Mesh/CoarseModel两组真实采集_全视图三模型视频对比_显示归一化_20260810_v2

${PY} -c '
import json,sys
from pathlib import Path
root=Path(sys.argv[1]); r=json.load(open(root/"report.json"))
assert r["passed"] is True and r["object_count"]==2
expected={"reconviagen_20260512_083943":18,"reconviagen_20260513_062152":16}
for row in r["cases"]:
    name=row["object_key"].split(":",1)[1]
    assert row["current_input_view_count"]==expected[name]
    assert row["mesh_files_copied_to_review_directory"] is False
    assert row["render_config"]["comparison_autoframe"] is True
    assert row["render_config"]["comparison_target_fill"]==0.72
    assert set(row["display"]["comparison_autoframe"])=={"current","reconviagen","pixal3d"}
    for key in ("current_no_vggt","reconviagen_original","pixal3d_official","threeway_display"):
        assert Path(row["videos"][key]["path"]).is_file()
assert not list(root.rglob("*.obj")) and not list(root.rglob("*.glb"))
print(json.dumps({"passed":True,"objects":expected,"output_dir":str(root),
 "scope_guard":r["scope_guard"]},indent=2,ensure_ascii=False))
' "${VIS}"

echo "current terminal remains open"
