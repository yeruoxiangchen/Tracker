#!/usr/bin/env bash
#
# 单物体主任务数据 A/B/C/R 增量重构
# ==================================
#
# 目标：
#   1. 立即重审当前 897-object 训练集，不等待 10k 全部渲染；
#   2. 自动并入“已有 _WORKER_COMPLETE.json”的全部新 Omni/Objaverse 渲染证据；
#   3. 只从 source plan 确定性抽取少量 fresh Objaverse 候选，不扫描全部源 Mesh；
#   4. 后续有新 Omni 分片完成时，复用 Mesh 审计并生成新的不可变快照；
#   5. 不覆盖旧数据；所有输出均为新不可变目录。
#
# 分层不是类别标签：
#   A = 有成功渲染证据的高置信单物体候选；
#   B = Mesh 有效，但多部件/低纹理/尚无渲染证据等软歧义候选；
#   C = scene-like 候选，必须人工判断；节点多不等价于“不是单物体”；
#   R = Mesh 硬损坏，或所有已观察渲染均为 source_asset_rejected。
#
# 注意：
#   - 第一次输出是 draft，不可直接作为正式训练冻结；
#   - A/B/C 都允许人工覆写；C 不是自动删除名单；
#   - 本流程完全使用 CPU，不占 GPU；
#   - 为避免与当前渲染争抢 /data IO，源池审计固定使用低 IO/CPU 优先级，
#     默认仅 4 workers。
#
# 推荐运行方式：
#   cd /home/zjr/Tracker
#   bash pose_point_depth_mv/单物体训练数据A_B_C增量重构命令_20260730.txt
#
# 若只想先跑 897：
#   RUN_SOURCE_POOL=0 bash pose_point_depth_mv/单物体训练数据A_B_C增量重构命令_20260730.txt

set -euo pipefail

cd /home/zjr/Tracker

PY=${PY:-/home/zjr/anaconda3/envs/reconviagen/bin/python}
AUDIT_WORKERS=${AUDIT_WORKERS:-4}
RUN_897=${RUN_897:-1}
# 旧897 + accepted Omni 123 已达到1020 objects。大候选审计默认关闭；
# 只有后续出现明确数据缺口时才显式启用。
RUN_SOURCE_POOL=${RUN_SOURCE_POOL:-0}
FRESH_OBJAVERSE_CANDIDATES=${FRESH_OBJAVERSE_CANDIDATES:-0}
SOURCE_SELECTION_SEED=${SOURCE_SELECTION_SEED:-20260730}

TRAIN897_MANIFEST=/data/zjr/reconvggt_pointpose_v9_ssfixed_odsplit_20260712/manifests/train.json
OLD_VAL_MANIFEST=/data/zjr/reconvggt_pointpose_v9_ssfixed_odsplit_20260712/manifests/val.json
OLD_HOLDOUT_MANIFEST=/data/zjr/reconvggt_pointpose_v9_ssfixed_odsplit_20260712/manifests/holdout.json
SOURCE_PLAN=/data/zjr/dataset10k_20260726/source_freeze_obj11546_omni215_v2/build_plan.json

RESTRUCTURE_ROOT=/data/zjr/single_object_restructure_20260730
TRAIN897_PREVIOUS=${RESTRUCTURE_ROOT}/train897_draft_v1
TRAIN897_OUT=${RESTRUCTURE_ROOT}/train897_fullreview_draft_v2
TRAIN897_LOG=${TRAIN897_OUT}.log

# 当前已完成渲染证据。工具只读取带 _WORKER_COMPLETE.json 的分片；
# 尚在写入的分片自动忽略。
OBJ_PILOT=/data/zjr/dataset10k_20260726/pilot_objaverse256_cyclescuda_v2
OMNI_PILOT=/data/zjr/dataset10k_20260726/pilot_omni256_cyclescuda_v4
BOTH_SMOKE=/data/zjr/dataset10k_20260726/smoke_both16_cyclescuda_v4

# 如果后面启动正式全量渲染，可在运行前增加：
#   export FORMAL_RENDER_ROOT=/data/zjr/dataset10k_20260726/<正式render目录>
RENDER_ARGS=(
  --render_root "${OBJ_PILOT}"
  --render_root "${OMNI_PILOT}"
  --render_root "${BOTH_SMOKE}"
)
if [ -n "${FORMAL_RENDER_ROOT:-}" ]; then
  RENDER_ARGS+=(--render_root "${FORMAL_RENDER_ROOT}")
fi

mkdir -p "${RESTRUCTURE_ROOT}"

echo "D0: 静态检查与单元测试"
"${PY}" -m py_compile \
  pose_point_depth_mv/dataset_tools/restructure_single_object_dataset.py \
  pose_point_depth_mv/dataset_tools/test_restructure_single_object_dataset.py
"${PY}" -m unittest \
  pose_point_depth_mv.dataset_tools.test_restructure_single_object_dataset -v
echo "D0_RC=0"

if [ "${RUN_897}" = 1 ]; then
  echo "D1: 重构当前 897-object 训练集（前台，CPU-only）"
  "${PY}" -u -m \
    pose_point_depth_mv.dataset_tools.restructure_single_object_dataset \
      --manifest "${TRAIN897_MANIFEST}" \
      --reuse_mesh_audit "${TRAIN897_PREVIOUS}" \
      --output_dir "${TRAIN897_OUT}" \
      --workers "${AUDIT_WORKERS}" \
      --write_previews \
      --preview_frames 4 \
      --max_previews_per_tier 1000 \
    2>&1 | tee "${TRAIN897_LOG}"
  D1_RC=${PIPESTATUS[0]}
  echo "D1_RC=${D1_RC}"
  echo "897 report: ${TRAIN897_OUT}/report.json"
  echo "897 review: ${TRAIN897_OUT}/review/index.html"
  echo "897 A+B draft manifest:"
  echo "  ${TRAIN897_OUT}/manifests/000_train_ab.json"
fi

# 组合候选快照始终保留：
#   - 旧897显式 manifest；
#   - 所有完成 marker 中的 Omni/Objaverse；
#   - 人工 override 对象。
# source-plan-only 候选默认不再抽 fresh Objaverse 或 Omni。显式设置
# RUN_SOURCE_POOL=1 FRESH_OBJAVERSE_CANDIDATES=<数量> 才启动缺口候选审计。
# 新的 Omni 必须来自完成 marker，不会扫描全部 5,889 个 Omni 源 Mesh。
SOURCE_SNAPSHOT_TAG=${SOURCE_SNAPSHOT_TAG:-mixed1k_optional_gap_objfresh${FRESH_OBJAVERSE_CANDIDATES}_20260730_v4}
SOURCE_OUT=${RESTRUCTURE_ROOT}/${SOURCE_SNAPSHOT_TAG}
SOURCE_LOG=${SOURCE_OUT}.log
SOURCE_UNIT=tracker-single-object-gap-candidates-v4

if [ "${RUN_SOURCE_POOL}" = 1 ]; then
  echo "D2: 启动可选缺口候选审计（CPU-only，低 IO 优先级）"
  echo "旧897=全部；完成render marker=全部；fresh Objaverse=${FRESH_OBJAVERSE_CANDIDATES}"
  if [ -s "${SOURCE_OUT}/report.json" ]; then
    echo "reuse complete source-pool snapshot: ${SOURCE_OUT}/report.json"
  elif [ -e "${SOURCE_OUT}" ]; then
    echo "D2 blocked: immutable partial/different output exists: ${SOURCE_OUT}"
    exit 98
  elif systemctl --user is-active --quiet "${SOURCE_UNIT}.service"; then
    echo "already active: ${SOURCE_UNIT}.service"
  else
    mkdir -p "$(dirname "${SOURCE_LOG}")"
    systemctl --user reset-failed "${SOURCE_UNIT}.service" >/dev/null 2>&1 || true
    systemd-run --user \
      --unit="${SOURCE_UNIT}" \
      --collect \
      --property=WorkingDirectory=/home/zjr/Tracker \
      --property="StandardOutput=append:${SOURCE_LOG}" \
      --property="StandardError=append:${SOURCE_LOG}" \
      /usr/bin/ionice -c 2 -n 7 \
      /usr/bin/nice -n 10 \
      "${PY}" -u -m \
        pose_point_depth_mv.dataset_tools.restructure_single_object_dataset \
          --source_plan "${SOURCE_PLAN}" \
          --manifest "${TRAIN897_MANIFEST}" \
          --exclude_manifest "${OLD_VAL_MANIFEST}" \
          --exclude_manifest "${OLD_HOLDOUT_MANIFEST}" \
          "${RENDER_ARGS[@]}" \
          --reuse_mesh_audit "${TRAIN897_OUT}" \
          --source_plan_limit "objaverse=${FRESH_OBJAVERSE_CANDIDATES}" \
          --source_plan_limit omni=0 \
          --source_selection_seed "${SOURCE_SELECTION_SEED}" \
          --output_dir "${SOURCE_OUT}" \
          --workers "${AUDIT_WORKERS}" \
          --write_previews \
          --preview_frames 4 \
          --max_previews_per_tier 64
    echo "started: ${SOURCE_UNIT}.service"
  fi
  echo "source report (完成后): ${SOURCE_OUT}/report.json"
  echo "filtered source plan (draft):"
  echo "  ${SOURCE_OUT}/training_candidate_source_plan/build_plan.json"
fi

echo
echo "D3: 在另一终端监视源池重构进度"
echo "  watch -n 20 'date -u; systemctl --user is-active ${SOURCE_UNIT}.service || true; tail -n 8 ${SOURCE_LOG} 2>/dev/null || true'"
echo
echo "也可一次检查："
systemctl --user --no-pager --full status "${SOURCE_UNIT}.service" 2>/dev/null \
  | sed -n '1,12p' || true
tail -n 8 "${SOURCE_LOG}" 2>/dev/null || true

echo
echo "D4: 有新渲染分片完成后的增量快照（默认不自动运行）"
echo "有新 Omni/Objaverse marker 后，使用同一 seed/限额和新的不可变目录名："
cat <<EOF
  cd /home/zjr/Tracker
  PY=${PY}
  PREVIOUS=${SOURCE_OUT}
  NEXT=${RESTRUCTURE_ROOT}/source17435_render_more_20260731_v3

  \${PY} -u -m pose_point_depth_mv.dataset_tools.restructure_single_object_dataset \\
    --source_plan ${SOURCE_PLAN} \\
    --manifest ${TRAIN897_MANIFEST} \\
    --exclude_manifest ${OLD_VAL_MANIFEST} \\
    --exclude_manifest ${OLD_HOLDOUT_MANIFEST} \\
    --render_root ${OBJ_PILOT} \\
    --render_root ${OMNI_PILOT} \\
    --render_root ${BOTH_SMOKE} \\
    --reuse_mesh_audit \${PREVIOUS} \\
    --source_plan_limit objaverse=${FRESH_OBJAVERSE_CANDIDATES} \\
    --source_plan_limit omni=0 \\
    --source_selection_seed ${SOURCE_SELECTION_SEED} \\
    --output_dir \${NEXT} \\
    --workers ${AUDIT_WORKERS} \\
    --write_previews \\
    --preview_frames 4 \\
    --max_previews_per_tier 64
EOF

echo
echo "D5: 查看分层统计"
cat <<EOF
  ${PY} -c 'import json,sys; r=json.load(open(sys.argv[1])); print(json.dumps(r[\"summary\"],indent=2,ensure_ascii=False)); print(json.dumps(r[\"policy\"],indent=2,ensure_ascii=False))' ${TRAIN897_OUT}/report.json
  ${PY} -c 'import json,sys; r=json.load(open(sys.argv[1])); print(json.dumps(r[\"summary\"],indent=2,ensure_ascii=False)); print(json.dumps(r[\"training_candidate_source_plan\"],indent=2,ensure_ascii=False))' ${SOURCE_OUT}/report.json
EOF

echo
echo "D6: 人工覆写并冻结 reviewed 版本（不要把空模板直接当作最终结论）"
echo "模板允许只填写已审核的行；空 final_tier 行会被忽略。"
cat <<EOF
  cp ${TRAIN897_OUT}/manual_overrides_template.csv /data/zjr/train897_single_object_overrides_reviewed_v1.csv
  # 编辑 CSV：final_tier=A/B/C/R，并填写 reason、reviewer。

  ${PY} -u -m pose_point_depth_mv.dataset_tools.restructure_single_object_dataset \\
    --manifest ${TRAIN897_MANIFEST} \\
    --reuse_mesh_audit ${TRAIN897_OUT} \\
    --overrides_csv /data/zjr/train897_single_object_overrides_reviewed_v1.csv \\
    --output_dir ${RESTRUCTURE_ROOT}/train897_reviewed_v1 \\
    --workers ${AUDIT_WORKERS} \\
    --require_review_for_training \\
    --hash_source_files \\
    --write_previews
EOF

echo
echo "解释："
echo "  - D1/D2 的 A+B 产物都是 draft；可用于检查数量、渲染和训练预算，不能替代正式人工冻结。"
echo "  - D6 使用 --require_review_for_training 后，只有人工确认过的 A/B 才进入训练候选 manifest。"
echo "  - fresh Objaverse 只审计固定 ${FRESH_OBJAVERSE_CANDIDATES} 个候选；若 reviewed 总数已接近1k，不再扩大搜索。"
echo "  - source-plan-only Omni 被排除；所有已完成 marker 的新 Omni 会自动保留并进入下一不可变快照。"
echo "  - 如果要据此重训 SS/SLAT，应从新的 reviewed manifest/source plan 重新构建 cache；不要修改旧 cache。"
echo "current terminal remains open"
