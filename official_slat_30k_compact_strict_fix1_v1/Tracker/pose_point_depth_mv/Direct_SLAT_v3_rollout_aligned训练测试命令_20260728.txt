# Direct-SLAT v3 rollout-aligned：897 数据训练与测试命令
# 2026-07-28
#
# v3 相对 v2 的唯一训练语义改动：
# 1. teacher state 直接监督部署时的 post-CFG velocity，而非只监督 positive branch；
# 2. 以概率 0.25 做一次 native Euler detached rollout，并监督下一状态 velocity；
# 3. wrong support 不再要求“比 correct 更差”，而是回归同状态的 post-CFG stock；
# 4. wrong-support / support-dropout / rollout 三种辅助图互斥，峰值只保留
#    correct + 一个 auxiliary graph，适配 24-GB 3090；
# 5. checkpoint 冻结 training_semantics、CFG strength/interval、rollout 和各 loss 参数；
#    v3 Mesh 导出若使用不同 CFG strength/interval 会直接报错。
#
# 固定执行顺序：
#   S0 静态/CPU 回归 -> S1 单卡 5-step 真模型 smoke -> S2 smoke 审计
#   -> S3 双卡 100-step -> S4 post-CFG teacher/mechanism
#   -> S5 same-noise Mesh -> S6 可选 Pixal3D utility
#   -> 只有这些结果通过，才人工批准 S7 继续到 1000 steps。
#
# 注意：
# - 当前仍使用原 897-object 训练缓存，不使用 10k 数据。
# - S4 是 teacher/post-CFG 诊断，S5/S6 才是最终输出效益证据。
# - 不要把 S4 通过单独解释为模型已经优于 stock/Pixal3D。

run_logged() {
  LOG_PATH=$1
  shift
  set +e
  "$@" 2>&1 | tee "${LOG_PATH}"
  COMMAND_RC=${PIPESTATUS[0]}
  printf '%s\n' "${COMMAND_RC}" > "${LOG_PATH}.exit_code"
  return "${COMMAND_RC}"
}

assert counts["rollout_events"]>0
assert counts["post_cfg_teacher_events"]>0
for row in r["history"]:
  for key in ("loss","flow_loss","stock_loss","grad_norm"):
    assert math.isfinite(float(row[key]))
print(json.dumps({"passed":True,"step":r["step"],"counts":counts,
 "means":r["mechanism_training_audit"]["means"]},indent=2))' \
