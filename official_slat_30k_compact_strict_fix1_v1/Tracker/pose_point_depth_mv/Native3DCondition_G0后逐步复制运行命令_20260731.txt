Native3DCondition G0之后逐步复制运行命令
================================================

本文档不是脚本。每次只复制一个完整的 `(...)` 命令块到终端。
任何一个硬门失败都停止，不运行后面的命令。

S1. SS8 adapter-only训练（GPU 5）
==================================

(
  set -uo pipefail
  cd /home/zjr/Tracker
  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  LIFT=/data/zjr/native3d_condition_reviewed1k_inputs_20260730_v3/lifting_train/manifest.json
  G0=/data/zjr/native3d_condition_ss_slat_20260730_v1/g0_native_condition_v2/report.json
  OUT=/data/zjr/native3d_condition_ss_slat_20260730_v1/ss8_overfit_seed42_v2
  LOG=/data/zjr/native3d_condition_ss_slat_20260730_v1/logs/SS8_train_v2.log

  test -s "${LIFT}" || exit 96
  "${PY}" -c "import json; assert json.load(open('${G0}'))['passed'] is True" || exit 96
  if [ -s "${OUT}/report.json" ]; then
    echo "reuse complete SS8: ${OUT}"
  elif [ -e "${OUT}" ]; then
    echo "SS8 partial output exists; preserve and inspect: ${OUT}"
    exit 98
  else
    mkdir -p "$(dirname "${LOG}")"
    env \
      CUDA_VISIBLE_DEVICES=5 \
      HF_HUB_OFFLINE=1 \
      TRANSFORMERS_OFFLINE=1 \
      ATTN_BACKEND=flash_attn \
      SPCONV_ALGO=native \
      PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
      "${PY}" -u -m pose_point_depth_mv.train_native_ss_flow \
        --cache_manifest "${LIFT}" \
        --output_dir "${OUT}" \
        --pretrained Stable-X/trellis-vggt-v0-2 \
        --indices all \
        --max_objects 8 \
        --max_steps 200 \
        --save_every 200 \
        --log_every 5 \
        --grad_accum 1 \
        --seed 42 \
        --lr 1e-4 \
        --hidden_dim 128 \
        --feature_source dino \
        --control_probability 0.25 \
        --control_rank_weight 0.0 \
        --amp_dtype bf16 \
      2>&1 | tee "${LOG}"
    RC=${PIPESTATUS[0]}
    printf '%s\n' "${RC}" > "${LOG}.exit_code"
    echo "S1_RC=${RC}"
    exit "${RC}"
  fi
  echo "current terminal remains open"
)
