# Direct SLAT Flow（frozen SS step900 + zero-init support adapter + SLAT LoRA）
# 2026-07-22
#
# 本文件现在先从已有 Pixal3D canonical renders 构建本地多视图版 lh-slats。
# 每个对象一个 NPZ，包含 coords[N,3] 与 feats[N,8]；它不是官方 150-view 原始 release，
# 因此仍必须通过投影、frame identity 与 frozen native decoder 三类硬门。
# v1 存在 behind-camera DINO feature leakage，已禁止训练入口接纳；只能使用 v2。
# frame IoU 只是 support identity 检查，不是投影正确性证据。v2 另外强制
# in-image/mask 多视图覆盖门和半圈 camera-cyclic corruption sensitivity 门。
# 禁止把 stock SLAT 采样输出当作训练 target；那只会蒸馏 baseline。
#
# 首版训练使用 GT SLAT coords 做 teacher forcing；support 从 frozen corrected SS 按
# 64 -> native stem 32 -> SS/physical 16 的坐标链路 gather。推理/Mesh 才使用
# corrected SS decoder 产生的 coords。SLAT native image condition 与 decoder 均冻结。

run_logged() (
  local log_path=$1
  shift
  set +e
  "$@" 2>&1 | tee "${log_path}"
  local pipe_status=("${PIPESTATUS[@]}")
  local command_rc=${pipe_status[0]}
  local tee_rc=${pipe_status[1]}
  echo "command exit code: ${command_rc}"
  echo "tee exit code: ${tee_rc}"
  if [ "${command_rc}" -ne 0 ]; then
    return "${command_rc}"
  fi
  return "${tee_rc}"
)

MESH32=${RUN}/mesh_slat_support_val32_step${BEST_STEP}_seed424344_exploratory_v1

run_s6_mesh() {
  local output_dir=$1
  local max_objects=$2
  local output_log=${output_dir}.log
  local resume_args=()
  local command_rc
  local stamp

  if [ -s "${output_dir}/report.json" ]; then
    echo "S6 already complete; no rerun needed: ${output_dir}"
    return 0
  fi
  if [ -e "${output_dir}" ]; then
    echo "S6 resuming incomplete output: ${output_dir}"
    resume_args=(--resume)
  fi
  if [ -e "${output_log}" ]; then
    stamp=$(date +%Y%m%d_%H%M%S)
    if ! mv "${output_log}" "${output_log}.incomplete_${stamp}"; then
      echo "S6 could not preserve old log: ${output_log}"
      return 98
    fi
  fi

  if run_logged "${output_log}" \
    env \
      CUDA_VISIBLE_DEVICES=4 \
      HF_HUB_OFFLINE=1 \
      TRANSFORMERS_OFFLINE=1 \
      ATTN_BACKEND=flash_attn \
      SPCONV_ALGO=native \
      PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
    "${PY}" -u -m pose_point_depth_mv.export_direct_slat_mesh_pairs \
      --cache_manifest "${VAL_CACHE}/manifest.json" \
      --train_cache_manifest "${TRAIN_CACHE}/manifest.json" \
      --checkpoint "${BEST_CKPT}" \
      --output_dir "${output_dir}" \
      --pretrained "${PRETRAINED}" \
      --joint_seeds 42,43,44 \
      --max_objects "${max_objects}" \
      --steps 25 \
      --cfg_strength 5.0 \
      --cfg_interval 0.5,1.0 \
      --rescale_t 3.0 \
      --surface_samples 20000 \
      --bootstrap_samples 5000 \
      --amp_dtype bf16 \
      "${resume_args[@]}"
  then
    if [ ! -s "${output_dir}/report.json" ]; then
      echo "S6 command returned success but report.json is missing: ${output_dir}"
      return 95
    fi
    return 0
  else
    command_rc=$?
    echo "S6 failed with exit code ${command_rc}: ${output_dir}"
    return "${command_rc}"
  fi
}
