# T5. 验收 nested views、文件 hash、对象/阶段固定噪声
###############################################################################

cd /home/zjr/Tracker

PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
MATCHED_OUT=/data/zjr/matched_view_2_4_8_seen16_seed424344_protocol_v1
T5_RC=0

if [ ! -s "${MATCHED_OUT}/protocol.json" ]; then
  echo "T5 blocked: run T4 first"
  T5_RC=96
else
  "${PY}" -c \
    'import hashlib,json,sys
from pathlib import Path
p=Path(sys.argv[1]); r=json.load(open(p))
body=dict(r); expected=body.pop("protocol_sha256")
actual=hashlib.sha256(json.dumps(body,sort_keys=True,separators=(",",":"),ensure_ascii=False).encode()).hexdigest()
assert actual==expected
assert r["object_count"]==16 and r["view_policy"]["strictly_nested"] is True
for case in r["cases"]:
    p2=set(case["variants"]["2"]["positions_in_frozen_8"])
    p4=set(case["variants"]["4"]["positions_in_frozen_8"])
    p8=set(case["variants"]["8"]["positions_in_frozen_8"])
    assert p2 < p4 < p8
    for variant in case["variants"].values():
        for frame in variant["frames"]:
            for key in ("image","mask"):
                item=frame[key]; q=Path(item["path"])
                assert q.is_file()
                assert hashlib.sha256(q.read_bytes()).hexdigest()==item["sha256"]
    for noise in case["noise"].values():
        assert noise["ss_seed"] != noise["slat_seed"]
print(json.dumps({
  "passed":True,
  "objects":r["object_count"],
  "joint_seeds":r["joint_seeds"],
  "protocol_sha256":r["protocol_sha256"],
  "view_positions":r["view_policy"]["positions"],
},indent=2,ensure_ascii=False))' \
    "${MATCHED_OUT}/protocol.json"
  T5_RC=$?
fi

echo "T5_RC=${T5_RC}"
echo "注意: T5 通过只表示输入/noise protocol 已冻结，不是 2/4/8 模型效果完成"
echo "current terminal remains open"
