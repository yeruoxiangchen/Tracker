mkdir -p "${LOG_ROOT}"

run_logged() {
  LOG_PATH=$1
  shift
  set +e
  "$@" 2>&1 | tee "${LOG_PATH}"
  COMMAND_RC=${PIPESTATUS[0]}
  printf '%s\n' "${COMMAND_RC}" > "${LOG_PATH}.exit_code"
  return "${COMMAND_RC}"
}

# step100 已存在，会直接复用。step200/300/400 串行运行。

  "100:${CKPT100}:${TEACHER100}" \
  "200:${CKPT200}:${TEACHER200}" \
  "300:${CKPT300}:${TEACHER300}" \
  "400:${CKPT400}:${TEACHER400}"; do
  IFS=: read -r STEP CHECKPOINT OUTPUT_DIR <<<"${SPEC}"
  if [ -s "${OUTPUT_DIR}/report.json" ]; then
    echo "reuse teacher step${STEP}: ${OUTPUT_DIR}"
  elif [ -e "${OUTPUT_DIR}" ]; then
    echo "E1 blocked: partial teacher output exists; preserve and inspect: ${OUTPUT_DIR}"
  else
    run_logged "${LOG_ROOT}/teacher_step${STEP}.log" env \
      CUDA_VISIBLE_DEVICES="${GPU}" \
      HF_HUB_OFFLINE=1 \
      TRANSFORMERS_OFFLINE=1 \
      ATTN_BACKEND=flash_attn \
      SPCONV_ALGO=native \
      MPLCONFIGDIR=/tmp/matplotlib \
      NUMBA_CACHE_DIR=/tmp/numba_cache \
      TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
      PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
      "${PY}" -u -m pose_point_depth_mv.eval_direct_slat_flow \
        --cache_manifest "${VAL_CACHE}/manifest.json" \
        --train_cache_manifest "${TRAIN_CACHE}/manifest.json" \
        --checkpoint "${CHECKPOINT}" \
        --output_dir "${OUTPUT_DIR}" \
        --pretrained "${PRETRAINED}" \
        --max_objects 8 \
        --noise_seeds 42 \
        --t_values 0.5,0.7,0.9 \
        --amp_dtype bf16 \
        --bootstrap_samples 2000 \
        --min_object_win_rate 0.50 \
        --decision_profile report_only
    E1_ITEM_RC=$?
    echo "teacher step${STEP} rc=${E1_ITEM_RC}"
    if [ "${E1_ITEM_RC}" -ne 0 ]; then
      echo "E1 stopped at step${STEP}"
      break

  fi
done

           for row in r["rows"])

assert all(row["mesh"]["formal"] is False for row in r["rows"])

    for key in ("raw_ratio_over_cap_rate",
                "raw_delta_to_stock_rms_mean",
                "effective_delta_to_stock_rms_mean"):
        assert math.isfinite(float(active[key]))
print(json.dumps({
  "passed":True,
  "steps":[row["step"] for row in r["rows"]],
  "teacher_objects":8,
  "mesh_objects":6,
  "formal":False,
  "automatic_checkpoint_selection":
      r["decision"]["automatic_checkpoint_selection"],
},indent=2))' \

# 1. 最先看 curve.csv / summary.txt 中的 Mesh：
#    Chamfer improvement、F-score delta、normal delta、LCR delta。
#
# 2. teacher/rollout gain 上升，但 Mesh 不上升：
#    判为 proxy -> rollout/decoder mismatch，不能继续单纯增加训练步数。
#
# 3. raw-ratio-over-cap 持续升高、raw delta ratio 升高，而 effective ratio 与
#    Mesh 基本不变：
#    判为模型在推 residual 到 trust-region 边界。
#
# 4. 本协议只有 6 objects × 1 seed，因此无论哪个 checkpoint 看起来最好，
#    都只能选候选；随后必须冻结 fresh blind、更多对象、多 seed 协议。
#
# 5. 不要在 E1-E4 完成前删除任何已有 step100/400 产物，也不要覆盖 partial
#    output。若出现 partial，保留目录并查看对应 logs/*.log 和 .exit_code。
