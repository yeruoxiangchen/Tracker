#!/usr/bin/env bash
set -euo pipefail

# 说明
# ====
#
# 用途：把 CoarseModel/datasets 下的本地真实采集，转换为当前冻结的
# mixed No-VGGT Native-SS + Native-SLAT 部署输入，并生成 runtime-O / 稀疏世界
# 坐标两份 Mesh 供人工检查。
#
# 当前四个目录的事实：
# 1. GOOD_MESH_TEST：可直接使用 9 帧、865 点的固定相机稀疏模型。
# 2. reconviagen_20260514_071732：原 points3D.txt 为空，C1 会从固定相机位姿
#    和 RGB/mask 三角化；实测 60 个原始点、38 个掩码支持点。
# 3. reconviagen_20260513_022427：目前只有旧 Mesh，RGB/mask/相机帧为空；C2 会
#    明确记录为 rejected，不会伪造输入。
# 4. heimei：使用 29 帧、1656 点的 object-focused 稀疏模型，随后均匀选择 8 帧。
#
# 这不是有 GT 的定量评测。旧 CoarseModel / ReconViaGen Mesh 只会复制到
# “仅供人工参考_不作为GT”，不会参与推理、对齐或指标计算。
#
# 每一节可以单独复制执行。GPU 默认 5，请先用 nvidia-smi 改为空闲卡。


# C0. 静态检查与最小单元测试（CPU，约 1 秒）
# ===========================================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python

${PY} -m py_compile \
  pose_point_depth_mv/dataset_tools/prepare_coarsemodel_real_raw_cache.py \
  pose_point_depth_mv/package_coarsemodel_real_no_vggt_results.py

${PY} -m unittest -v \
  pose_point_depth_mv.dataset_tools.test_prepare_coarsemodel_real_raw_cache \
  pose_point_depth_mv.test_package_coarsemodel_real_no_vggt_results

echo "C0 complete"


# C1. 为 20260514 采集补建固定相机点云（CPU，通常几十秒）
# =======================================================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
ROOT=/data/zjr/coarsemodel_real_no_vggt_test_20260810_v1
DATASET=/home/zjr/Tracker/CoarseModel/datasets/reconviagen_20260514_071732
DERIVED=${ROOT}/derived_sparse/reconviagen_20260514_071732/0
REPORT=${ROOT}/derived_sparse/reconviagen_20260514_071732_report.json

if [ -s "${DERIVED}/points3D.txt" ] && [ -s "${DERIVED}/images.txt" ] && [ -s "${DERIVED}/cameras.txt" ]; then
  echo "reuse 20260514 derived sparse model: ${DERIVED}"
elif [ -e "${DERIVED}" ]; then
  echo "C1 blocked: partial derived sparse output exists; preserve and inspect: ${DERIVED}"
else
  ${PY} -u trellis_point_prior_mv/triangulate_slam_like_points.py \
    --datasets "${DATASET}" \
    --input_sparse_subdir sparse/0 \
    --output_sparse_subdir "${DERIVED}" \
    --max_frames 9 \
    --frame_select first \
    --max_features 8192 \
    --feature_mask_mode none \
    --matcher exhaustive \
    --ratio_test 0.82 \
    --min_pair_matches 8 \
    --max_reproj_error 6.0 \
    --min_triangulation_angle_deg 0.5 \
    --min_support_views 2 \
    --min_support_ratio 0.05 \
    --merge_voxel_size 0.001 \
    --min_output_points 40 \
    --max_output_points 50000 \
    --output_report "${REPORT}"
fi

wc -l "${DERIVED}/points3D.txt"
echo "C1 complete: ${DERIVED}"


# C2. 将四个目录适配为统一 raw cache（CPU，约 10 秒）
# =================================================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
ROOT=/data/zjr/coarsemodel_real_no_vggt_test_20260810_v1
DERIVED=${ROOT}/derived_sparse/reconviagen_20260514_071732/0
RAW=${ROOT}/raw_cache_v1

${PY} -u -m pose_point_depth_mv.dataset_tools.prepare_coarsemodel_real_raw_cache \
  --dataset /home/zjr/Tracker/CoarseModel/datasets/GOOD_MESH_TEST \
  --dataset /home/zjr/Tracker/CoarseModel/datasets/reconviagen_20260514_071732 \
  --dataset /home/zjr/Tracker/CoarseModel/datasets/reconviagen_20260513_022427 \
  --dataset /home/zjr/Tracker/CoarseModel/datasets/heimei \
  --sparse_override reconviagen_20260514_071732="${DERIVED}" \
  --output_dir "${RAW}" \
  --min_registered_pairs 8 \
  --allow_missing \
  --resume

${PY} -c '
import json,sys
r=json.load(open(sys.argv[1]))
accepted={x["object_id"] for x in r["objects"]}
rejected={x["dataset_name"] for x in r["rejections"]}
assert r["passed"] is True
assert accepted=={"GOOD_MESH_TEST","reconviagen_20260514_071732","heimei"}, accepted
assert "reconviagen_20260513_022427" in rejected, rejected
print(json.dumps({"accepted":sorted(accepted),"rejected":sorted(rejected)},indent=2,ensure_ascii=False))
' "${RAW}/raw_cache_report.json"

echo "C2 complete: ${RAW}/raw_cache_report.json"


# C3. 构建 8 视图 runtime-O 输入（CPU，约 5 秒）
# =============================================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
ROOT=/data/zjr/coarsemodel_real_no_vggt_test_20260810_v1
RAW=${ROOT}/raw_cache_v1
RUNTIME=${ROOT}/runtime_o_min32_v1

${PY} -u -m pose_point_depth_mv.dataset_tools.prepare_omni_real_runtime_inputs \
  --raw_cache_report "${RAW}/raw_cache_report.json" \
  --output_dir "${RUNTIME}" \
  --selected_view_count 8 \
  --view_selection_policy lexical_even_valid_mask_fallback \
  --min_object_points 32 \
  --min_mask_observations 2 \
  --min_mask_support_ratio 0.50 \
  --min_completed_objects 3 \
  --resume

${PY} -c '
import json,sys
r=json.load(open(sys.argv[1]))
assert r["passed"] is True and r["completed_object_count"]==3
print(json.dumps({x["object_key"]:x["runtime_frame_stats"]["support"]["mask_supported_point_count"] for x in r["objects"]},indent=2,ensure_ascii=False))
' "${RUNTIME}/runtime_input_manifest.json"

echo "注意：32 点门槛只为保留 20260514 的探索性测试；训练和正式评测仍使用原质量门。"
echo "C3 complete: ${RUNTIME}/runtime_input_manifest.json"


# C4. 生成 DINO-only 模型输入（单 GPU，约 1--3 分钟）
# =================================================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
GPU=${GPU:-5}
ROOT=/data/zjr/coarsemodel_real_no_vggt_test_20260810_v1
RUNTIME=${ROOT}/runtime_o_min32_v1
MODEL_INPUT=${ROOT}/dino_only_inputs_v1
RESUME=()
if [ -e "${MODEL_INPUT}" ]; then RESUME=(--resume); fi

CUDA_VISIBLE_DEVICES="${GPU}" \
HF_HUB_OFFLINE=1 \
TRANSFORMERS_OFFLINE=1 \
ATTN_BACKEND=flash_attn \
SPCONV_ALGO=native \
MPLCONFIGDIR=/tmp/matplotlib \
NUMBA_CACHE_DIR=/tmp/numba_cache \
TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
${PY} -u -m pose_point_depth_mv.dataset_tools.prepare_omni_real_dino_only_model_inputs \
  --runtime_input_manifest "${RUNTIME}/runtime_input_manifest.json" \
  --output_dir "${MODEL_INPUT}" \
  --device cuda \
  "${RESUME[@]}"

echo "C4 complete: ${MODEL_INPUT}/model_input_manifest.json"


# C5. 用当前冻结 mixed No-VGGT SS+SLat 推理（单 GPU）
# ================================================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
GPU=${GPU:-5}
ROOT=/data/zjr/coarsemodel_real_no_vggt_test_20260810_v1
RUN=/data/zjr/native_no_vggt_mixed_real376_synth868_20260808_v1
MODEL_INPUT=${ROOT}/dino_only_inputs_v1
INFER=${ROOT}/native_no_vggt_mixed_seed42_v1
SS_CKPT=${RUN}/ss_mixed_step2000_seed42_1gpu_v1/checkpoints/step_002000.pt
SLAT_CKPT=${RUN}/slat_mixed_step2000_seed42_2gpu_v1/checkpoints/step_002000.pt
SS_CONTRACT=${RUN}/contracts/ss_real_full_ema_v1.json
SLAT_CONTRACT=${RUN}/contracts/slat_real_full_ema_v1.json
STOCK_FREEZE=/data/zjr/native_slat_genrecon_v2_mixed1k_20260802_v1/stock_slat_freeze_v2.json

if [ -s "${INFER}/inference_manifest.json" ]; then
  echo "reuse complete inference: ${INFER}/inference_manifest.json"
else
  CUDA_VISIBLE_DEVICES="${GPU}" \
  HF_HUB_OFFLINE=1 \
  TRANSFORMERS_OFFLINE=1 \
  ATTN_BACKEND=flash_attn \
  SPCONV_ALGO=native \
  MPLCONFIGDIR=/tmp/matplotlib \
  NUMBA_CACHE_DIR=/tmp/numba_cache \
  TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
  PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
  ${PY} -u -m pose_point_depth_mv.infer_omni_real_native_no_vggt_mixed \
    --model_input_manifest "${MODEL_INPUT}/model_input_manifest.json" \
    --native_ss_checkpoint "${SS_CKPT}" \
    --native_slat_checkpoint "${SLAT_CKPT}" \
    --ss_migration_contract "${SS_CONTRACT}" \
    --slat_migration_contract "${SLAT_CONTRACT}" \
    --stock_slat_freeze "${STOCK_FREEZE}" \
    --output_dir "${INFER}" \
    --seeds 42 \
    --weights ema \
    --amp_dtype bf16 \
    --device cuda
fi

echo "C5 complete: ${INFER}/inference_manifest.json"


# C6. 整理输入、runtime-O Mesh、世界坐标 Mesh 和旧参考 Mesh（CPU）
# ===============================================================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
ROOT=/data/zjr/coarsemodel_real_no_vggt_test_20260810_v1
RAW=${ROOT}/raw_cache_v1
RUNTIME=${ROOT}/runtime_o_min32_v1
INFER=${ROOT}/native_no_vggt_mixed_seed42_v1
VIS=/home/zjr/Tracker/pose_point_depth_mv/outputs/可视化Mesh/CoarseModel真实采集_NoVGGT测试_20260810_v1

${PY} -u -m pose_point_depth_mv.package_coarsemodel_real_no_vggt_results \
  --inference_manifest "${INFER}/inference_manifest.json" \
  --runtime_input_manifest "${RUNTIME}/runtime_input_manifest.json" \
  --raw_cache_report "${RAW}/raw_cache_report.json" \
  --output_dir "${VIS}"

echo "可视化目录: ${VIS}"
find "${VIS}" -maxdepth 3 -type f -print | sort


# C7. 最终一次只读核验
# ===================

cd /home/zjr/Tracker
PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
REPORT=/home/zjr/Tracker/pose_point_depth_mv/outputs/可视化Mesh/CoarseModel真实采集_NoVGGT测试_20260810_v1/report.json

${PY} -c '
import json,sys
r=json.load(open(sys.argv[1]))
assert r["passed"] is True and r["object_count"]==3
print(json.dumps({"passed":True,"objects":[x["object_key"] for x in r["cases"]],"scope_guard":r["scope_guard"]},indent=2,ensure_ascii=False))
' "${REPORT}"

echo "current terminal remains open"
