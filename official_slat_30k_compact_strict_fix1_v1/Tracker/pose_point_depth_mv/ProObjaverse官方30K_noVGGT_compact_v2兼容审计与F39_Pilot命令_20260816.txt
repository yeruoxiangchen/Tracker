B. F39 root环境：30K protocol freeze + 8-GPU 256-object compact pilot
======================================================================

先修改/确认：

  RAW
  SELECTION
  SS_REPORT
  FREEZE

PROTOCOL 和 PILOT 必须是尚不存在的新目录，避免覆盖旧证据。

set +e
(
  set -euo pipefail

  PROJECT=/root/Tracker
  CONDA_SH=/root/miniconda3/etc/profile.d/conda.sh
  PY=/root/miniconda3/envs/reconviagen-bw/bin/python

  test "$(id -u)" -eq 0
  test -d "${PROJECT}"
  test -r "${PROJECT}/pose_point_depth_mv/prepare_proobjaverse_official_slat_compact_cache.py"
  test -r "${CONDA_SH}"
  test -x "${PY}"

  cd "${PROJECT}"

  source "${CONDA_SH}"
  conda activate reconviagen-bw

  export PYTHONPATH="${PROJECT}:${PROJECT}/ReconViaGen:${PROJECT}/ReconViaGen/wheels/vggt"
  export CUDA_HOME=/usr/local/cuda
  export PATH="${CUDA_HOME}/bin:${PATH}"
  export LD_LIBRARY_PATH="${CUDA_HOME}/lib64:${LD_LIBRARY_PATH:-}"
  export CUMM_CUDA_ARCH_LIST=12.0
  export SPCONV_ALGO=native
  export ATTN_BACKEND=flash_attn
  export TORCH_FORCE_NO_WEIGHTS_ONLY_LOAD=1
  export TORCH_HOME=/root/.cache/torch
  export HF_HUB_OFFLINE=1
  export TRANSFORMERS_OFFLINE=1
  export OMP_NUM_THREADS=4
  export MKL_NUM_THREADS=4
  export OPENBLAS_NUM_THREADS=1
  export NUMEXPR_NUM_THREADS=1

  RAW=/root/autodl-tmp/ProObjaverse-300K-ReconViaGen-30K
  SELECTION=/root/autodl-tmp/ProObjaverse-300K-ReconViaGen-30K-state/combined_audit/combined_selection_30k.json

  # 下面两项必须替换为F39上的真实、内容一致路径。
  SS_REPORT=/root/autodl-tmp/path/to/original_2k_training_native_ss_report.json
  FREEZE=/root/autodl-tmp/path/to/stock_slat_freeze_v2.json

  PROTOCOL=/root/autodl-tmp/proobjaverse_official_slat_protocol30k_seed20260813_v1
  PILOT=/root/autodl-tmp/proobjaverse_official_slat_compact_v2_pilot256_8gpu_v1

  echo "===== preflight guards ====="

  test -x "${PY}"
  test -d "${RAW}"
  test -s "${SELECTION}"
  test -s "${SS_REPORT}"
  test -s "${FREEZE}"
  test ! -e "${PROTOCOL}"
  test ! -e "${PILOT}"

  nvidia-smi \
    --query-gpu=index,name,memory.used,memory.total,utilization.gpu \
    --format=csv,noheader,nounits

  GPU_COUNT=$(nvidia-smi --query-gpu=index --format=csv,noheader,nounits | wc -l)
  test "${GPU_COUNT}" -eq 8

  echo "===== freeze exact 30K protocol ====="

  "${PY}" -u -m pose_point_depth_mv.proobjaverse_official_slat_protocol \
    freeze \
    --data_root "${RAW}" \
    --selection "${SELECTION}" \
    --output_dir "${PROTOCOL}" \
    --pair_count 30000 \
    --train_count 29872 \
    --seed 20260813

  SPLIT=${PROTOCOL}/train.json
  test -s "${PROTOCOL}/protocol.json"
  test -s "${SPLIT}"

  mkdir -p "${PILOT}/logs"
  GPU_LOG=${PILOT}/gpu_memory_samples.csv
  START=$(date +%s)

  echo "===== start 1-second GPU-memory sampler ====="

  (
    while true; do
      nvidia-smi \
        --query-gpu=timestamp,index,memory.used \
        --format=csv,noheader,nounits
      sleep 1
    done
  ) >"${GPU_LOG}" &
  MONITOR_PID=$!

  echo "===== launch 8 workers / 256 total objects ====="

  PIDS=()
  for WORKER in 0 1 2 3 4 5 6 7; do
    CUDA_VISIBLE_DEVICES="${WORKER}" \
    "${PY}" -u -m \
      pose_point_depth_mv.prepare_proobjaverse_official_slat_compact_cache \
      --split_manifest "${SPLIT}" \
      --native_ss_report "${SS_REPORT}" \
      --stock_slat_freeze "${FREEZE}" \
      --output_dir "${PILOT}" \
      --dino_model dinov2_vitl14_reg \
      --selected_views 8 \
      --max_objects 256 \
      --ss_context_tokens 4096 \
      --worker_index "${WORKER}" \
      --worker_count 8 \
      --materialize_only \
      >"${PILOT}/logs/worker_${WORKER}.log" 2>&1 &
    WORKER_PID=$!
    PIDS+=("${WORKER_PID}")
    echo "worker=${WORKER} pid=${WORKER_PID} log=${PILOT}/logs/worker_${WORKER}.log"
  done

  WORKER_RC=0
  for PID in "${PIDS[@]}"; do
    if ! wait "${PID}"; then
      WORKER_RC=1
    fi
  done

  kill "${MONITOR_PID}" 2>/dev/null || true
  wait "${MONITOR_PID}" 2>/dev/null || true

  test "${WORKER_RC}" -eq 0

  echo "===== finalize immutable compact manifests ====="

  CUDA_VISIBLE_DEVICES=0 \
  "${PY}" -u -m \
    pose_point_depth_mv.prepare_proobjaverse_official_slat_compact_cache \
    --split_manifest "${SPLIT}" \
    --native_ss_report "${SS_REPORT}" \
    --stock_slat_freeze "${FREEZE}" \
    --output_dir "${PILOT}" \
    --dino_model dinov2_vitl14_reg \
    --selected_views 8 \
    --max_objects 256 \
    --ss_context_tokens 4096 \
    --finalize_only

  END=$(date +%s)
  ELAPSED=$((END - START))

  test -s "${PILOT}/slat_manifest.json"
  test -s "${PILOT}/lifting_manifest.json"
  test -s "${PILOT}/report.json"

  echo "===== pilot statistics ====="

  "${PY}" - "${PILOT}" "${ELAPSED}" "${GPU_LOG}" <<'PY'
import csv
import json
import sys
from pathlib import Path

root = Path(sys.argv[1])
elapsed = int(sys.argv[2])
gpu_log = Path(sys.argv[3])

report = json.loads((root / "report.json").read_text(encoding="utf-8"))
objects = int(report["object_count"])
assert report["passed"] is True
assert objects == 256

cache_files = []
for path in root.rglob("*"):
    if not path.is_file():
        continue
    relative = path.relative_to(root)
    if relative.parts[0] == "logs":
        continue
    if path.name in {"gpu_memory_samples.csv", "pilot_summary.json"}:
        continue
    cache_files.append(path)

total_bytes = sum(path.stat().st_size for path in cache_files)
payload_bytes = int(report["compact_payload_bytes"])

peak_by_gpu = {}
with gpu_log.open(newline="", encoding="utf-8") as handle:
    for row in csv.reader(handle):
        if len(row) < 3:
            continue
        try:
            gpu = int(row[-2].strip())
            used_mib = int(row[-1].strip())
        except ValueError:
            continue
        peak_by_gpu[gpu] = max(peak_by_gpu.get(gpu, 0), used_mib)

objects_per_hour = objects * 3600.0 / elapsed
scale = 30000.0 / objects

summary = {
    "passed": True,
    "object_count": objects,
    "total_cache_bytes": total_bytes,
    "compact_payload_bytes": payload_bytes,
    "total_MiB_per_object": total_bytes / objects / 2**20,
    "payload_MiB_per_object": payload_bytes / objects / 2**20,
    "elapsed_seconds": elapsed,
    "objects_per_hour": objects_per_hour,
    "sampled_peak_gpu_memory_MiB": peak_by_gpu,
    "peak_memory_contract": (
        "1-second nvidia-smi system-wide sampled high-water mark; "
        "diagnostic, not exact torch allocator peak"
    ),
    "projected_full_30k_total_cache_GiB": total_bytes * scale / 2**30,
    "projected_full_30k_payload_GiB": payload_bytes * scale / 2**30,
    "projected_full_30k_hours": elapsed * scale / 3600.0,
}

(root / "pilot_summary.json").write_text(
    json.dumps(summary, indent=2, ensure_ascii=False) + "\n",
    encoding="utf-8",
)
print(json.dumps(summary, indent=2, ensure_ascii=False))
PY

  echo "F39 256-OBJECT COMPACT PILOT PASS"
  echo "summary=${PILOT}/pilot_summary.json"
)
B_RC=$?
echo "B_RC=${B_RC} (0=PASS；非0时终端仍保持打开，请检查worker日志)"
echo "current terminal remains open"
