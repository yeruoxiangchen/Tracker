# 冻结原则：

###############################################################################
# R4. 只读盘点11个归档中的实际video object及共享字段合同
###############################################################################

set +e
set +u
set +o pipefail
(
  set -euo pipefail
  cd /home/zjr/Tracker

  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  ROOT=/data/zjr/omni_real_video_pilot_20260804_v1
  ARCHIVES=/data/zjr/OmniObject3D/raw/omniobject3d___OmniObject3D-New/raw/videos_processed
  SCANS=/data/zjr/OmniObject3D/raw_scans_extracted_omni215_v1/categories
  CATEGORIES=${ROOT}/manifests/r2_categories_35obj.txt
  P7=${ROOT}/audit/R2_DOWNLOAD_SUMMARY.json
  Q3=${ROOT}/audit/R3_STRATIFIED_SAMPLE_CONTRACT_PASS.json
  OUT=${ROOT}/R4_actual_video_object_inventory_v1
  LOG=${ROOT}/logs/R4_actual_video_object_inventory.log

  test -x "${PY}" || exit 96
  test -s "${P7}" || exit 96
  test -s "${Q3}" || exit 96
  "${PY}" -c 'import json,sys; p=json.load(open(sys.argv[1])); assert p["passed"] is True' "${Q3}"

  if [ -s "${OUT}/inventory.json" ]; then
    "${PY}" -c \
      'import json,sys; p=json.load(open(sys.argv[1])); print(json.dumps({k:p[k] for k in ("passed","category_count","video_object_count","passed_object_count","scope_guard")},indent=2,ensure_ascii=False)); assert p["passed"] is True' \
      "${OUT}/inventory.json"
  elif [ -e "${OUT}" ]; then
    echo "R4 partial output exists; preserve and inspect: ${OUT}"
    exit 98
  else
    mkdir -p "$(dirname "${LOG}")"
    "${PY}" -u -m pose_point_depth_mv.dataset_tools.prepare_omni_real_video_cache \
      inventory \
      --archive_root "${ARCHIVES}" \
      --category_list "${CATEGORIES}" \
      --scan_root "${SCANS}" \
      --p7_report "${P7}" \
      --output_dir "${OUT}"
  fi
) 2>&1 | tee /data/zjr/omni_real_video_pilot_20260804_v1/logs/R4_actual_video_object_inventory.log

R4_RC=${PIPESTATUS[0]}
echo "R4_RC=${R4_RC}"
if [ "${R4_RC}" -eq 0 ]; then
  echo "R4 passed; use video_object_count, not the old scan_object_count=35"
else
  echo "R4 failed; inspect the report/log; current terminal remains open"
fi
echo "current terminal remains open"


###############################################################################
# R5. 固定6对象：mask内COLMAP点 -> Scan.obj proper Sim(3)自动诊断
###############################################################################

set +e
set +u
set +o pipefail
(
  set -euo pipefail
  cd /home/zjr/Tracker

  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  ROOT=/data/zjr/omni_real_video_pilot_20260804_v1
  SAMPLE=${ROOT}/sample_audit_stratified6_v1/sample_audit.json
  CONTRACT=${ROOT}/audit/R3_STRATIFIED_SAMPLE_CONTRACT_PASS.json
  OUT=${ROOT}/R5_sim3_stratified6_v1
  LOG=${ROOT}/logs/R5_sim3_stratified6.log

  if [ -s "${OUT}/sim3_report.json" ]; then
    "${PY}" -c \
      'import json,sys; p=json.load(open(sys.argv[1])); print(json.dumps({"automatic_passed":p["automatic_passed"],"sample_count":p["sample_count"],"automatic_pass_count":p["automatic_pass_count"],"thresholds":p["thresholds"],"objects":[{"object_id":r["object_id"],"points":r["source_stats"]["mask_supported_point_count"],"scale":r["similarity_scale"],"median":r["median_normalized"],"p90":r["p90_normalized"],"inlier":r["inlier_rate_3pct"],"passed":r["automatic_passed"]} for r in p["objects"]]},indent=2,ensure_ascii=False)); assert p["automatic_passed"] is True' \
      "${OUT}/sim3_report.json"
  elif [ -e "${OUT}" ]; then
    echo "R5 partial output exists; preserve and inspect: ${OUT}"
    exit 98
  else
    env \
      MPLCONFIGDIR=/tmp/matplotlib \
      NUMBA_CACHE_DIR=/tmp/numba_cache \
      "${PY}" -u -m pose_point_depth_mv.dataset_tools.audit_omni_real_video_sim3 \
        --sample_report "${SAMPLE}" \
        --contract_report "${CONTRACT}" \
        --output_dir "${OUT}" \
        --mask_threshold 127 \
        --min_mask_observations 2 \
        --min_mask_ratio 0.60 \
        --min_object_points 100 \
        --max_source_points 10000 \
        --max_scan_vertices 100000 \
        --icp_iterations 20 \
        --icp_trim_fraction 0.80 \
        --max_median_normalized 0.03 \
        --max_p90_normalized 0.08 \
        --min_inlier_rate_3pct 0.70
  fi
) 2>&1 | tee /data/zjr/omni_real_video_pilot_20260804_v1/logs/R5_sim3_stratified6.log

R5_RC=${PIPESTATUS[0]}
echo "R5_RC=${R5_RC}"
if [ "${R5_RC}" -eq 0 ]; then
  echo "R5 automatic gate passed; run R5P and inspect six alignment previews"
else
  echo "R5 did not pass; do not lower thresholds or run R6 before diagnosis"
fi
echo "current terminal remains open"


###############################################################################
# R5P. 复制6张三视图对齐图到Tracker，供一次性抽样人工检查
###############################################################################

set +e
set +u
set +o pipefail
(
  set -euo pipefail
  SRC=/data/zjr/omni_real_video_pilot_20260804_v1/R5_sim3_stratified6_v1
  DST=/home/zjr/Tracker/pose_point_depth_mv/outputs/omni_real_video_sim3_stratified6_20260804_v1

  test -s "${SRC}/sim3_report.json" || exit 96
  mkdir -p "${DST}"
  for OBJECT_DIR in "${SRC}"/*; do
    test -d "${OBJECT_DIR}" || continue
    OBJECT_ID=$(basename "${OBJECT_DIR}")
    test -s "${OBJECT_DIR}/alignment_xyz.png" || continue
    cp -a "${OBJECT_DIR}/alignment_xyz.png" "${DST}/${OBJECT_ID}_alignment_xyz.png"
  done
  FINAL_COUNT=$(find "${DST}" -maxdepth 1 -type f -name '*_alignment_xyz.png' | wc -l)
  echo "alignment_preview_count=${FINAL_COUNT}"
  test "${FINAL_COUNT}" -eq 6
  find "${DST}" -maxdepth 1 -type f -name '*_alignment_xyz.png' -print | sort
)
R5P_RC=$?
echo "R5P_RC=${R5P_RC}"
echo "inspect: /home/zjr/Tracker/pose_point_depth_mv/outputs/omni_real_video_sim3_stratified6_20260804_v1"
echo "Gray must outline the same object surface as red; reject obvious axis/part mismatch or scale collapse."
echo "current terminal remains open"


###############################################################################
# R5G. 人工看完6张对齐图后冻结抽样Sim(3)合同；默认不通过
###############################################################################

set +e
set +u
set +o pipefail
(
  set -euo pipefail
  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  ROOT=/data/zjr/omni_real_video_pilot_20260804_v1
  REPORT=${ROOT}/R5_sim3_stratified6_v1/sim3_report.json
  OUTPUT=${ROOT}/audit/R5_SIM3_STRATIFIED_SAMPLE_PASS.json

  # 看完6张图后改成1。默认0只结束本子shell。
  SIM3_MANUAL_PASS=1

  if [ "${SIM3_MANUAL_PASS}" -ne 1 ]; then
    echo "R5G blocked: inspect all six alignment previews, then set SIM3_MANUAL_PASS=1"
    exit 95
  fi

  "${PY}" -c '
import json,sys
from datetime import datetime,timezone
from pathlib import Path
report_path=Path(sys.argv[1]).resolve(); output=Path(sys.argv[2]).resolve()
report=json.loads(report_path.read_text(encoding="utf-8"))
assert report["format"]=="pose_point_depth_mv.omni_real_video_sample_sim3.v1"
assert report["sample_count"]==6 and report["automatic_passed"] is True
payload={"format":"pose_point_depth_mv.omni_real_video_sim3_contract.v1","decided_at_utc":datetime.now(timezone.utc).isoformat(),"sim3_report":str(report_path),"sample_count":6,"manual_alignment_preview_passed":True,"future_policy":"no per-object manual alignment audit; batch automatic residual summaries and isolate failures","passed":True}
output.parent.mkdir(parents=True,exist_ok=True)
output.write_text(json.dumps(payload,indent=2,ensure_ascii=False)+"\n",encoding="utf-8")
print(json.dumps(payload,indent=2,ensure_ascii=False))
' "${REPORT}" "${OUTPUT}"
)
R5G_RC=$?
echo "R5G_RC=${R5G_RC}"
if [ "${R5G_RC}" -eq 0 ]; then
  echo "R5G passed; R6 raw cache is now allowed"
else
  echo "R5G blocked/failed; current terminal remains open"
fi
echo "current terminal remains open"
