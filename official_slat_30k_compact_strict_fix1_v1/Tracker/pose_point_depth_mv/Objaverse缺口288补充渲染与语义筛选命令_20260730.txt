#!/usr/bin/env bash
# 旧897语义二筛后的Objaverse缺口补充：288 candidates -> reviewed约1k

set -euo pipefail
cd /home/zjr/Tracker

PY=${PY:-/home/zjr/anaconda3/envs/reconviagen/bin/python}
BLENDER=${BLENDER:-/tmp/blender-3.0.1-linux-x64/blender}
GPU_LIST_CSV=${GPU_LIST_CSV:-1,4,5,6}

SOURCE_PLAN=/data/zjr/dataset10k_20260726/source_freeze_obj11546_omni215_v2/build_plan.json
MESH_AUDIT=/data/zjr/single_object_restructure_20260730/objaverse29999_render256_draft_v1/objects.json
TRAIN897=/data/zjr/reconvggt_pointpose_v9_ssfixed_odsplit_20260712/manifests/train.json
OLD_VAL=/data/zjr/reconvggt_pointpose_v9_ssfixed_odsplit_20260712/manifests/val.json
OLD_HOLDOUT=/data/zjr/reconvggt_pointpose_v9_ssfixed_odsplit_20260712/manifests/holdout.json
OLD_OBJ_RENDER=/data/zjr/dataset10k_20260726/pilot_objaverse256_cyclescuda_v2
OLD_BOTH_SMOKE=/data/zjr/dataset10k_20260726/smoke_both16_cyclescuda_v4

GAP_PLAN=/home/zjr/Tracker/pose_point_depth_mv/objaverse_gap288_source_plan_20260730_v1
GAP_RENDER=/data/zjr/dataset10k_20260726/gap_objaverse288_cyclescuda_v1
GAP_PREVIEW=/home/zjr/Tracker/pixal3d_multiview/outputs/data_previews/gap_objaverse288_cyclescuda_v1
GAP_LOG=/data/zjr/dataset10k_20260726/logs/gap_objaverse288_cyclescuda_v1
GAP_REVIEW=/home/zjr/Tracker/pose_point_depth_mv/objaverse_gap288_semantic_review_20260730_v1

echo "G0: static checks and tests"
"${PY}" -m py_compile \
  pose_point_depth_mv/dataset_tools/prepare_objaverse_gap_source_plan.py \
  pose_point_depth_mv/dataset_tools/prepare_objaverse_semantic_review.py \
  pose_point_depth_mv/dataset_tools/run_mixed_multiview_render_worker.py
"${PY}" -m unittest -v \
  pose_point_depth_mv.dataset_tools.test_prepare_objaverse_gap_source_plan \
  pose_point_depth_mv.dataset_tools.test_prepare_objaverse_semantic_review
echo "G0_RC=0"

echo "G1: freeze exactly 16 x 18 fresh Objaverse candidates"
if [ -s "${GAP_PLAN}/report.json" ] && [ -s "${GAP_PLAN}/build_plan.json" ]; then
  echo "reuse immutable gap plan: ${GAP_PLAN}"
elif [ -e "${GAP_PLAN}" ]; then
  echo "G1 blocked: partial immutable plan exists: ${GAP_PLAN}"
  exit 98
else
  "${PY}" -u -m pose_point_depth_mv.dataset_tools.prepare_objaverse_gap_source_plan \
    --source_plan "${SOURCE_PLAN}" \
    --mesh_audit_objects "${MESH_AUDIT}" \
    --exclude_manifest "${TRAIN897}" \
    --exclude_manifest "${OLD_VAL}" \
    --exclude_manifest "${OLD_HOLDOUT}" \
    --exclude_render_root "${OLD_OBJ_RENDER}" \
    --exclude_render_root "${OLD_BOTH_SMOKE}" \
    --allowed_auto_tier A \
    --allowed_auto_tier B \
    --selection_seed 20260730 \
    --shard_count 16 \
    --objects_per_shard 18 \
    --output_dir "${GAP_PLAN}"
fi
echo "G1_RC=0"

echo "G2: launch four Objaverse-only workers"
IFS=',' read -r -a GPU_LIST <<<"${GPU_LIST_CSV}"
if [ "${#GPU_LIST[@]}" -ne 4 ]; then
  echo "G2 blocked: exactly four GPUs are required; got ${#GPU_LIST[@]}"
  exit 96
fi
if [ ! -x "${BLENDER}" ]; then
  echo "G2 blocked: Blender is missing: ${BLENDER}"
  exit 96
fi
for WORKER_INDEX in "${!GPU_LIST[@]}"; do
  GPU_INDEX=${GPU_LIST[${WORKER_INDEX}]}
  UNIT=tracker-objgap288-cyclescuda-w${WORKER_INDEX}-v1
  if systemctl --user is-active --quiet "${UNIT}.service"; then
    echo "already active: ${UNIT}.service"
    continue
  fi
  systemctl --user reset-failed "${UNIT}.service" >/dev/null 2>&1 || true
  systemd-run --user \
    --unit="${UNIT}" \
    --collect \
    --property=WorkingDirectory=/home/zjr/Tracker \
    /usr/bin/env \
      CUDA_VISIBLE_DEVICES="${GPU_INDEX}" \
      HF_HUB_OFFLINE=1 \
      TRANSFORMERS_OFFLINE=1 \
      PYOPENGL_PLATFORM=egl \
      ATTN_BACKEND=flash_attn \
      SPCONV_ALGO=native \
      MPLCONFIGDIR=/tmp/matplotlib \
      NUMBA_CACHE_DIR=/tmp/numba_cache \
      TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
      PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
    "${PY}" -u pose_point_depth_mv/dataset_tools/run_mixed_multiview_render_worker.py \
      --source_plan "${GAP_PLAN}/build_plan.json" \
      --render_root "${GAP_RENDER}" \
      --preview_root "${GAP_PREVIEW}" \
      --log_root "${GAP_LOG}" \
      --worker_index "${WORKER_INDEX}" \
      --num_workers 4 \
      --sources objaverse \
      --python "${PY}" \
      --blender_path "${BLENDER}" \
      --blender_engine CYCLES \
      --blender_samples 16 \
      --blender_bounds_tolerance 0.001 \
      --blender_cycles_device CUDA \
      --seed 20260730 \
      --sequences_per_object 2 \
      --candidate_views 24 \
      --selected_views 8 \
      --vis_count_per_shard 2
  echo "started ${UNIT}.service on physical GPU ${GPU_INDEX}"
done
echo "G2_RC=0"

if [ -d "${GAP_RENDER}/objaverse" ]; then
  DONE=$(find "${GAP_RENDER}/objaverse" -mindepth 2 -maxdepth 2 \
    -type f -name _WORKER_COMPLETE.json | wc -l)
else
  DONE=0
fi
echo "G3: completed shards ${DONE}/16"
echo "monitor:"
echo "  watch -n 20 '${PY} -u pose_point_depth_mv/dataset_tools/monitor_mixed_render_shards.py --render_root ${GAP_RENDER} --log_root ${GAP_LOG} --source objaverse --expected_shards 16 --expected_tasks_per_shard 36 --num_workers 4 --unit tracker-objgap288-cyclescuda-w0-v1.service --unit tracker-objgap288-cyclescuda-w1-v1.service --unit tracker-objgap288-cyclescuda-w2-v1.service --unit tracker-objgap288-cyclescuda-w3-v1.service --gpu_index 1 --gpu_index 4 --gpu_index 5 --gpu_index 6'"

if [ "${DONE}" -eq 16 ]; then
  if [ -s "${GAP_REVIEW}/candidates.json" ]; then
    echo "reuse pending semantic review: ${GAP_REVIEW}"
  elif [ -e "${GAP_REVIEW}" ]; then
    echo "G4 blocked: partial review workspace exists: ${GAP_REVIEW}"
    exit 98
  else
    "${PY}" -u -m pose_point_depth_mv.dataset_tools.prepare_objaverse_semantic_review \
      prepare-render \
      --render_root "${GAP_RENDER}" \
      --source_group gap_objaverse288 \
      --review_prefix G \
      --output_dir "${GAP_REVIEW}" \
      --batch_size 20 \
      --columns 2
  fi
  echo "G4_RC=0"
  echo "review sheets: ${GAP_REVIEW}/batches/gap_objaverse288"
  echo "decision CSV: ${GAP_REVIEW}/review_decisions.csv"
else
  echo "G4 not started; wait for all 16 immutable completion markers"
fi

echo "current terminal remains open"
