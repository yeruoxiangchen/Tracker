A3. 完成后查看并汇总Mesh到outputs2
==================================

cd /home/zjr/Tracker

PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
AUDIT=/data/zjr/proobjaverse_official_native_ss_train2000_20260815_v1/dev48_support_explosion1_lcr_worst3_audit_20260815_v1
DEST=/home/zjr/Tracker/pose_point_depth_mv/outputs2/NativeSS_support爆炸与LCR最差3例审计_20260815_v1

cat "${AUDIT}/summary.txt"
${PY} -c '
import json,sys
r=json.load(open(sys.argv[1]))
print(json.dumps({
  "audit_completed":r["passed"],
  "case_count":r["case_count"],
  "cases":[{
    "kind":x["kind"],
    "rank":x["rank"],
    "uid":x["object_uid"],
    "runtime_passed":x["model_runtime_passed"],
    "failure_count":x["failure_count"],
    "gt":x["gt_target_mesh"],
    "rerun_report":x["rerun_report"],
  } for x in r["cases"]],
  "scope_guard":r["scope_guard"],
},indent=2,ensure_ascii=False))
' "${AUDIT}/audit_report.json"

${PY} -c '
import hashlib,json,shutil,sys
from pathlib import Path

report_path=Path(sys.argv[1]).resolve(strict=True)
destination=Path(sys.argv[2]).resolve()
report=json.loads(report_path.read_text(encoding="utf-8"))
assert report["passed"] is True and report["case_count"] == 4
destination.mkdir(parents=True,exist_ok=True)

def sha256(path):
    digest=hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda:handle.read(1024*1024),b""):
            digest.update(chunk)
    return digest.hexdigest()

files=[]
for case in report["cases"]:
    case_dir=destination/case["label"]
    case_dir.mkdir(parents=True,exist_ok=True)
    sources=[("GT",None,Path(case["gt_target_mesh"]).resolve(strict=True))]
    for row in case["branch_records"]:
        if row.get("passed") is True and row.get("mesh"):
            sources.append((str(row["branch"]),int(row["seed"]),Path(row["mesh"]).resolve(strict=True)))
    for branch,seed,source in sources:
        filename="GT_target_mesh.obj" if branch == "GT" else f"seed{seed}_{branch}.obj"
        target=case_dir/filename
        shutil.copy2(source,target)
        files.append({
            "case":case["label"],"kind":case["kind"],"uid":case["object_uid"],
            "branch":branch,"seed":seed,"source":str(source),
            "destination":str(target),"sha256":sha256(target),
        })

shutil.copy2(report_path,destination/"源审计报告.json")
manifest={
    "format":"pose_point_depth_mv.native_ss_support_lcr_mesh_bundle.v1",
    "passed":True,"source_audit_report":str(report_path),
    "case_count":report["case_count"],"mesh_count":len(files),"files":files,
    "note":"原始/data审计产物保留；outputs2为人工检查副本。",
}
(destination/"Mesh汇总清单.json").write_text(
    json.dumps(manifest,indent=2,ensure_ascii=False)+"\n",encoding="utf-8"
)
print(json.dumps({
    "passed":True,"output":str(destination),
    "cases":report["case_count"],"meshes":len(files),
    "manifest":str(destination/"Mesh汇总清单.json"),
},indent=2,ensure_ascii=False))
' "${AUDIT}/audit_report.json" "${DEST}"

echo "生成Mesh位于：${AUDIT}/reruns/<case>/mesh_pairs/<pair>/<branch>/mesh.obj"
echo "GT Mesh位于：${AUDIT}/reruns/<case>/GT_target_mesh.obj"
echo "人工检查副本：${DEST}"
echo "current terminal remains open"
