# Objaverse 256-object pilot：009/012 marker 恢复、013 补跑与正式汇总
#
# 适用状态（2026-07-27）：
#   render root: pilot_objaverse256_cyclescuda_v2
#   已生成 manifest: 15/16（缺 shard_013）
#   已生成完成 marker: 13/16（缺 shard_009、012、013）
#   shard_009/012 的 raw manifest 把空 GLB 场景记为 other_failed；
#   canonical taxonomy 已将精确错误 "scene has no mesh geometry"
#   解释为 source_asset_rejected。009/012 会原地复核并补 marker，不重新渲染；
#   worker 1 随后只渲染尚未运行的 shard_013。
#
# 本文件不使用 exit 98，也不会关闭当前终端。首次恢复按 R0、R1、R2 顺序在同一
# shell 执行。R3、R4、R5 各自重新定义所需变量，可以在新的终端中独立执行。


###############################################################################
# R0. 固定变量与只读状态检查
###############################################################################

cd /home/zjr/Tracker

PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
WORK=/data/zjr/dataset10k_20260726
OBJ_SOURCE_FREEZE=${WORK}/source_freeze_objaverse_v1
OBJ_PILOT_ROOT=${WORK}/pilot_objaverse256_cyclescuda_v2
OBJ_PILOT_PREVIEW=/home/zjr/Tracker/pixal3d_multiview/outputs/data_previews/objaverse256_cyclescuda_v2
OBJ_PILOT_LOG_ROOT=${WORK}/logs/pilot_objaverse256_cyclescuda_v2
OBJ_PILOT_REPORT=${OBJ_PILOT_ROOT}/pilot_report.json
BLENDER=/tmp/blender-3.0.1-linux-x64/blender
BLENDER_ENGINE=CYCLES
BLENDER_CYCLES_DEVICE=CUDA

printf '完成 marker: '
find "${OBJ_PILOT_ROOT}/objaverse" -mindepth 2 -maxdepth 2 \
  -type f -name _WORKER_COMPLETE.json 2>/dev/null | wc -l

printf '已生成 manifest: '
find "${OBJ_PILOT_ROOT}/objaverse" -mindepth 2 -maxdepth 2 \
  -type f -name manifest.json 2>/dev/null | wc -l

for SHARD in 009 012 013; do
  if [ -s "${OBJ_PILOT_ROOT}/objaverse/shard_${SHARD}/_WORKER_COMPLETE.json" ]; then
    echo "shard_${SHARD}: complete"
  elif [ -s "${OBJ_PILOT_ROOT}/objaverse/shard_${SHARD}/manifest.json" ]; then
    echo "shard_${SHARD}: complete manifest without marker"
  else
    echo "shard_${SHARD}: not rendered"
  fi
done

nvidia-smi --query-gpu=index,name,memory.used,memory.total,utilization.gpu \
  --format=csv,noheader,nounits


###############################################################################
# R1. 恢复前硬门：测试、builder 冻结绑定、009/012 真实 inventory
###############################################################################

cd /home/zjr/Tracker

R1_RC=0

"${PY}" -m unittest \
  pose_point_depth_mv.test_mixed_multiview_render_guards
TEST_RC=$?
if [ "${TEST_RC}" -ne 0 ]; then
  echo "R1 blocked: regression tests failed"
  R1_RC=91
fi

"${PY}" -m py_compile \
  pose_point_depth_mv/dataset_tools/render_failure_taxonomy.py \
  pose_point_depth_mv/dataset_tools/run_mixed_multiview_render_worker.py \
  pose_point_depth_mv/dataset_tools/analyze_mixed_render_pilot.py
COMPILE_RC=$?
if [ "${COMPILE_RC}" -ne 0 ]; then
  echo "R1 blocked: py_compile failed"
  R1_RC=92
fi

CURRENT_BUILDER_SHA=$(
  "${PY}" -c \
    'import hashlib,sys; print(hashlib.sha256(open(sys.argv[1],"rb").read()).hexdigest())' \
    pose_point_depth_mv/dataset_tools/build_objaverse_multiview_sparse_data.py
)
FROZEN_BUILDER_SHA=$(
  "${PY}" -c \
    'import json,sys; print(json.load(open(sys.argv[1]))["code_bindings"]["dataset_builder"]["sha256"])' \
    "${OBJ_PILOT_ROOT}/objaverse/shard_000/manifest.json"
)

echo "current builder sha: ${CURRENT_BUILDER_SHA}"
echo "frozen  builder sha: ${FROZEN_BUILDER_SHA}"
if [ "${CURRENT_BUILDER_SHA}" != "${FROZEN_BUILDER_SHA}" ]; then
  echo "R1 blocked: builder changed; do not mix code versions in this pilot"
  R1_RC=93
fi

"${PY}" -c \
  'from pathlib import Path
from pose_point_depth_mv.dataset_tools.run_mixed_multiview_render_worker import load_plan,load_expected_objects,validate_render_inventory
plan_path=Path(__import__("sys").argv[1])
root=Path(__import__("sys").argv[2])
plan=load_plan(plan_path)
shards={int(row["index"]):row for row in plan["objaverse"]["shards"]}
failed=False
for index in (9,12):
    shard=shards[index]
    expected=load_expected_objects(plan_path.parent/shard["path"],shard,16)
    result=validate_render_inventory(root/f"shard_{index:03d}"/"manifest.json",expected,2)
    print(f"shard_{index:03d}:",result)
    failed = failed or not result[0]
raise SystemExit(3 if failed else 0)' \
  "${OBJ_SOURCE_FREEZE}/build_plan.json" \
  "${OBJ_PILOT_ROOT}/objaverse"
INVENTORY_RC=$?
if [ "${INVENTORY_RC}" -ne 0 ]; then
  echo "R1 blocked: shard_009/012 inventory is not recoverable"
  R1_RC=94
fi

echo "R1_RC=${R1_RC}"
echo "current terminal remains open"


###############################################################################
# R2. 启动恢复：worker 0 补 shard_012 marker；worker 1 补 009 并渲染 013
###############################################################################

cd /home/zjr/Tracker

# 默认 GPU 1/2。运行前以 R0 的 nvidia-smi 为准；若被占用，可覆盖：
# RECOVERY_GPU_LIST_CSV=3,4
RECOVERY_GPU_LIST_CSV=${RECOVERY_GPU_LIST_CSV:-1,2}
IFS=',' read -r -a RECOVERY_GPU_LIST <<<"${RECOVERY_GPU_LIST_CSV}"
R2_RC=0

if [ "${R1_RC:-97}" -ne 0 ]; then
  echo "R2 blocked: R1_RC=${R1_RC:-not_set}"
  R2_RC=95
elif [ "${#RECOVERY_GPU_LIST[@]}" -ne 2 ]; then
  echo "R2 blocked: exactly two GPU indices are required"
  R2_RC=96
elif [ ! -x "${BLENDER}" ]; then
  echo "R2 blocked: Blender is missing: ${BLENDER}"
  R2_RC=96
else
  for WORKER_INDEX in 0 1; do
    GPU_INDEX=${RECOVERY_GPU_LIST[${WORKER_INDEX}]}
    UNIT=tracker-mixed10k-objpilot256-recovery-w${WORKER_INDEX}-v2

    if systemctl --user is-active --quiet "${UNIT}"; then
      echo "already active: ${UNIT}"
      continue
    fi

    systemctl --user reset-failed "${UNIT}" >/dev/null 2>&1 || true
    systemd-run --user \
      --unit="${UNIT}" \
      --collect \
      --property=WorkingDirectory=/home/zjr/Tracker \
      /usr/bin/env \
        CUDA_VISIBLE_DEVICES="${GPU_INDEX}" \
        HF_HUB_OFFLINE=1 \
        TRANSFORMERS_OFFLINE=1 \
        PYOPENGL_PLATFORM=egl \
        ATTN_BACKEND=flash_attn \
        SPCONV_ALGO=native \
        MPLCONFIGDIR=/tmp/matplotlib \
        NUMBA_CACHE_DIR=/tmp/numba_cache \
        TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
        PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
      "${PY}" -u \
        pose_point_depth_mv/dataset_tools/run_mixed_multiview_render_worker.py \
          --source_plan "${OBJ_SOURCE_FREEZE}/build_plan.json" \
          --render_root "${OBJ_PILOT_ROOT}" \
          --preview_root "${OBJ_PILOT_PREVIEW}" \
          --log_root "${OBJ_PILOT_LOG_ROOT}" \
          --worker_index "${WORKER_INDEX}" \
          --num_workers 4 \
          --sources objaverse \
          --python "${PY}" \
          --blender_path "${BLENDER}" \
          --blender_engine "${BLENDER_ENGINE}" \
          --blender_samples 16 \
          --blender_bounds_tolerance 0.001 \
          --blender_cycles_device "${BLENDER_CYCLES_DEVICE}" \
          --seed 20260727 \
          --sequences_per_object 2 \
          --candidate_views 24 \
          --selected_views 8 \
          --vis_count_per_shard 2 \
          --max_objects_per_shard 16
    START_RC=$?
    echo "${UNIT}: systemd-run rc=${START_RC}, GPU=${GPU_INDEX}"
    if [ "${START_RC}" -ne 0 ]; then
      R2_RC=${START_RC}
    fi
  done
fi

echo "R2_RC=${R2_RC}"
echo "current terminal remains open"

# R4. 达到 16/16 后生成正式 pilot_report.json
###############################################################################

cd /home/zjr/Tracker

PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
OBJ_PILOT_ROOT=/data/zjr/dataset10k_20260726/pilot_objaverse256_cyclescuda_v2
OBJ_PILOT_REPORT=${OBJ_PILOT_ROOT}/pilot_report.json

PILOT_DONE=$(
  find "${OBJ_PILOT_ROOT}/objaverse" -mindepth 2 -maxdepth 2 \
    -type f -name _WORKER_COMPLETE.json 2>/dev/null | wc -l
)
R4_RC=0

if [ "${PILOT_DONE}" -ne 16 ]; then
  echo "R4 blocked: pilot is incomplete: ${PILOT_DONE}/16"
  R4_RC=96
elif [ -s "${OBJ_PILOT_REPORT}" ]; then
  echo "reuse existing pilot report: ${OBJ_PILOT_REPORT}"
  "${PY}" -c \
    'import json,sys; r=json.load(open(sys.argv[1])); print(json.dumps(r,indent=2,ensure_ascii=False)); raise SystemExit(0 if r.get("passed") is True else 3)' \
    "${OBJ_PILOT_REPORT}"
  R4_RC=$?
else
  "${PY}" -u \
    pose_point_depth_mv/dataset_tools/analyze_mixed_render_pilot.py \
      --render_root "${OBJ_PILOT_ROOT}" \
      --source objaverse \
      --expected_shards 16 \
      --sequences_per_object 2 \
      --max_low_texture_ratio 0.22 \
      --target_objects 6000 \
      --reserve_fraction 0.15 \
      --output "${OBJ_PILOT_REPORT}"
  R4_RC=$?
fi

echo "R4_RC=${R4_RC}"
echo "pilot report: ${OBJ_PILOT_REPORT}"
echo "current terminal remains open"
