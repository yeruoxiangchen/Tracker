#!/usr/bin/env bash
# Stock2 / Full2 / ReconViaGen Proper-Sim(3)纯形状补充评测（2026-08-10）
# ========================================================================
#
# 直接前台运行：
#   bash 'pose_point_depth_mv/Stock2_Full2_ReconViaGen_ProperSim3纯形状补充评测命令_20260810.txt'
#
# 在第93章与第94章现有输出的seed42共同16对象上比较：
#
#   ReconViaGen original = Stock SS + Stock SLat + Stock decoder
#   stock2               = Native-SS v2 + Stock SLat + Stock decoder
#   full2                = Native-SS v2 + Native-SLAT v2 + Stock decoder
#
# 同一报告保留raw_canonical和sim3_shape_only。Sim(3)轨道对每个预测独立执行
# GT-assisted proper isotropic Sim(3)，允许SO(3)旋转、平移和统一尺度，禁止reflection
# 及各向异性缩放。对齐采样与最终20K评分采样使用独立随机流。
#
# 这是已消费开发集上的post-hoc shape-only补充，formal=false；不能用于相机位姿、
# 绝对尺度、世界放置或AR精度主张。CPU前台运行，不使用GPU，预计约10--20分钟。

set -euo pipefail
cd /home/zjr/Tracker

PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
R93=/data/zjr/native_ss_stock_slat_mesh_transfer_20260802_v1/matched_t3_final32_seed42_external_baselines_v1/unified_native_stock_pixal_genrecon_geometry32_v2/report.json
R94=/data/zjr/native_slat_genrecon_v2_mixed1k_20260802_v1/val32_seed424344_step2000_ema_v2/report.json
OUT=/data/zjr/native_slat_genrecon_v2_mixed1k_20260802_v1/N11_stock2_full2_reconviagen_common16_proper_sim3_shape_v1
LOG=${OUT}.log

# 先执行静态编译和定向单元测试。
"${PY}" -m py_compile \
  pose_point_depth_mv/evaluate_stock2_full2_reconviagen_sim3_shape.py \
  pose_point_depth_mv/test_stock2_full2_reconviagen_sim3_shape.py

"${PY}" -m unittest \
  pose_point_depth_mv.test_stock2_full2_reconviagen_sim3_shape

# 正式Common16 Shape-only补充评测。日志放在OUT旁边，避免污染不可变输出目录。
"${PY}" -u -m \
  pose_point_depth_mv.evaluate_stock2_full2_reconviagen_sim3_shape \
  --chapter93_report "${R93}" \
  --chapter94_report "${R94}" \
  --output_dir "${OUT}" \
  --seed 42 \
  --expected_objects 16 \
  --candidate_samples 2000 \
  --alignment_samples 10000 \
  --candidate_iterations 12 \
  --final_iterations 50 \
  --surface_samples 20000 \
  --bootstrap_samples 10000 \
  --metric_seed 20260810 \
  --resume \
  2>&1 | tee "${LOG}"

test -s "${OUT}/report.json"
test -s "${OUT}/summary.txt"

"${PY}" - "${OUT}/report.json" <<'PY'
import json
import sys

report = json.load(open(sys.argv[1], encoding="utf-8"))
assert report["passed"] is True
assert report["formal"] is False
assert report["post_hoc"] is True
assert report["object_count"] == 16
assert report["record_count"] == 48

def compact(track):
    output = {}
    for name, comparison in track["comparisons"].items():
        metrics = comparison["metrics"]
        output[name] = {
            "chamfer_l1_improvement_mean": metrics["chamfer_l1"]["mean"],
            "chamfer_l1_improvement_median": metrics["chamfer_l1"]["median"],
            "chamfer_l1_win_rate": metrics["chamfer_l1"]["positive_rate"],
            "chamfer_l1_mean_95_ci": metrics["chamfer_l1"]["bootstrap_mean_95_ci"],
            "fscore_0p02_delta_mean": metrics["fscore_0p02"]["mean"],
            "normal_delta_mean": metrics["normal_consistency"]["mean"],
        }
    return output

print(json.dumps({
    "passed": report["passed"],
    "formal": report["formal"],
    "post_hoc": report["post_hoc"],
    "objects": report["object_count"],
    "records": report["record_count"],
    "raw_comparisons": compact(report["summary"]["tracks"]["raw_canonical"]),
    "sim3_comparisons": compact(report["summary"]["tracks"]["sim3_shape_only"]),
    "report": sys.argv[1],
}, ensure_ascii=False, indent=2))
PY

echo "Stock2 / Full2 / ReconViaGen proper-Sim(3) shape test finished: rc=0"

# 解释：
# 1. stock2_vs_reconviagen在raw和Sim(3)均明显为正：
#    stock2同时保留端到端canonical优势和纯形状优势。
# 2. raw为正但Sim(3)接近零：
#    原优势主要来自canonical pose/scale恢复，不能写成纯形状优势。
# 3. raw接近零但Sim(3)为正：
#    纯形状更好，但姿态或尺度误差掩盖了收益。
# 4. full2_vs_stock2的Sim(3) CI跨零：
#    Native-SLAT v2仍未建立相对Stock SLat的纯形状增益。
# 5. passed=true只代表输入绑定、对齐合同和评分完整，不代表质量比较自动通过。
