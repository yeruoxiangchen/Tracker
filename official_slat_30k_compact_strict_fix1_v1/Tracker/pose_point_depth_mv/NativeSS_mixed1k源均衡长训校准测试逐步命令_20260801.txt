# 本文件不是bash脚本。不要执行 `bash 本文件`。
# 按 A0、A1、T0...顺序，每次只复制一个完整的 (...) 命令块到终端。
#
# 冻结协议：
#   train：868 objects / 1417 sequences，保留自然来源比例并按object均衡采样；

# 排除失败SS64训练使用过的排序前64对象。checkpoint、CFG和final三组互不重叠。
# 长训从Stock初始化，不resume旧SS64 step500。T0默认用物理GPU2和GPU5执行双卡DDP；
# V0/V2/V4仍是单卡评估。复制T0/V0/V2/V4前仍需用nvidia-smi确认目标GPU空闲。

# T0：868对象、2000 optimizer updates的双卡DDP长训。
# world2*每卡grad_accum4=全局有效batch8；2000*8/868约18.4次对象曝光，
# 接近旧Direct SS的1000*16/897约17.8次，训练合同不因双卡而加倍。
# 可在复制前执行 `export GPUS=5,6` 覆盖默认物理GPU2,5；必须恰好提供两张卡。
(
  set -euo pipefail
  cd /home/zjr/Tracker
  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  TORCHRUN=/home/zjr/anaconda3/envs/reconviagen/bin/torchrun
  GPUS=${GPUS:-2,5}
  SPLIT=/data/zjr/native3d_condition_reviewed1k_inputs_20260731_v4/native_ss_sourcebalanced_split_seed20260801_v1
  TRAIN=${SPLIT}/train868_manifest.json
  ROOT=/data/zjr/native3d_condition_ss_mixed1k_20260801_v1
  OUT=${ROOT}/ss868_sourceholdout_seed42_v1
  LOG=${ROOT}/logs/T0_ss868_train.log

  test -s "${TRAIN}" || exit 96
  test -x "${TORCHRUN}" || exit 96
  [ "$(awk -F, '{print NF}' <<<"${GPUS}")" -eq 2 ] || exit 96
  mkdir -p "${ROOT}/logs"
  if [ -s "${OUT}/report.json" ]; then
    "${PY}" -c "import json; p=json.load(open('${OUT}/report.json')); assert p['passed'] is True and p['step']==2000; print('reuse complete T0')"
  elif [ -e "${OUT}" ]; then
    echo "T0 partial output exists; preserve it and prepare an explicit resume directory: ${OUT}"
    exit 98
  else
    set +e
    env \
      CUDA_VISIBLE_DEVICES="${GPUS}" \
      PYTHONUNBUFFERED=1 \
      HF_HUB_OFFLINE=1 \
      TRANSFORMERS_OFFLINE=1 \
      ATTN_BACKEND=flash_attn \
      SPCONV_ALGO=native \
      PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
      "${TORCHRUN}" --standalone --nproc_per_node=2 --module \
        pose_point_depth_mv.train_native_ss_genrecon \
        --cache_manifest "${TRAIN}" \
        --output_dir "${OUT}" \
        --max_steps 2000 \
        --save_every 250 \
        --log_every 20 \
        --grad_accum 4 \
        --num_workers 0 \
        --seed 42 \
        --lora_rank 8 \
        --lora_alpha 16 \
        --condition_channels 1024 \
        --new_lr 1e-4 \
        --lora_lr 3e-5 \
        --new_weight_decay 0.01 \
        --warmup_steps 40 \
        --ema_decay 0.9995 \
        --p_uncond 0.1 \
        --t_logit_mean 1 \
        --t_logit_std 1 \
        --min_condition_views 1 \
        --max_condition_views 16 \
        --amp_dtype bf16 \
        --gradient_checkpointing \
      2>&1 | tee "${LOG}"
    RC=${PIPESTATUS[0]}
    set -e
    echo "T0_RC=${RC}"
    [ "${RC}" -eq 0 ] || exit "${RC}"
  fi
)

# T1：训练完成合同与四个粗选checkpoint完整性检查。
(
  set -euo pipefail
  cd /home/zjr/Tracker
  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  RUN=/data/zjr/native3d_condition_ss_mixed1k_20260801_v1/ss868_sourceholdout_seed42_v1
  TRAIN=/data/zjr/native3d_condition_reviewed1k_inputs_20260731_v4/native_ss_sourcebalanced_split_seed20260801_v1/train868_manifest.json

  "${PY}" - "${RUN}" "${TRAIN}" <<'PY'
import hashlib, json, sys, torch
from pathlib import Path
from pose_point_depth_mv.native_ss_genrecon import validate_native_ss_genrecon_checkpoint
run, train = Path(sys.argv[1]), Path(sys.argv[2])
report = json.load(open(run / 'report.json'))
assert report['passed'] is True and report['step'] == 2000
distributed = report['model_summary']['optimization']['distributed']
assert distributed['world_size'] == 2
assert distributed['per_rank_grad_accum'] == 4
assert distributed['global_effective_batch'] == 8
assert report['global_micro_samples'] == 2000 * 8
assert report['data_identity']['object_count'] == 868
assert report['data_identity']['sample_count'] == 1417
assert report['data_identity']['manifest_sha256'] == hashlib.sha256(train.read_bytes()).hexdigest()
for step in (500, 1000, 1500, 2000):
    path = run / 'checkpoints' / f'step_{step:06d}.pt'
    checkpoint = torch.load(path, map_location='cpu')
    validate_native_ss_genrecon_checkpoint(checkpoint, pretrained='Stable-X/trellis-vggt-v0-2')
    assert checkpoint['step'] == step
    assert checkpoint['args']['grad_accum'] == 4
    assert checkpoint['model_summary']['optimization']['distributed'] == distributed
    assert checkpoint['data_identity'] == report['data_identity']
    assert checkpoint['ema']['updates'] == step
print(json.dumps({
    'step': report['step'],
    'objects': report['data_identity']['object_count'],
    'sequences': report['data_identity']['sample_count'],
    'distributed': distributed,
    'effective_exposures_per_object': report['global_micro_samples'] / 868,
    'ema': report['ema'],
}, indent=2))
print('T1 PASS')
PY
)
