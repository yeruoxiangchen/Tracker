run_logged() {
  LOG_PATH=$1
  shift
  set +e
  "$@" 2>&1 | tee "${LOG_PATH}"
  COMMAND_RC=${PIPESTATUS[0]}
  printf '%s\n' "${COMMAND_RC}" > "${LOG_PATH}.exit_code"
  return "${COMMAND_RC}"
}

# V1. 冻结正式 fresh-blind 协议（CPU，只运行一次）

# 冻结动作只读取 E4 对象身份以排除旧对象，不读取任何 fresh 对象的 Mesh 结果。

  "seeds":r["sampling"]["joint_seeds"],
  "expected_pairs":len(r["selected"])*len(r["sampling"]["joint_seeds"]),
  "excluded_objects":len(r["selection"]["excluded_object_uids"]),
},indent=2))' \

    --rescale_t 3.0 \
