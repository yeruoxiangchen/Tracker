# Native Full / Stock 同集 32 对象补测 Pixal3D 与 official GenRecon

# 使用方式：不要执行 `bash 本文件`。

# 每一段结束都会打印返回码，并保持当前终端打开。
#
# 数据边界：
# - 严格复用 T3 已评价 Native Full / Stock 的 32 个对象及原始顺序。
# - Native Full / Stock 绑定 T3 seed42 canonical Mesh。
# - Pixal3D 使用同一对象中最大 foreground-mask 的一张图。
# - official GenRecon 使用同一对象全部 2/4/8 posed views。
# - 外部基线只跑 seed42；不得与 T3 的 seed42/43/44 三种子均值直接拼接。
# - 这是回顾性 matched comparison，不是 untouched formal SOTA claim。

GENRECON_ROOT=/home/zjr/Tracker/GenRecon
GENRECON_CONFIG=${GENRECON_ROOT}/configs/pipelines/original.json
GENRECON_SS=${GENRECON_ROOT}/checkpoints/sparse_structure.pt
GENRECON_SHAPE=${GENRECON_ROOT}/checkpoints/shape_slat.pt

GENRECON_SS_DECODER=/home/zjr/.cache/huggingface/hub/models--microsoft--TRELLIS-image-large/snapshots/25e0d31ffbebe4b5a97464dd851910efc3002d96/ckpts/ss_dec_conv3d_16l8_fp16
GENRECON_SHAPE_DECODER=/home/zjr/.cache/huggingface/hub/models--microsoft--TRELLIS.2-4B/snapshots/af44b45f2e35a493886929c6d786e563ec68364d/ckpts/shape_dec_next_dc_f16c32_fp16

echo "BENCH_GPU=${BENCH_GPU} CASE_LIMIT=${CASE_LIMIT}"

###############################################################################
# Q0：CPU 静态检查，并冻结与 T3 完全同集的 32-object 协议
# 预计少数高面数 Omni GT 导出较慢；看到 1/32 ... 32/32 是正常进度。
###############################################################################

cd /home/zjr/Tracker
Q0_RC=0

"${PY}" -m py_compile \
  pose_point_depth_mv/prepare_native_ss_matched_baselines.py \
  pose_point_depth_mv/infer_official_genrecon_objects.py \
  pose_point_depth_mv/evaluate_native_ss_pixal_genrecon.py \
  GenRecon/genrecon/pipelines/images_to_3d.py \
  || Q0_RC=$?

if [ "${Q0_RC}" -eq 0 ]; then
  "${PY}" -m unittest pose_point_depth_mv.test_native_ss_pixal_genrecon \

fi

    "${SPLIT_AUDIT}" \

fi

if [ "${Q0_RC}" -eq 0 ]; then

echo "Q0_RC=${Q0_RC}"

###############################################################################
# Q1：Pixal3D official-final
# 每个 case 使用新进程，完成的 case 可安全复用；单卡串行，最多使用 1 张 GPU。
###############################################################################

cd /home/zjr/Tracker
Q1_RC=0

if [ ! -s "${PROTOCOL}" ]; then
  echo "Q1 blocked: run Q0 first"
  Q1_RC=96
else
  mapfile -t CASE_IDS < <(
    "${PY}" -c \
      'import json,sys
p=json.load(open(sys.argv[1])); assert len(p["cases"])==32
limit=int(sys.argv[2]); rows=p["cases"] if limit==0 else p["cases"][:limit]
for case in rows: print(case["case_id"])' \
      "${PROTOCOL}" "${CASE_LIMIT}"
  )
fi

if [ "${Q1_RC}" -eq 0 ]; then
  mkdir -p "${PIXAL_LOG_ROOT}"
  Q1_INDEX=0
  for CASE_ID in "${CASE_IDS[@]}"; do
    Q1_INDEX=$((Q1_INDEX + 1))
    case "${CASE_ID}" in
      *[!A-Za-z0-9._-]*)
        echo "Q1 unsafe frozen case ID: ${CASE_ID}"
        Q1_RC=97
        break
        ;;
    esac

    FREE_MIB=$(
      nvidia-smi -i "${BENCH_GPU}" \
        --query-gpu=memory.free \
        --format=csv,noheader,nounits 2>/dev/null | tr -d '[:space:]'
    )
    if ! [[ "${FREE_MIB}" =~ ^[0-9]+$ ]] || [ "${FREE_MIB}" -lt "${MIN_FREE_GPU_MIB}" ]; then
      echo "Q1 blocked before ${CASE_ID}: GPU ${BENCH_GPU} free=${FREE_MIB:-?} MiB"
      Q1_RC=95
      break
    fi

    printf -v CASE_NUMBER '%02d' "${Q1_INDEX}"
    CASE_LOG=${PIXAL_LOG_ROOT}/${CASE_NUMBER}_${CASE_ID}.log
    echo "[Q1] ${Q1_INDEX}/${#CASE_IDS[@]} ${CASE_ID} GPU=${BENCH_GPU}"

    CUDA_VISIBLE_DEVICES="${BENCH_GPU}" \
    ATTN_BACKEND=sdpa \
    SPARSE_ATTN_BACKEND=sdpa \
    HF_HUB_OFFLINE=1 \
    TRANSFORMERS_OFFLINE=1 \
    PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
    "${PIXAL_PY}" -u -m pose_point_depth_mv.compare_pixal3d_singleview_smoke \
      infer \
      --protocol "${PROTOCOL}" \
      --case_ids "${CASE_ID}" \
      --model_path "${PIXAL_SNAPSHOT}" \
      --naf_repo "${NAF_REPO}" \
      --naf_checkpoint "${NAF_CKPT}" \
      --naf_source_manifest "${NAF_SOURCE_MANIFEST}" \
      --device cuda \
      --low_vram \
      --skip_rembg \
      --camera_mode known_crop_fov \
      --resolution 1024 \
      --sampling_steps 12 \
      2>&1 | tee "${CASE_LOG}"
    CASE_RC=${PIPESTATUS[0]}
    printf '%s\n' "${CASE_RC}" > "${CASE_LOG}.exit_code"
    if [ "${CASE_RC}" -ne 0 ]; then
      Q1_RC=${CASE_RC}
      break
    fi
  done
fi

if [ -d "${PROTOCOL_ROOT}/pixal3d" ]; then
  PIXAL_DONE=$(find "${PROTOCOL_ROOT}/pixal3d" -type f -name result_official_postprocessed.json | wc -l)
else
  PIXAL_DONE=0
fi
echo "Q1_RC=${Q1_RC}"
echo "Pixal complete: ${PIXAL_DONE}/32"
echo "current terminal remains open"


###############################################################################
# Q2：official GenRecon 512 geometry
# 使用同一对象全部 posed views；Texture-SLAT 不改变几何，因此不运行 texture。
# decoder 与 DINO runtime assets 已验证在本地缓存，默认离线运行。
###############################################################################

cd /home/zjr/Tracker
Q2_RC=0

if [ ! -s "${PROTOCOL}" ]; then
  echo "Q2 blocked: run Q0 first"
  Q2_RC=96
else
  mapfile -t CASE_IDS < <(
    "${PY}" -c \
      'import json,sys
p=json.load(open(sys.argv[1])); assert len(p["cases"])==32
limit=int(sys.argv[2]); rows=p["cases"] if limit==0 else p["cases"][:limit]
for case in rows: print(case["case_id"])' \
      "${PROTOCOL}" "${CASE_LIMIT}"
  )
fi

if [ "${GENRECON_HF_OFFLINE}" = 1 ]; then
  GENRECON_HF_HUB_OFFLINE=1
  GENRECON_TRANSFORMERS_OFFLINE=1
else
  GENRECON_HF_HUB_OFFLINE=0
  GENRECON_TRANSFORMERS_OFFLINE=0
fi

if [ "${Q2_RC}" -eq 0 ]; then
  mkdir -p "${GENRECON_LOG_ROOT}"
  Q2_INDEX=0
  for CASE_ID in "${CASE_IDS[@]}"; do
    Q2_INDEX=$((Q2_INDEX + 1))
    case "${CASE_ID}" in
      *[!A-Za-z0-9._-]*)
        echo "Q2 unsafe frozen case ID: ${CASE_ID}"
        Q2_RC=97
        break
        ;;
    esac

    # Resume must not require an idle GPU merely to revisit immutable cases.
    # Q3 performs the full JSON/mesh SHA validation before scoring.
    if [ -s "${GENRECON_OUTPUT}/${CASE_ID}/result.json" ] && \
       [ -s "${GENRECON_OUTPUT}/${CASE_ID}/mesh_canonical.obj" ]; then
      echo "[Q2] ${Q2_INDEX}/${#CASE_IDS[@]} reuse complete ${CASE_ID}"
      continue
    fi

    FREE_MIB=$(
      nvidia-smi -i "${BENCH_GPU}" \
        --query-gpu=memory.free \
        --format=csv,noheader,nounits 2>/dev/null | tr -d '[:space:]'
    )
    if ! [[ "${FREE_MIB}" =~ ^[0-9]+$ ]] || [ "${FREE_MIB}" -lt "${MIN_FREE_GPU_MIB}" ]; then
      echo "Q2 blocked before ${CASE_ID}: GPU ${BENCH_GPU} free=${FREE_MIB:-?} MiB"
      Q2_RC=95
      break
    fi

    printf -v CASE_NUMBER '%02d' "${Q2_INDEX}"
    CASE_LOG=${GENRECON_LOG_ROOT}/${CASE_NUMBER}_${CASE_ID}.log
    echo "[Q2] ${Q2_INDEX}/${#CASE_IDS[@]} ${CASE_ID} GPU=${BENCH_GPU}"

    CUDA_VISIBLE_DEVICES="${BENCH_GPU}" \
    PYTHONPATH="${GENRECON_ROOT}:/home/zjr/Tracker" \
    PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
    HF_HUB_OFFLINE="${GENRECON_HF_HUB_OFFLINE}" \
    TRANSFORMERS_OFFLINE="${GENRECON_TRANSFORMERS_OFFLINE}" \
    "${GENRECON_PY}" -u -m pose_point_depth_mv.infer_official_genrecon_objects \
      --protocol "${PROTOCOL}" \
      --case_id "${CASE_ID}" \
      --output_root "${GENRECON_OUTPUT}" \
      --genrecon_root "${GENRECON_ROOT}" \
      --pipeline_config "${GENRECON_CONFIG}" \
      --ss_checkpoint "${GENRECON_SS}" \
      --shape_checkpoint "${GENRECON_SHAPE}" \
      --ss_train_config "${GENRECON_SS_CONFIG}" \
      --shape_train_config "${GENRECON_SHAPE_CONFIG}" \
      --dino_model "${GENRECON_DINO}" \
      --device cuda \
      --seed 42 \
      --occ_threshold -1.0 \
      --proj_batch_voxels 4096 \
      2>&1 | tee "${CASE_LOG}"
    CASE_RC=${PIPESTATUS[0]}
    printf '%s\n' "${CASE_RC}" > "${CASE_LOG}.exit_code"
    if [ "${CASE_RC}" -ne 0 ]; then
      Q2_RC=${CASE_RC}
      break
    fi
  done
fi

if [ -d "${GENRECON_OUTPUT}" ]; then
  GENRECON_DONE=$(find "${GENRECON_OUTPUT}" -type f -name result.json | wc -l)
else
  GENRECON_DONE=0
fi
echo "Q2_RC=${Q2_RC}"
echo "GenRecon complete: ${GENRECON_DONE}/32"
echo "current terminal remains open"


###############################################################################
# Q3：统一 canonical 几何评分（CPU）
# 必须先得到 Pixal3D 32/32 和 GenRecon 32/32。
###############################################################################

cd /home/zjr/Tracker
Q3_RC=0

if [ -d "${PROTOCOL_ROOT}/pixal3d" ]; then
  PIXAL_DONE=$(find "${PROTOCOL_ROOT}/pixal3d" -type f -name result_official_postprocessed.json | wc -l)
else
  PIXAL_DONE=0
fi
if [ -d "${GENRECON_OUTPUT}" ]; then
  GENRECON_DONE=$(find "${GENRECON_OUTPUT}" -type f -name result.json | wc -l)
else
  GENRECON_DONE=0
fi

if [ "${PIXAL_DONE}" -ne 32 ] || [ "${GENRECON_DONE}" -ne 32 ]; then
  echo "Q3 blocked: Pixal=${PIXAL_DONE}/32 GenRecon=${GENRECON_DONE}/32"
  Q3_RC=96
elif [ -s "${SCORE_OUT}/report.json" ] && [ -s "${SCORE_OUT}/summary.txt" ]; then
  echo "reuse immutable unified score: ${SCORE_OUT}"
elif [ -e "${SCORE_OUT}" ]; then
  echo "Q3 blocked: partial score output exists; preserve and inspect: ${SCORE_OUT}"
  Q3_RC=98
else
  "${PY}" -u -m pose_point_depth_mv.evaluate_native_ss_pixal_genrecon \
    --protocol "${PROTOCOL}" \
    --pixal_transform "${PIXAL_TRANSFORM}" \
    --genrecon_output_root "${GENRECON_OUTPUT}" \
    --output_dir "${SCORE_OUT}" \
    --surface_samples 20000 \
    --bootstrap_samples 10000 \
    --seed 20260802 \
    || Q3_RC=$?
fi

echo "Q3_RC=${Q3_RC}"
echo "score report: ${SCORE_OUT}/report.json"
echo "current terminal remains open"


###############################################################################
# Q4：完整性复核并显示最终摘要
###############################################################################

cd /home/zjr/Tracker
Q4_RC=0

if [ ! -s "${SCORE_OUT}/report.json" ]; then
  echo "Q4 blocked: run Q3 first"
  Q4_RC=96
else
  "${PY}" -c \
    'import json,sys
from pose_point_depth_mv.compare_pixal3d_singleview_smoke import canonical_sha256
p=sys.argv[1]; r=json.load(open(p)); body=dict(r); saved=body.pop("report_sha256")
assert canonical_sha256(body)==saved
assert r["passed"] is True and r["formal"] is False
assert r["object_count"]==32
assert r["methods"]==["native_full","stock","pixal3d_official","genrecon_official"]
assert r["matched_t3_surface_seed_replay"] is True
assert float(r["t3_metric_replay_max_abs"]) <= 1.0e-4
print(json.dumps({
 "passed":True,
 "formal":r["formal"],
 "objects":r["object_count"],
 "methods":r["methods"],
 "t3_metric_replay_max_abs":r["t3_metric_replay_max_abs"],
 "summary":r["summary"],
 "scope_guard":r["scope_guard"],
 "cross_domain_guard":r["cross_domain_guard"],
},indent=2,ensure_ascii=False))' \
    "${SCORE_OUT}/report.json" \
    || Q4_RC=$?
fi

if [ "${Q4_RC}" -eq 0 ]; then
  sed -n '1,180p' "${SCORE_OUT}/summary.txt"
fi

echo "Q4_RC=${Q4_RC}"
echo "report: ${SCORE_OUT}/report.json"
echo "current terminal remains open"
