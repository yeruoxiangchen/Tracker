###############################################################################
# P3. 归档安全、SHA256和video-to-scan对象ID映射硬门
###############################################################################

set +e
set +u
set +o pipefail
(
  set -euo pipefail
  cd /home/zjr/Tracker

  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  DATA_ROOT=/data/zjr/OmniObject3D
  OFFICIAL_ROOT=${DATA_ROOT}/raw/omniobject3d___OmniObject3D-New
  ARCHIVE_ROOT=${OFFICIAL_ROOT}/raw/videos_processed
  SCAN_ROOT=${DATA_ROOT}/raw_scans_extracted_omni215_v1/categories
  PILOT_ROOT=/data/zjr/omni_real_video_pilot_20260804_v1
  CATEGORY=canned_beverage
  ARCHIVE=${ARCHIVE_ROOT}/${CATEGORY}.tar.gz
  MEMBER_LIST=${PILOT_ROOT}/manifests/${CATEGORY}_archive_members.txt
  AUDIT=${PILOT_ROOT}/audit/${CATEGORY}_archive_scan_mapping.json

  test -s "${ARCHIVE}" || exit 96
  test -d "${SCAN_ROOT}/${CATEGORY}" || exit 96
  gzip -t "${ARCHIVE}"
  tar -tzf "${ARCHIVE}" > "${MEMBER_LIST}"

  "${PY}" - "${ARCHIVE}" "${SCAN_ROOT}/${CATEGORY}" "${AUDIT}" <<'PY'
import hashlib
import json
import re
import sys
import tarfile
from pathlib import Path, PurePosixPath

archive = Path(sys.argv[1]).resolve()
scan_category = Path(sys.argv[2]).resolve()
output = Path(sys.argv[3]).resolve()
category = archive.name.removesuffix(".tar.gz")
object_pattern = re.compile(rf"^{re.escape(category)}_[0-9]+$")

digest = hashlib.sha256()
with archive.open("rb") as handle:
    for block in iter(lambda: handle.read(1024 * 1024), b""):
        digest.update(block)

unsafe = []
video_ids = set()
member_count = 0
with tarfile.open(archive, "r:gz") as bundle:
    for member in bundle.getmembers():
        member_count += 1
        pure = PurePosixPath(member.name)
        if pure.is_absolute() or ".." in pure.parts:
            unsafe.append({"name": member.name, "reason": "path traversal"})
        if member.issym() or member.islnk() or member.isdev():
            unsafe.append({"name": member.name, "reason": "link/device member"})
        for part in pure.parts:
            if object_pattern.fullmatch(part):
                video_ids.add(part)

scan_ids = {
    path.name
    for path in scan_category.iterdir()
    if path.is_dir() and (path / "Scan" / "Scan.obj").is_file()
}
payload = {
    "format": "pose_point_depth_mv.omni_real_video_archive_audit.v1",
    "category": category,
    "archive": str(archive),
    "archive_bytes": archive.stat().st_size,
    "archive_sha256": digest.hexdigest(),
    "member_count": member_count,
    "unsafe_members": unsafe,
    "video_object_ids": sorted(video_ids),
    "scan_object_ids": sorted(scan_ids),
    "video_without_scan": sorted(video_ids - scan_ids),
    "scan_without_video": sorted(scan_ids - video_ids),
}
# 真实视频可以只覆盖scan池的子集。硬门只要求每个video object都有对应scan；
# scan_without_video保留为覆盖率信息，不应导致R1失败。
payload["video_scan_coverage"] = len(video_ids) / len(scan_ids) if scan_ids else 0.0
payload["passed"] = bool(video_ids) and not unsafe and not payload["video_without_scan"]
output.parent.mkdir(parents=True, exist_ok=True)
output.write_text(json.dumps(payload, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
print(json.dumps(payload, indent=2, ensure_ascii=False))
if not payload["passed"]:
    raise SystemExit(2)
PY

  sha256sum "${ARCHIVE}" "${MEMBER_LIST}" "${AUDIT}"
) 2>&1 | tee /data/zjr/omni_real_video_pilot_20260804_v1/logs/P3_archive_mapping_audit.log

P3_RC=${PIPESTATUS[0]}
echo "P3_RC=${P3_RC}"
if [ "${P3_RC}" -eq 0 ]; then
  echo "P3 complete; current terminal remains open"
else
  echo "P3 failed; inspect the audit; current terminal remains open"

  EXTRACT_DIR=${PILOT_ROOT}/extracted/${CATEGORY}

import collections
import json
import shutil
import subprocess
import sys
from pathlib import Path

root = Path(sys.argv[1]).resolve()
output = Path(sys.argv[2]).resolve()
video_ext = {".mp4", ".mov", ".avi", ".mkv", ".m4v"}
image_ext = {".jpg", ".jpeg", ".png", ".webp", ".bmp"}
files = sorted(path for path in root.rglob("*") if path.is_file() and not path.name.startswith("_EXTRACT_COMPLETE"))
videos = [path for path in files if path.suffix.lower() in video_ext]
images = [path for path in files if path.suffix.lower() in image_ext]
extension_counts = collections.Counter(path.suffix.lower() or "<none>" for path in files)
probes = []
if shutil.which("ffprobe"):
    for path in videos:
        result = subprocess.run(
            [
                "ffprobe", "-v", "error", "-show_entries",
                "format=duration,size:stream=index,codec_type,codec_name,width,height,avg_frame_rate,nb_frames",
                "-of", "json", str(path),
            ],
            text=True,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            check=False,
        )
        probes.append({
            "path": str(path),
            "returncode": result.returncode,
            "probe": json.loads(result.stdout) if result.returncode == 0 else None,
            "stderr": result.stderr[-2000:],
        })
payload = {
    "format": "pose_point_depth_mv.omni_real_video_media_inventory.v1",
    "root": str(root),
    "file_count": len(files),
    "video_count": len(videos),
    "image_count": len(images),
    "extension_counts": dict(sorted(extension_counts.items())),
    "video_paths": [str(path) for path in videos],
    "first_image_paths": [str(path) for path in images[:40]],
    "ffprobe": probes,
}
payload["passed"] = bool(videos or images) and all(row["returncode"] == 0 for row in probes)
output.parent.mkdir(parents=True, exist_ok=True)
output.write_text(json.dumps(payload, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
print(json.dumps(payload, indent=2, ensure_ascii=False))
if not payload["passed"]:
    raise SystemExit(2)
PY

  if command -v ffmpeg >/dev/null 2>&1; then
