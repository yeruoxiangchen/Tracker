# 修正此前 Pixal3D 对比误用了 decoded pre-remesh mesh.obj 的问题：

#      o_voxel.postprocess.to_glb(..., remesh=True, texture_size=4096)

#        不改 Pixal 尺度/形状，也不改其余三路。

# 在相邻 case 之间累计 CUDA 显存；已完成且绑定合法的 case 会在初始化
# Pixal3D pipeline 前直接复用。
#

# 会为指定对象生成一份新的 V4 step200 Direct-SLAT Mesh。旧 Pixal3D

# Pixal3D 官方 final GLB 需要重新执行 Pixal3D，因为旧结果没有保存
# o_voxel remesh 所需的 attr_volume/coords。
#
# GPU
# ===

#
#   export GPU=5
#

run_logged() {
  LOG_PATH=$1
  shift
  set +e
  "$@" 2>&1 | tee "${LOG_PATH}"
  COMMAND_RC=${PIPESTATUS[0]}
  printf '%s\n' "${COMMAND_RC}" > "${LOG_PATH}.exit_code"
  return "${COMMAND_RC}"
}

run_pixal_official_case_isolated() {
  PROTOCOL_PATH=$1
  LOG_ROOT=$2
  mkdir -p "${LOG_ROOT}"
  mapfile -t CASE_IDS < <(
    "${PY}" -c \
      'import json,sys
r=json.load(open(sys.argv[1]))
for row in r["cases"]:
 print(row["case_id"])' \
      "${PROTOCOL_PATH}"
  )
  if [ "${#CASE_IDS[@]}" -eq 0 ]; then
    echo "Pixal3D case isolation blocked: protocol has no cases"
    return 96
  fi
  CASE_INDEX=0
  for CASE_ID in "${CASE_IDS[@]}"; do
    CASE_INDEX=$((CASE_INDEX + 1))
    case "${CASE_ID}" in
      *[!A-Za-z0-9._-]*)
        echo "unsafe Pixal3D case ID: ${CASE_ID}"
        return 97
        ;;
    esac
    printf -v CASE_NUMBER '%03d' "${CASE_INDEX}"
    echo "Pixal3D isolated case ${CASE_INDEX}/${#CASE_IDS[@]}: ${CASE_ID}"
    run_logged "${LOG_ROOT}/${CASE_NUMBER}_${CASE_ID}.log" env \
      CUDA_VISIBLE_DEVICES="${GPU}" \
      ATTN_BACKEND=sdpa \
      SPARSE_ATTN_BACKEND=sdpa \
      HF_HUB_OFFLINE=1 \
      TRANSFORMERS_OFFLINE=1 \
      PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
      "${PIXAL_PY}" -u -m pose_point_depth_mv.compare_pixal3d_singleview_smoke \
        infer \
        --protocol "${PROTOCOL_PATH}" \
        --case_ids "${CASE_ID}" \
        --model_path "${PIXAL_SNAPSHOT}" \
        --naf_repo "${NAF_REPO}" \
        --naf_checkpoint "${NAF_CKPT}" \
        --naf_source_manifest "${NAF_SOURCE_MANIFEST}" \
        --device cuda \
        --low_vram \
        --skip_rembg \
        --camera_mode known_crop_fov \
        --resolution 1024 \
        --sampling_steps 12
    CASE_RC=$?
    if [ "${CASE_RC}" -ne 0 ]; then
      echo "Pixal3D isolated case failed: ${CASE_ID}, rc=${CASE_RC}"
      return "${CASE_RC}"
    fi
  done
  return 0
}
