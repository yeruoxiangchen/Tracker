Objaverse 5K：完成 shard 并行后处理与 Direct DINO-only Cache
================================================================

目标
----

1. 当前 5K Blender 渲染继续运行，不停止、不读取 active shard 半成品。
2. CPU watcher 只消费带有效 _WORKER_COMPLETE.json 的 shard，逐 shard 生成：
   draft A/B 后处理 -> point prior -> PointPose。
3. GPU watcher 只消费 _CPU_SHARD_READY.json，直接运行冻结 DINO；不导入、
   不加载、不执行 VGGT。
4. 每个 shard 独立、可恢复；合并时只写绝对引用，不复制大 pt 文件。

注意：自动 A/B 仍是扩量实验用 draft 语义筛选，不等同于人工 reviewed 数据冻结。
当前脚本保留旧 train/val/test 和 Objaverse16 的显式泄漏排除。

A1. 启动 CPU shard watcher（与当前 5K 渲染并行）
================================================

(
  set -euo pipefail
  cd /home/zjr/Tracker

  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  RENDER=/data/zjr/objaverse5k_single_subject_render512_20260810_v1
  WORK=/data/zjr/objaverse5k_direct_dino_cache_shards_20260811_v1
  LOG_ROOT=${WORK}/logs
  LOG=${LOG_ROOT}/cpu_shard_watcher.log
  UNIT=tracker-objaverse5k-cpu-cache-shards-v1

  mkdir -p "${LOG_ROOT}"

  if systemctl --user is-active --quiet "${UNIT}.service"; then
    echo "already active: ${UNIT}.service"
  else
    systemctl --user reset-failed "${UNIT}.service" >/dev/null 2>&1 || true
    systemd-run --user \
      --unit="${UNIT}" \
      --collect \
      --property=WorkingDirectory=/home/zjr/Tracker \
      --property="StandardOutput=append:${LOG}" \
      --property="StandardError=append:${LOG}" \
      /usr/bin/ionice -c 2 -n 7 \
      /usr/bin/nice -n 10 \
      "${PY}" -u -m \
        pose_point_depth_mv.dataset_tools.process_objaverse5k_cache_shards \
        --stage cpu \
        --render_root "${RENDER}" \
        --work_root "${WORK}" \
        --python "${PY}" \
        --exclude_manifest \
          /data/zjr/reviewed_mixed1k_semantic_object_ss_repaired_v1_20260730/train.json \
        --exclude_manifest \
          /data/zjr/reviewed_mixed1k_semantic_object_ss_repaired_v1_20260730/val.json \
        --exclude_manifest \
          /data/zjr/reviewed_mixed1k_semantic_object_ss_repaired_v1_20260730/test.json \
        --exclude_manifest \
          /data/zjr/objaverse16_no_vggt_mixed_20260810_v1/O0_frozen_objaverse_test16_v1.json \
        --audit_workers 8 \
        --prior_seed 42 \
        --log_every 10 \
        --expected_shards 16 \
        --poll_seconds 60 \
        --watch
  fi

  echo "unit: ${UNIT}.service"
  echo "log:  ${LOG}"
)


A2. 启动 Direct DINO-only GPU watcher
=====================================

# 先用 nvidia-smi 选择空闲卡；2026-08-11 烟测时 GPU 0 空闲。
# 如 GPU 0 被重新占用，只改 DINO_GPU，不需要重跑 A1。

(
  set -euo pipefail
  cd /home/zjr/Tracker

  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  WORK=/data/zjr/objaverse5k_direct_dino_cache_shards_20260811_v1
  LOG_ROOT=${WORK}/logs
  LOG=${LOG_ROOT}/direct_dino_shard_watcher.log
  UNIT=tracker-objaverse5k-direct-dino-shards-v1
  DINO_GPU=${DINO_GPU:-0}

  mkdir -p "${LOG_ROOT}"
  nvidia-smi -i "${DINO_GPU}" \
    --query-gpu=index,memory.free,memory.total,utilization.gpu \
    --format=csv,noheader,nounits

  if systemctl --user is-active --quiet "${UNIT}.service"; then
    echo "already active: ${UNIT}.service"
  else
    systemctl --user reset-failed "${UNIT}.service" >/dev/null 2>&1 || true
    systemd-run --user \
      --unit="${UNIT}" \
      --collect \
      --property=WorkingDirectory=/home/zjr/Tracker \
      --property="StandardOutput=append:${LOG}" \
      --property="StandardError=append:${LOG}" \
      /usr/bin/env \
        CUDA_VISIBLE_DEVICES="${DINO_GPU}" \
        HF_HUB_OFFLINE=1 \
        TRANSFORMERS_OFFLINE=1 \
        ATTN_BACKEND=flash_attn \
        SPCONV_ALGO=native \
        MPLCONFIGDIR=/tmp/matplotlib \
        NUMBA_CACHE_DIR=/tmp/numba_cache \
        TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
        PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
        "${PY}" -u -m \
          pose_point_depth_mv.dataset_tools.process_objaverse5k_cache_shards \
          --stage dino \
          --work_root "${WORK}" \
          --python "${PY}" \
          --device cuda \
          --dino_model dinov2_vitl14_reg \
          --ss_context_tokens 4096 \
          --log_every 10 \
          --expected_shards 16 \
          --poll_seconds 60 \
          --watch
  fi

  echo "GPU:  ${DINO_GPU}"
  echo "unit: ${UNIT}.service"
  echo "log:  ${LOG}"
)

A4. 将当前已经完成的 DINO shard 冻结成阶段训练 manifest
========================================================

# 每当 admitted shard 集合变化，必须换一个新的 SNAPSHOT 名字；不要覆盖旧快照。
# 例：若当前 DINO ready=6，则用 stage06_v1。

(
  set -euo pipefail
  cd /home/zjr/Tracker
  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  WORK=/data/zjr/objaverse5k_direct_dino_cache_shards_20260811_v1
  READY=$(find "${WORK}/shards" -name _DINO_ONLY_LIFTING_COMPLETE.json | wc -l)
  test "${READY}" -gt 0

  SNAPSHOT=stage06_v1       # 按 READY 修改；新 shard 加入后换新名字
  OUT=${WORK}/merged_${SNAPSHOT}

  "${PY}" -u -m \
    pose_point_depth_mv.dataset_tools.merge_dino_only_lifting_shards \
      --input_root "${WORK}" \
      --output_dir "${OUT}" \
      --expected_shards "${READY}"

  echo "training manifest: ${OUT}/lifting_manifest.json"
)
