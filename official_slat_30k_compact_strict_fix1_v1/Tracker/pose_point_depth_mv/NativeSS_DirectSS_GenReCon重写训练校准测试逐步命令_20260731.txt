# D0：静态编译和12项CPU合同测试。运行时间短，直接看终端输出。
(
  set -euo pipefail
  cd /home/zjr/Tracker
  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python

  "${PY}" -m py_compile \
    pose_point_depth_mv/native_ss_genrecon.py \
    pose_point_depth_mv/train_native_ss_genrecon.py \
    pose_point_depth_mv/evaluate_native_ss_genrecon.py \
    pose_point_depth_mv/test_native_ss_genrecon.py

  "${PY}" - <<'PY'
import inspect
import pose_point_depth_mv.test_native_ss_genrecon as tests

selected = [
    (name, fn)
    for name, fn in vars(tests).items()
    if name.startswith('test_') and inspect.isfunction(fn)
]
for name, fn in selected:
    fn()
    print('PASS', name)
assert len(selected) == 12, len(selected)
print('D0 PASS: 12 Native SS Direct-SS/GenReCon contract tests')
PY
)

# S0：训练8-object机制smoke，200 optimizer steps。长时间命令，终端和日志同时输出。
(
  set -euo pipefail
  cd /home/zjr/Tracker
  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  RUN_ROOT=/data/zjr/native3d_condition_ss_genrecon_20260731_v1
  TRAIN=/data/zjr/native3d_condition_reviewed1k_inputs_20260730_v3/lifting_train/manifest.json
  OUT=${RUN_ROOT}/ss8_seed42_v1
  LOG=${RUN_ROOT}/logs/S0_ss8_train.log

  test -s "${TRAIN}" || exit 96
  mkdir -p "${RUN_ROOT}/logs"
  if [ -s "${OUT}/report.json" ]; then
    "${PY}" -c "import json; p=json.load(open('${OUT}/report.json')); assert p['passed'] is True and p['step']==200, p; print('reuse complete S0:', '${OUT}')"
  elif [ -e "${OUT}" ]; then
    echo "S0 partial output exists; preserve and inspect: ${OUT}"
    exit 98
  else
    set +e
    env \
      CUDA_VISIBLE_DEVICES=5 \
      HF_HUB_OFFLINE=1 \
      TRANSFORMERS_OFFLINE=1 \
      ATTN_BACKEND=flash_attn \
      SPCONV_ALGO=native \
      PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
      "${PY}" -u -m pose_point_depth_mv.train_native_ss_genrecon \
        --cache_manifest "${TRAIN}" \
        --output_dir "${OUT}" \
        --max_objects 8 \
        --max_steps 200 \
        --save_every 50 \
        --log_every 5 \
        --grad_accum 8 \
        --num_workers 0 \
        --seed 42 \
        --lora_rank 8 \
        --lora_alpha 16 \
        --condition_channels 1024 \
        --new_lr 1e-4 \
        --lora_lr 3e-5 \
        --new_weight_decay 0.01 \
        --p_uncond 0.1 \
        --t_logit_mean 1 \
        --t_logit_std 1 \
        --min_condition_views 1 \
        --max_condition_views 16 \
        --amp_dtype bf16 \
        --gradient_checkpointing \
      2>&1 | tee "${LOG}"
    RC=${PIPESTATUS[0]}
    set -e
    echo "S0_RC=${RC}"
    [ "${RC}" -eq 0 ] || exit "${RC}"
  fi
  echo "S0 finished; report=${OUT}/report.json"
)

# S1：验证SS8 checkpoint合同，确认不存在Direct-SLAT字段。运行时间短。
(

  cd /home/zjr/Tracker
  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python

  "${PY}" - "${OUT}/report.json" "${OUT}/checkpoints/step_000200.pt" <<'PY'
import json, sys, torch
from pose_point_depth_mv.native_ss_genrecon import validate_native_ss_genrecon_checkpoint

report = json.load(open(sys.argv[1]))
checkpoint = torch.load(sys.argv[2], map_location='cpu')
validate_native_ss_genrecon_checkpoint(
    checkpoint,
    pretrained='Stable-X/trellis-vggt-v0-2',
)
summary = {
    'report_passed': report.get('passed'),
    'step': checkpoint.get('step'),
    'format': checkpoint.get('format'),
    'training_semantics': checkpoint['model_summary'].get('training_semantics'),
    'cfg_semantics': checkpoint['model_summary'].get('cfg_semantics'),
    'post_cfg_cap': checkpoint['model_summary'].get('post_cfg_cap'),
    'direct_slat_dependency': checkpoint['model_summary'].get('direct_slat_dependency'),
    'training_object_count': checkpoint['data_identity'].get('object_count'),
    'training_object_uids': checkpoint['data_identity'].get('object_uids'),
    'initial_stock_audit': report.get('initial_stock_audit'),
}
print(json.dumps(summary, indent=2, ensure_ascii=False))
assert report.get('passed') is True, summary
assert checkpoint.get('step') == 200, summary
assert summary['post_cfg_cap'] is False, summary
assert summary['direct_slat_dependency'] is False, summary
assert summary['training_object_count'] == 8, summary
assert len(summary['training_object_uids']) == 8, summary
assert report['initial_stock_audit']['passed'] is True, summary
print('S1 PASS')
PY
)
