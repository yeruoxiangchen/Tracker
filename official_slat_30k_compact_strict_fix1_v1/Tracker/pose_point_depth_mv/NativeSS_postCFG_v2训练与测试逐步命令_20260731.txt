Native SS post-CFG v2 训练与测试逐步命令
==========================================

本文档不是脚本。每次只复制一个完整的 (...) 命令块到终端。
按 P0 -> P11 顺序运行；任一硬门失败就停止。
本轮只验证 Native SS。不要启动 SS1K、Native SLAT 或 Mesh。
所有输出写入新目录，不复用 20260730 的 v1 训练结果。

P2. Native SS8 post-CFG v2 训练（GPU 5）
========================================

(
  set -uo pipefail
  cd /home/zjr/Tracker
  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  LIFT=/data/zjr/native3d_condition_reviewed1k_inputs_20260730_v3/lifting_train/manifest.json
  ROOT=/data/zjr/native3d_condition_ss_postcfg_20260731_v2
  OUT=${ROOT}/ss8_overfit_seed42_v1
  LOG=${ROOT}/logs/P2_SS8_train.log

  test -s "${LIFT}" || exit 96
  if [ -s "${OUT}/report.json" ]; then
    echo "reuse complete P2: ${OUT}"
  elif [ -e "${OUT}" ]; then
    echo "P2 partial output exists; preserve and inspect: ${OUT}"
    exit 98
  else
    mkdir -p "$(dirname "${LOG}")"
    env \
      CUDA_VISIBLE_DEVICES=5 \
      HF_HUB_OFFLINE=1 \
      TRANSFORMERS_OFFLINE=1 \
      ATTN_BACKEND=flash_attn \
      SPCONV_ALGO=native \
      PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
      "${PY}" -u -m pose_point_depth_mv.train_native_ss_flow \
        --cache_manifest "${LIFT}" \
        --output_dir "${OUT}" \
        --pretrained Stable-X/trellis-vggt-v0-2 \
        --indices all \
        --max_objects 8 \
        --max_steps 200 \
        --save_every 200 \
        --log_every 5 \
        --grad_accum 1 \
        --seed 42 \
        --lr 1e-4 \
        --weight_decay 0 \
        --grad_clip 1.0 \
        --hidden_dim 128 \
        --feature_source dino \
        --condition_scale 1.0 \
        --training_semantics post_cfg_bounded_v2 \
        --guided_delta_policy post_cfg_v2 \
        --support_interval_policy cfg_active_only_v1 \
        --delta_bound_mode smooth_rms_v1 \
        --train_cfg_strength 5.0 \
        --train_cfg_interval 0.5,1.0 \
        --native_steps 25 \
        --rescale_t 3.0 \
        --guidance_rescale 0 \
        --delta_scale 1.0 \
        --delta_rms_ratio_cap 0.10 \
        --raw_delta_excess_weight 0.10 \
        --delta_norm_weight 1e-4 \
        --t_schedule native_cfg_active \
        --control_mode pose_cyclic1 \
        --control_probability 0.25 \
        --control_rank_weight 0 \
        --amp_dtype bf16 \
      2>&1 | tee "${LOG}"
    RC=${PIPESTATUS[0]}
    printf '%s\n' "${RC}" > "${LOG}.exit_code"
    echo "P2_RC=${RC}"
    exit "${RC}"
  fi
  echo 'current terminal remains open'
)
