    return float(statistics.fmean(values))

    assert {r["object_uid"] for r in rows} == set(d1["protocol"]["object_uids"])

            float(r["full_stock_count_ratio"])
            for r in selected
            if r["full_stock_count_ratio"] is not None

            "object_count": len(selected),

        "gap_objaverse": 10,
        "pilot_objaverse": 9,
        "omni": 9,

        name: {
            "mean": float(candidate["summary"][name]["mean"]),
            "median": float(candidate["summary"][name]["median"]),
            "positive_rate": float(candidate["summary"][name]["positive_rate"]),
            "bootstrap_mean_95_ci": candidate["summary"][name]["bootstrap_mean_95_ci"],
        }
        for name in metrics
    }

        "full_stock_count_ratio"
    ]
    return {"overall": overall, "by_source": by_group, "checks": checks}

candidates = {float(c["cfg_strength"]): c for c in d1["candidates"]}

        "checkpoint_step": 500,
        "weights": "ema",
        "object_count": 48,
        "joint_seeds": [42, 43, 44],
        "condition_scale_policy": "learned_projection_only",
        "post_cfg_cap": False,
    },
    "cfg": {
        "1": summarize_candidate(candidates[1.0], candidates[1.0]["checks"]),
        "3": summarize_candidate(candidates[3.0], candidates[3.0]["checks"]),

        f"{row['recall_gain']['mean']:+.8f} "

        f"{row['full_stock_count_ratio']['mean']:.8f} "

        row = result["cfg"][cfg]["by_source"][group]

            f"{row['latent_mse_gain']:+.8f} {row['full_stock_count_ratio']:.8f}"

def mean(values):
    return float(statistics.fmean(values))

def metric_mean(rows, name):
    values = [
        float(row[name]) for row in rows
        if row.get(name) is not None
    ]
    return mean(values) if values else None

def summarize(candidate):
    rows = candidate["object_rows"]
    by_group = {}
    for group in groups:
        selected = [r for r in rows if classify(path_by_uid[r["object_uid"]]) == group]
        by_group[group] = {
            "object_count": len(selected),
            **{name: metric_mean(selected, name) for name in metrics},
            "full_stock_count_ratio_defined_object_count": sum(
                r["full_stock_count_ratio"] is not None for r in selected
            ),
        }
    assert [by_group[g]["object_count"] for g in groups] == [20, 10, 9, 9]
    return {
        "overall": {
            **{name: float(candidate["summary"][name]["mean"]) for name in metrics[:-1]},
            "full_stock_count_ratio": float(
                candidate["count_summary"]["full_stock_count_ratio"]["mean"]
            ),
            "iou_object_win_rate": float(candidate["summary"]["iou_gain"]["positive_rate"]),
        },
        "by_source": by_group,

import json
import statistics
import sys
from pathlib import Path

pointpose_path, output_path, *specs = sys.argv[1:]
reports = {}
for spec in specs:
    step_text, path = spec.split(":", 1)
    reports[int(step_text)] = json.load(open(path, encoding="utf-8"))
steps = [100,200,300,400,500]
assert sorted(reports) == steps

path_by_uid = {}
for row in json.load(open(pointpose_path, encoding="utf-8"))["samples"]:
    path_by_uid.setdefault(str(row["object_uid"]), str(row["image_paths"][0]))

def classify(path):
    for token, name in (
        ("objaverse_sparse_mv_artraj_pbr_5000_v9_select8", "legacy_objaverse"),
        ("gap_objaverse288_cyclescuda_v1", "gap_objaverse"),
        ("pilot_objaverse256_cyclescuda_v2", "pilot_objaverse"),
        ("pilot_omni256_cyclescuda_v4", "omni"),
    ):
        if token in path:
            return name
    raise AssertionError(f"unknown render source: {path}")

groups = ("legacy_objaverse", "gap_objaverse", "pilot_objaverse", "omni")
metrics = ("iou_gain", "precision_gain", "recall_gain", "latent_mse_gain", "full_stock_count_ratio")
reference_hash = None
trajectory = {}
for step in steps:
    report = reports[step]
    q = report["protocol"]
    assert q["checkpoint_step"] == step and q["weights"] == "ema", q
    assert q["object_start"] == 16 and q["object_end"] == 64, q
    assert q["joint_seeds"] == [42,43,44] and q["steps"] == 25, q
    assert q["cfg_interval"] == [0.5,1.0] and q["guidance_rescale"] == 0.0, q
    assert q["rescale_t"] == 3.0 and q["amp_dtype"] == "bf16", q
    assert q["condition_scale_policy"] == "learned_projection_only", q
    assert q["post_cfg_cap"] is False, q
    reference_hash = reference_hash or q["object_uid_hash"]
    assert q["object_uid_hash"] == reference_hash
    candidates = {float(c["cfg_strength"]): c for c in report["candidates"]}
    assert set(candidates) == ({1.0, 3.0} if step == 500 else {1.0}), candidates
    candidate = candidates[1.0]
    assert candidate["object_count"] == 48
    rows = candidate["object_rows"]
    by_source = {}
    for group in groups:
        selected = [r for r in rows if classify(path_by_uid[r["object_uid"]]) == group]
        def metric_mean(name):
            values = [
                float(r[name]) for r in selected
                if r.get(name) is not None
            ]
            return float(statistics.fmean(values)) if values else None
        by_source[group] = {
            "object_count": len(selected),
            **{name: metric_mean(name) for name in metrics},
            "full_stock_count_ratio_defined_object_count": sum(
                r["full_stock_count_ratio"] is not None for r in selected
            ),
        }
    assert [by_source[g]["object_count"] for g in groups] == [20,10,9,9]
    trajectory[str(step)] = {
        "overall": {
            "iou_gain": float(candidate["summary"]["iou_gain"]["mean"]),
            "iou_object_win_rate": float(candidate["summary"]["iou_gain"]["positive_rate"]),
            "precision_gain": float(candidate["summary"]["precision_gain"]["mean"]),
            "recall_gain": float(candidate["summary"]["recall_gain"]["mean"]),
            "latent_mse_gain": float(candidate["summary"]["latent_mse_gain"]["mean"]),
            "full_stock_count_ratio": float(
                candidate["count_summary"]["full_stock_count_ratio"]["mean"]
            ),
        },
        "by_source": by_source,
        "checks": candidate["checks"],
    }

best_iou = max(trajectory, key=lambda s: trajectory[s]["overall"]["iou_gain"])
best_recall = max(trajectory, key=lambda s: trajectory[s]["overall"]["recall_gain"])
first_latent_positive = next(
    (s for s in map(str, steps) if trajectory[s]["overall"]["latent_mse_gain"] > 0), None
)
first_contracted = next(
    (s for s in map(str, steps)
     if trajectory[s]["overall"]["full_stock_count_ratio"] < 0.85), None
)
result = {
    "contract": {
        "diagnostic_only_s3_consumed": True,
        "steps": steps,
        "weights": "ema",
        "cfg_strength": 1.0,
        "object_count": 48,
        "joint_seeds": [42,43,44],
        "condition_scale_policy": "learned_projection_only",
        "post_cfg_cap": False,
    },
    "trajectory": trajectory,
    "diagnostic": {
        "best_iou_step": int(best_iou),
        "best_recall_step": int(best_recall),
        "first_positive_latent_gain_step": (
            None if first_latent_positive is None else int(first_latent_positive)
        ),
        "first_mean_count_ratio_below_0_85": (
            None if first_contracted is None else int(first_contracted)
        ),
    },
}
Path(output_path).write_text(json.dumps(result, indent=2) + "\n", encoding="utf-8")

print("EMA CFG=1 full checkpoint trajectory")
print("step IoU_gain    Recall_gain Latent_gain Count_ratio IoU_win")
for step in steps:
    r = trajectory[str(step)]["overall"]
    print(
        f"{step:4d} {r['iou_gain']:+.8f} {r['recall_gain']:+.8f} "
        f"{r['latent_mse_gain']:+.8f} {r['full_stock_count_ratio']:.8f} "
        f"{r['iou_object_win_rate']:.6f}"
    )
print("diagnostic:", json.dumps(result["diagnostic"], ensure_ascii=False))
for step in steps:
    print(f"\nstep={step} by source")
    print("source             n  IoU_gain    Recall_gain Latent_gain Count_ratio")
    for group in groups:
        r = trajectory[str(step)]["by_source"][group]
        print(
            f"{group:18s} {r['object_count']:2d} {r['iou_gain']:+.8f} "
            f"{r['recall_gain']:+.8f} {r['latent_mse_gain']:+.8f} "
            f"{r['full_stock_count_ratio']:.8f}"
        )
print("D10 PASS; summary:", output_path)
PY
  fi
)
