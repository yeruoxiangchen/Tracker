val_contract = validate_genrecon_cache_contract(
    val, training_config_hash=train.config_hash
)
for label, dataset in (('train', train), ('val', val)):
    for index in range(len(dataset)):
        dataset[index]
        if (index + 1) % 100 == 0:
            print(label, index + 1, '/', len(dataset), flush=True)

val_uids = {str(row['object_uid']) for row in val.rows}

    'train_objects': len(train_uids),
    'val_samples': len(val),
    'val_objects': len(val_uids),
