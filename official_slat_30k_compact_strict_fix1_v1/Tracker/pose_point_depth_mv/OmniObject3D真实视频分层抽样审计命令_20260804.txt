# Q1. 固定6对象分层样本：安全抽取、自动合同审计、RGB-mask预览
###############################################################################

set +e
set +u
set +o pipefail
(
  set -euo pipefail
  cd /home/zjr/Tracker

  PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
  ROOT=/data/zjr/omni_real_video_pilot_20260804_v1
  ARCHIVES=/data/zjr/OmniObject3D/raw/omniobject3d___OmniObject3D-New/raw/videos_processed
  SCANS=/data/zjr/OmniObject3D/raw_scans_extracted_omni215_v1/categories
  EXTRACT=${ROOT}/sample_extract_stratified6_v1
  OUT=${ROOT}/sample_audit_stratified6_v1
  LOG=${ROOT}/logs/Q1_stratified6_sample_audit.log

  if [ -s "${OUT}/sample_audit.json" ]; then
    "${PY}" -c \
      'import json,sys; p=json.load(open(sys.argv[1])); assert p["format"]=="pose_point_depth_mv.omni_real_video_sample_audit.v1"; assert p["sample_count"]==6; assert p["automatic_passed"] is True; print(json.dumps(p,indent=2,ensure_ascii=False))' \
      "${OUT}/sample_audit.json"
  elif [ -e "${OUT}" ]; then
    echo "Q1 partial output exists; preserve and inspect: ${OUT}"
    exit 98
  else
    mkdir -p "$(dirname "${LOG}")"
    "${PY}" -u \
      pose_point_depth_mv/dataset_tools/audit_omni_real_video_sample.py \
      --archive_root "${ARCHIVES}" \
      --scan_root "${SCANS}" \
      --extract_root "${EXTRACT}" \
      --output_dir "${OUT}" \
      --sample canned_beverage:canned_beverage_001 \
      --sample fork:fork_001 \
      --sample guitar:guitar_001 \
      --sample laptop:laptop_001 \
      --sample tvstand:tvstand_001 \
      --sample boxed_beverage:boxed_beverage_001 \
      --min_frames 32 \
      --min_registration_ratio 0.70 \
      --min_point_count 100 \
      --preview_frames 12
  fi
) 2>&1 | tee /data/zjr/omni_real_video_pilot_20260804_v1/logs/Q1_stratified6_sample_audit.log
