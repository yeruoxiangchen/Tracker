# Objaverse16：旧 synthetic1k / mixed / ReconViaGen Stock 的 SLat-only 交换测试
#
# 目的：去掉 SS 差异，固定同一份 64^3 oracle support、seed42、坐标键控初始噪声、
# 25步/CFG5和冻结 Mesh decoder，只比较三路 SLat。
#
# 注意：oracle support 来自冻结 GT SS target_coords，因此这是 SLat 阶段诊断，不是部署指标，
# formal=false。完整流水线结论仍以 O11/O9 为旁证。
#
# 运行（前台；GPU默认4）：
#   GPU=4 bash 'pose_point_depth_mv/Objaverse16旧Mixed与ReconViaGen_SLat固定SS交换测试命令_20260810.txt'

set -euo pipefail
cd /home/zjr/Tracker

PY=/home/zjr/anaconda3/envs/reconviagen/bin/python
GPU=${GPU:-4}
RUN=/data/zjr/objaverse16_no_vggt_mixed_20260810_v1
OUT=${RUN}/O12_slat_oracle_support_swap16_seed42_v1

mkdir -p "${OUT}"

CUDA_VISIBLE_DEVICES="${GPU}" \
HF_HUB_OFFLINE=1 \
TRANSFORMERS_OFFLINE=1 \
ATTN_BACKEND=flash_attn \
SPCONV_ALGO=native \
TORCH_EXTENSIONS_DIR=/tmp/torch_extensions \
NUMBA_CACHE_DIR=/tmp/numba_cache \
MPLCONFIGDIR=/tmp/matplotlib \
PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
TORCH_CUDA_ARCH_LIST=8.6 \
"${PY}" -u -m pose_point_depth_mv.evaluate_objaverse16_slat_oracle_swap \
  --selection_manifest "${RUN}/O0_frozen_objaverse_test16_v1.json" \
  --stock_lifting_manifest "${RUN}/O3_lifting_full_v1/manifest.json" \
  --model_input_manifest "${RUN}/O5_model_inputs_target_free_v1/model_input_manifest.json" \
  --synthetic_inference_manifest "${RUN}/O10_synthetic1k_no_vggt_seed42_v1/inference_manifest.json" \
  --mixed_inference_manifest "${RUN}/O6_native_no_vggt_mixed_seed42_v1/inference_manifest.json" \
  --existing_o11_report "${RUN}/O11_synthetic1k_current_reconviagen_axisfixed_20k_v1/report.json" \
  --output_dir "${OUT}" \
  --pretrained Stable-X/trellis-vggt-v0-2 \
  --seed 42 \
  --surface_samples 20000 \
  --fscore_thresholds 0.01,0.02,0.05 \
  --device cuda \
  --amp_dtype bf16 \
  --resume \
  2>&1 | tee "${OUT}/run.log"

test -s "${OUT}/report.json"
test -s "${OUT}/summary.txt"

"${PY}" - "${OUT}/report.json" <<'PY'
import json
import sys

report = json.load(open(sys.argv[1], encoding="utf-8"))
assert report["passed"] is True
assert report["formal"] is False
assert report["object_count"] == 16
assert report["record_count"] == 48
print(json.dumps({
    "passed": report["passed"],
    "objects": report["object_count"],
    "records": report["record_count"],
    "decisions": {
        name: value["decision"]
        for name, value in report["paired_comparisons"].items()
    },
    "report": sys.argv[1],
}, ensure_ascii=False, indent=2))
PY

echo "Objaverse16 SLat oracle-support swap finished: rc=0"
